1//! The relay actor.
2//!
3//! The [`RelayActor`] handles all the relay connections.  It is helped by the
4//! [`ActiveRelayActor`] which handles a single relay connection.
5//!
6//! - The [`RelayActor`] manages all connections to relay servers.
7//!   - It starts a new [`ActiveRelayActor`] for each relay server needed.
8//!   - The [`ActiveRelayActor`] will exit when unused.
9//!     - Unless it is for the home relay, this one never exits.
10//!   - Each [`ActiveRelayActor`] uses a relay [`Client`].
11//!     - The relay [`Client`] is a `Stream` and `Sink` directly connected to the
12//!       `TcpStream` connected to the relay server.
13//!   - Each [`ActiveRelayActor`] will try and maintain a connection with the relay server.
14//!     - If connections fail, exponential backoff is used for reconnections.
15//! - When `AsyncUdpSocket` needs to send datagrams:
16//!   - It puts them on a queue to the [`RelayActor`].
17//!   - The [`RelayActor`] ensures the correct [`ActiveRelayActor`] is running and
18//!     forwards datagrams to it.
19//!   - The ActiveRelayActor sends datagrams directly to the relay server.
20//! - The relay receive path is:
21//!   - Whenever [`ActiveRelayActor`] is connected it reads from the underlying `TcpStream`.
22//!   - Received datagrams are placed on an mpsc channel that now bypasses the
23//!     [`RelayActor`] and goes straight to the `AsyncUpdSocket` interface.
24//!
25//! [`Client`]: krikos_relay::client::Client

163    #[error("active-relay actor accounting exhausted")]
164    CounterExhausted,

171impl From<AdmissionError> for ActiveRelayAdmissionError {
172    fn from(error: AdmissionError) -> Self {
173        match error {
174            AdmissionError::CapacityFull => e!(Self::CapacityFull),
175            AdmissionError::CounterExhausted => e!(Self::CounterExhausted),
176        }
177    }

210    pub(super) async fn run(
211        mut self,
212        mut receiver: mpsc::Receiver<RelayActorMessage>,
213        mut datagram_send_channel: mpsc::Receiver<RelaySendItem>,
214    ) {
215        #[cfg(not(wasm_browser))]
216        if let Some(url) = self.config.initial_relay.clone() {
217            self.config
218                .my_relay
219                .set(url.clone(), RelayConnectionState::Connecting);
220            self.set_home_relay(url).await;
221        }
222
223        // When this future is present, it is sending pending datagrams to an
224        // ActiveRelayActor.  We can not process further datagrams during this time.
225        let mut datagram_send_fut = std::pin::pin!(MaybeFuture::None);
226
227        loop {
228            #[cfg(wasm_browser)]
229            let has_active_relay_tasks = !self.active_relay_tasks.is_empty();
230            #[cfg(wasm_browser)]
231            let active_relay_completion = self.active_relay_tasks.join_next();
232            #[cfg(not(wasm_browser))]
233            let active_relay_completion = self.active_relay_completions_rx.recv();
234            #[cfg(not(wasm_browser))]
235            let has_active_relay_tasks = self.active_relay_task_count > 0;
236
237            tokio::select! {
238                biased;
239                _ = self.cancel_token.cancelled() => {
240                    debug!("shutting down");
241                    break;
242                }
243                completion = active_relay_completion, if has_active_relay_tasks => {
244                    #[cfg(wasm_browser)]
245                    if let Some(res) = completion {
246                        match res {
247                            Ok(permit) => drop(permit),
248                            Err(err) if err.is_panic() => {
249                                error!("ActiveRelayActor task panicked: {err:#?}");
250                            }
251                            Err(err) if err.is_cancelled() => {
252                                error!("ActiveRelayActor cancelled: {err:#?}");
253                            }
254                            Err(err) => error!("ActiveRelayActor failed: {err:#?}"),
255                        }
256                    }
257                    #[cfg(not(wasm_browser))]
258                    if let Some(completion) = completion {
259                        self.active_relay_task_count = self
260                            .active_relay_task_count
261                            .checked_sub(1)
262                            .expect("active-relay task count must not underflow");
263                        match completion {
264                            ActiveRelayCompletion::Finished(permit) => drop(permit),
265                            ActiveRelayCompletion::Panicked(permit) => {
266                                drop(permit);
267                                error!("ActiveRelayActor task panicked");
268                            }
269                        }
270                    }
271                    self.reap_active_relays();
272                }
273                msg = receiver.recv() => {
274                    let Some(msg) = msg else {
275                        debug!("Inbox dropped, shutting down.");
276                        break;
277                    };
278                    let cancel_token = self.cancel_token.child_token();
279                    cancel_token.run_until_cancelled(self.handle_msg(msg)).await;
280                }
281                // Only poll for new datagrams if we are not blocked on sending them.
282                item = datagram_send_channel.recv(), if datagram_send_fut.is_none() => {
283                    let Some(item) = item else {
284                        debug!("Datagram send channel dropped, shutting down.");
285                        break;
286                    };
287                    let token = self.cancel_token.child_token();
288                    if let Some(Some(fut)) = token.run_until_cancelled(
289                        self.try_send_datagram(item)
290                    ).await {
291                        datagram_send_fut.as_mut().set_future(fut);
292                    }
293                }
294                // Only poll this future if it is in use.
295                _ = &mut datagram_send_fut, if datagram_send_fut.is_some() => {
296                    datagram_send_fut.as_mut().set_none();
297                }
298            }
299        }
300
301        // try shutdown
302        #[cfg(wasm_browser)]
303        if time::timeout(Duration::from_secs(3), self.close_all_active_relays())
304            .await
305            .is_err()
306        {
307            warn!("Failed to shut down all ActiveRelayActors");
308        }
309        #[cfg(not(wasm_browser))]
310        {
311            let runtime = self.runtime.clone();
312            let clock = runtime.context().clock();
313            match RuntimeTimeout::after(
314                clock,
315                Duration::from_secs(3),
316                self.close_all_active_relays(),
317            ) {
318                Ok(timeout) => match timeout.await {
319                    Ok(()) => {}
320                    Err(krikos_runtime::TimeoutError::Elapsed) => {
321                        warn!("Failed to shut down all ActiveRelayActors");
322                    }
323                    Err(krikos_runtime::TimeoutError::Clock(error)) => {
324                        runtime.latch_failure(error.to_string());
325                    }
326                },
327                Err(error) => runtime.latch_failure(error.to_string()),
328            }
329        }
330    }

332    async fn handle_msg(&mut self, msg: RelayActorMessage) {
333        match msg {
334            RelayActorMessage::NetworkChange { report } => {
335                self.on_network_change(report).await;
336            }
337            RelayActorMessage::MaybeCloseRelaysOnRebind => {
338                self.maybe_close_relays_on_rebind().await;
339            }
340            RelayActorMessage::CheckConnectionAfterNetworkChange => {
341                self.check_connection_after_network_change().await;
342            }
343        }

346    /// Sends datagrams to the correct [`ActiveRelayActor`], or returns a future.
347    ///
348    /// If the datagram can not be sent immediately, because the destination channel is
349    /// full, a future is returned that will complete once the datagrams have been sent to
350    /// the [`ActiveRelayActor`].
351    async fn try_send_datagram(
352        &mut self,
353        item: RelaySendItem,
354    ) -> Option<impl Future<Output = ()> + use<>> {
355        let url = item.url.clone();
356        let handle = match self
357            .active_relay_handle_for_endpoint(&item.url, &item.remote_endpoint)
358            .await
359        {
360            Ok(handle) => handle,
361            Err(error) => {
362                self.config.metrics.active_relay_datagrams_rejected.inc();
363                warn!(?url, %error, "Dropped datagram(s): active-relay capacity unavailable");
364                return None;
365            }
366        };
367        match handle.datagrams_send_queue.try_send(item) {
368            Ok(()) => None,
369            Err(mpsc::error::TrySendError::Closed(_)) => {
370                warn!(?url, "Dropped datagram(s): ActiveRelayActor closed.");
371                None
372            }
373            Err(mpsc::error::TrySendError::Full(item)) => {
374                let sender = handle.datagrams_send_queue.clone();
375                let fut = async move {
376                    if sender.send(item).await.is_err() {
377                        warn!(?url, "Dropped datagram(s): ActiveRelayActor closed.");
378                    }
379                };
380                Some(fut)
381            }
382        }
383    }

385    async fn on_network_change(&mut self, report: Report) {
386        let prev = self.config.my_relay.get();
387        let prev_url = prev.as_ref().map(RelayStatus::url);
388        if report.preferred_relay.as_ref() == prev_url {
389            // No change.
390            return;
391        }
392
393        if let Some(relay_url) = report.preferred_relay {
394            self.config.metrics.relay_home_change.inc();
395
396            // On change, notify all currently connected relay servers and
397            // start connecting to our home relay if we are not already.
398            info!("home is now relay {}, was {:?}", relay_url, prev_url);
399            // Publish `Connecting` initially. If an `ActiveRelayActor` already
400            // exists for this URL it will republish its actual status (e.g.
401            // `Connected`) when it receives the `SetHomeRelay(true)` message
402            // sent below.
403            self.config
404                .my_relay
405                .set(relay_url.clone(), RelayConnectionState::Connecting);
406            self.set_home_relay(relay_url).await;
407        } else {
408            self.config.my_relay.clear();
409        }
410    }

412    async fn set_home_relay(&mut self, home_url: RelayUrl) {
413        let home_url_ref = &home_url;
414        n0_future::join_all(self.active_relays.iter().map(|(url, handle)| async move {
415            let is_preferred = url == home_url_ref;
416            handle
417                .inbox_addr
418                .send(ActiveRelayMessage::SetHomeRelay(is_preferred))
419                .await
420                .ok()
421        }))
422        .await;
423        // Ensure we have an ActiveRelayActor for the current home relay.
424        if let Err(error) = self.active_relay_handle(home_url.clone()) {
425            self.config.my_relay.set_status(
426                &home_url,
427                RelayConnectionState::Disconnected {
428                    last_error: Some(Arc::new(AnyError::from(error))),
429                },
430            );
431        }
432    }

434    /// Returns the handle for the [`ActiveRelayActor`] to reach `remote_endpoint`.
435    ///
436    /// The endpoint is expected to be reachable on `url`, but if no [`ActiveRelayActor`] for
437    /// `url` exists but another existing [`ActiveRelayActor`] already knows about the endpoint,
438    /// that other endpoint is used.
439    async fn active_relay_handle_for_endpoint(
440        &mut self,
441        url: &RelayUrl,
442        remote_endpoint: &EndpointId,
443    ) -> Result<ActiveRelayHandle, ActiveRelayAdmissionError> {
444        if let Some(handle) = self.active_relays.get(url) {
445            return Ok(handle.clone());
446        }
447
448        let mut found_relay: Option<RelayUrl> = None;
449        // If we don't have an open connection to the remote endpoint's home relay, see if
450        // we have an open connection to a relay endpoint where we'd heard from that peer
451        // already.  E.g. maybe they dialed our home relay recently.
452        {
453            // Futures which return Some(RelayUrl) if the relay knows about the remote endpoint.
454            let check_futs = self.active_relays.iter().map(|(url, handle)| async move {
455                let (tx, rx) = oneshot::channel();
456                handle
457                    .prio_inbox_addr
458                    .send(ActiveRelayPrioMessage::HasEndpointRoute(
459                        *remote_endpoint,
460                        tx,
461                    ))
462                    .await
463                    .ok();
464                match rx.await {
465                    Ok(true) => Some(url.clone()),
466                    _ => None,
467                }
468            });
469            let mut futures = FuturesUnorderedBounded::from_iter(check_futs);
470            while let Some(maybe_url) = futures.next().await {
471                if maybe_url.is_some() {
472                    found_relay = maybe_url;
473                    break;
474                }
475            }
476        }
477        let url = found_relay.unwrap_or(url.clone());
478        self.active_relay_handle(url)
479    }

481    /// Returns the handle of the [`ActiveRelayActor`].
482    fn active_relay_handle(
483        &mut self,
484        url: RelayUrl,
485    ) -> Result<ActiveRelayHandle, ActiveRelayAdmissionError> {
486        match self.active_relays.get(&url) {
487            Some(e) => Ok(e.clone()),
488            None => {
489                let handle = self.start_active_relay(url.clone())?;
490                if Some(&url) == self.config.my_relay.get().as_ref().map(RelayStatus::url)
491                    && let Err(err) = handle
492                        .inbox_addr
493                        .try_send(ActiveRelayMessage::SetHomeRelay(true))
494                {
495                    error!("Home relay not set, send to new actor failed: {err:#}.");
496                }
497                self.active_relays.insert(url, handle.clone());
498                self.log_active_relay();
499                Ok(handle)
500            }
501        }
502    }

504    fn start_active_relay(
505        &mut self,
506        url: RelayUrl,
507    ) -> Result<ActiveRelayHandle, ActiveRelayAdmissionError> {
508        let permit = self.admission.try_acquire().map_err(|error| {
509            self.config.metrics.active_relay_capacity_rejections.inc();
510            ActiveRelayAdmissionError::from(error)
511        })?;
512        debug!(?url, "Adding relay connection");
513
514        let auth_token = self
515            .config
516            .relay_map
517            .get(&url)
518            .and_then(|cfg| cfg.auth_token.clone());
519        let connection_opts = RelayConnectionOptions {
520            secret_key: self.config.secret_key.clone(),
521            #[cfg(not(wasm_browser))]
522            dns_resolver: self.config.dns_resolver.clone(),
523            proxy_url: self.config.proxy_url.clone(),
524            prefer_ipv6: self.config.ipv6_reported.clone(),
525            tls_config: self.config.tls_config.clone(),
526            auth_token,
527        };
528
529        // TODO: Replace 64 with PER_CLIENT_SEND_QUEUE_DEPTH once that's unused
530        let (send_datagram_tx, send_datagram_rx) = mpsc::channel(64);
531        let (prio_inbox_tx, prio_inbox_rx) = mpsc::channel(32);
532        let (inbox_tx, inbox_rx) = mpsc::channel(64);
533        let span = info_span!("active-relay", %url);
534        let opts = ActiveRelayActorOptions {
535            url,
536            prio_inbox_: prio_inbox_rx,
537            inbox: inbox_rx,
538            relay_datagrams_send: send_datagram_rx,
539            relay_datagrams_recv: self.relay_datagram_recv_queue.clone(),
540            connection_opts,
541            #[cfg(not(wasm_browser))]
542            relay_connector: self.config.relay_connector.clone(),
543            #[cfg(not(wasm_browser))]
544            runtime: self.runtime.clone(),
545            stop_token: self.cancel_token.child_token(),
546            metrics: self.config.metrics.clone(),
547            my_relay: self.config.my_relay.clone(),
548        };
549        let handle = ActiveRelayHandle {
550            prio_inbox_addr: prio_inbox_tx,
551            inbox_addr: inbox_tx,
552            datagrams_send_queue: send_datagram_tx,
553        };
554        #[cfg(wasm_browser)]
555        let actor = ActiveRelayActor::new(opts).expect("browser relay timer setup is infallible");
556        #[cfg(not(wasm_browser))]
557        let actor = match ActiveRelayActor::new(opts) {
558            Ok(actor) => actor,
559            Err(error) => {
560                self.runtime.latch_failure(error);
561                return Err(e!(ActiveRelayAdmissionError::ConstructionFailed));
562            }
563        };
564        #[cfg(wasm_browser)]
565        self.active_relay_tasks.spawn(
566            async move {
567                actor.run().await;
568                permit
569            }
570            .instrument(span),
571        );
572        #[cfg(not(wasm_browser))]
573        {
574            let completions = self.active_relay_completions_tx.clone();
575            let future = async move {
576                let actor = actor.run().instrument(span);
577                match AssertUnwindSafe(actor).catch_unwind().await {
578                    Ok(()) => {
579                        let _ = completions
580                            .send(ActiveRelayCompletion::Finished(permit))
581                            .await;
582                    }
583                    Err(panic) => {
584                        let _ = completions
585                            .send(ActiveRelayCompletion::Panicked(permit))
586                            .await;
587                        std::panic::resume_unwind(panic);
588                    }
589                }
590            };
591            if self
592                .runtime
593                .spawn(
594                    krikos_runtime::TaskKind::Relay,
595                    "active-relay-actor",
596                    Box::pin(future),
597                )
598                .is_ok()
599            {
600                self.active_relay_task_count = self
601                    .active_relay_task_count
602                    .checked_add(1)
603                    .expect("active-relay task count must not overflow");
604            } else {
605                self.config.metrics.active_relay_spawn_rejections.inc();
606                return Err(e!(ActiveRelayAdmissionError::SpawnRejected));
607            }
608        }
609        self.log_active_relay();
610        Ok(handle)
611    }

613    /// Triggers an immediate health check on all relay connections after a network change.
614    async fn check_connection_after_network_change(&mut self) {
615        self.send_check_connection().await;
616    }
617
618    /// Closes the relay connections not originating from a local IP address.
619    ///
620    /// Called in response to a rebind, any relay connection originating from an address
621    /// that's not known to be currently a local IP address should be closed.  All the other
622    /// relay connections are pinged.
623    async fn maybe_close_relays_on_rebind(&mut self) {
624        self.send_check_connection().await;
625        self.log_active_relay();
626    }
627
628    /// Sends a [`ActiveRelayMessage::CheckConnection`] to all active relays with current
629    /// local IPs.
630    async fn send_check_connection(&self) {
631        #[cfg(not(wasm_browser))]
632        let ifs = interfaces::State::new().await;
633        #[cfg(not(wasm_browser))]
634        let local_ips: Vec<_> = ifs
635            .interfaces
636            .values()
637            .flat_map(|netif| netif.addrs())
638            .map(|ipnet| ipnet.addr())
639            .collect();
640        // In browsers, we don't have this information. This will do the right thing
641        // in the ActiveRelayActor, though.
642        #[cfg(wasm_browser)]
643        let local_ips = Vec::new();
644        let send_futs = self.active_relays.values().map(|handle| {
645            let local_ips = local_ips.clone();
646            async move {
647                handle
648                    .inbox_addr
649                    .send(ActiveRelayMessage::CheckConnection { local_ips })
650                    .await
651                    .ok();
652            }
653        });
654        n0_future::join_all(send_futs).await;
655    }
656
657    /// Cleans up [`ActiveRelayActor`]s which have stopped running.
658    fn reap_active_relays(&mut self) {
659        self.active_relays
660            .retain(|_url, handle| !handle.inbox_addr.is_closed());
661
662        // Make sure home relay exists
663        if let Some(status) = self.config.my_relay.get() {
664            let url = status.url().clone();
665            if let Err(error) = self.active_relay_handle(url.clone()) {
666                self.config.my_relay.set_status(
667                    &url,
668                    RelayConnectionState::Disconnected {
669                        last_error: Some(Arc::new(AnyError::from(error))),
670                    },
671                );
672            }
673        }
674        self.log_active_relay();
675    }
676
677    /// Stops all [`ActiveRelayActor`]s and awaits for them to finish.
678    async fn close_all_active_relays(&mut self) {
679        self.cancel_token.cancel();
680        #[cfg(wasm_browser)]
681        {
682            let tasks = std::mem::take(&mut self.active_relay_tasks);
683            tasks.join_all().await;
684        }
685        #[cfg(not(wasm_browser))]
686        while self.active_relay_task_count > 0 {
687            match self.active_relay_completions_rx.recv().await {
688                Some(completion) => {
689                    drop(completion);
690                    self.active_relay_task_count = self
691                        .active_relay_task_count
692                        .checked_sub(1)
693                        .expect("active-relay shutdown task count must not underflow");
694                }
695                None => break,
696            }
697        }
698
699        self.log_active_relay();
700    }
701
702    fn log_active_relay(&self) {
703        debug!("{} active relay conns{}", self.active_relays.len(), {
704            let mut s = String::new();
705            if !self.active_relays.is_empty() {
706                s += ":";
707                for endpoint in self.active_relay_sorted() {
708                    s += &format!(" relay-{endpoint}");
709                }
710            }
711            s
712        });
713    }
714
715    fn active_relay_sorted(&self) -> impl Iterator<Item = RelayUrl> + use<> {
716        let mut ids: Vec<_> = self.active_relays.keys().cloned().collect();
717        ids.sort();
718
719        ids.into_iter()
720    }
