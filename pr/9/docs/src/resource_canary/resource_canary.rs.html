261#[tokio::main]
262async fn main() -> ExitCode {
263    let filter = EnvFilter::try_from_default_env()
264        .unwrap_or_else(|_| EnvFilter::new("resource_canary=warn"));
265    if let Err(error) = tracing_subscriber::fmt().with_env_filter(filter).try_init() {
266        eprintln!("resource canary tracing initialization failed: {error}");
267    }
268    match run(Cli::parse()).await {
269        Ok(()) => ExitCode::SUCCESS,
270        Err(error) => {
271            eprintln!("resource canary failed: {error}");
272            ExitCode::FAILURE
273        }
274    }

311async fn run_preflight(
312    host_profile: HostProfile,
313    baseline_seconds: u64,
314    output: &Path,
315) -> Result<(), Box<dyn Error>> {
316    if baseline_seconds > MAX_BASELINE_SECONDS {
317        return Err(format!(
318            "baseline_seconds {baseline_seconds} exceeds maximum {MAX_BASELINE_SECONDS}"
319        )
320        .into());
321    }
322    let artifact_dir = create_artifact_dir(output, "preflight")?;
323    println!("{}", artifact_dir.display());
324    let result = async {
325        let (report, result) = observe_preflight(
326            host_profile,
327            Duration::from_secs(baseline_seconds),
328            &artifact_dir,
329        )
330        .await?;
331        let report_path = artifact_dir.join("preflight.json");
332        write_json_new(&report_path, &report)?;
333        result.map_err(|error| Box::new(error) as Box<dyn Error>)
334    }
335    .await;
336    finish_artifact_run(&artifact_dir, result)
337}

339#[allow(
340    clippy::too_many_arguments,
341    reason = "CLI timing and scale remain explicit at the process boundary"
342)]
343async fn run_canary(
344    host_profile: HostProfile,
345    lane: LaneSelection,
346    scale_percent: u8,
347    warmup_seconds: u64,
348    measurement_seconds: u64,
349    cooldown_seconds: u64,
350    sample_interval_seconds: u64,
351    baseline_seconds: u64,
352    output: &Path,
353) -> Result<(), Box<dyn Error>> {
354    if scale_percent > 100 {
355        return Err("scale_percent must be in 1..=100".into());
356    }
357    if baseline_seconds > MAX_BASELINE_SECONDS {
358        return Err(format!(
359            "baseline_seconds {baseline_seconds} exceeds maximum {MAX_BASELINE_SECONDS}"
360        )
361        .into());
362    }
363    let timing = DurationProfile::new(
364        Duration::from_secs(warmup_seconds),
365        Duration::from_secs(measurement_seconds),
366        Duration::from_secs(cooldown_seconds),
367        Duration::from_secs(sample_interval_seconds),
368    )?;
369    let requested_evidence =
370        CanaryMode::classify(&timing, scale_percent, EvidencePrerequisites::ready()).is_evidence();
371    let release_build = is_release_build();
372    let source_clean = source_tree_clean()?;
373    let all_lanes = lane == LaneSelection::All;
374    let mode = CanaryMode::classify(
375        &timing,
376        scale_percent,
377        EvidencePrerequisites::new(all_lanes, release_build, source_clean, host_profile),
378    );
379    if requested_evidence && !mode.is_evidence() {
380        return Err(format!(
381            "evidence prerequisites failed: all_lanes={all_lanes}, release_build={release_build}, \
382             source_clean={source_clean}, host_profile={}",
383            host_profile.as_str()
384        )
385        .into());
386    }
387    let artifact_dir = create_artifact_dir(output, "run")?;
388    println!("{}", artifact_dir.display());
389
390    let result = async {
391        let (preflight, preflight_result) = observe_preflight(
392            host_profile,
393            Duration::from_secs(baseline_seconds),
394            &artifact_dir,
395        )
396        .await?;
397        write_json_new(&artifact_dir.join("preflight.json"), &preflight)?;
398        if let Err(error) = preflight_result {
399            return Err(Box::new(error) as Box<dyn Error>);
400        }
401
402        let mut lanes = Vec::new();
403        if lane.includes(LaneSelection::Dns) {
404            let report = execute_dns_lane(scale_percent, timing, &preflight, &artifact_dir).await?;
405            let accepted = report.accepted;
406            lanes.push(report);
407            write_run_report(
408                &artifact_dir,
409                host_profile,
410                mode,
411                scale_percent,
412                timing,
413                false,
414                &lanes,
415            )?;
416            if !accepted {
417                return Err("DNS resource lane failed acceptance".into());
418            }
419        }
420        if lane.includes(LaneSelection::Relay) {
421            let report =
422                execute_relay_lane(scale_percent, timing, &preflight, &artifact_dir).await?;
423            let accepted = report.accepted;
424            lanes.push(report);
425            write_run_report(
426                &artifact_dir,
427                host_profile,
428                mode,
429                scale_percent,
430                timing,
431                false,
432                &lanes,
433            )?;
434            if !accepted {
435                return Err("relay resource lane failed acceptance".into());
436            }
437        }
438        if lane.includes(LaneSelection::Endpoint) {
439            let report =
440                execute_endpoint_lane(scale_percent, timing, &preflight, &artifact_dir).await?;
441            let accepted = report.accepted;
442            lanes.push(report);
443            write_run_report(
444                &artifact_dir,
445                host_profile,
446                mode,
447                scale_percent,
448                timing,
449                false,
450                &lanes,
451            )?;
452            if !accepted {
453                return Err("endpoint resource lane failed acceptance".into());
454            }
455        }
456        write_run_report(
457            &artifact_dir,
458            host_profile,
459            mode,
460            scale_percent,
461            timing,
462            true,
463            &lanes,
464        )?;
465        Ok::<(), Box<dyn Error>>(())
466    }
467    .await;
468    finish_artifact_run(&artifact_dir, result)
469}

471async fn execute_dns_lane(
472    scale_percent: u8,
473    timing: DurationProfile,
474    preflight: &PreflightReport,
475    artifact_dir: &Path,
476) -> Result<LaneReport, Box<dyn Error>> {
477    let production = WorkloadProfile::production_twice()?;
478    let udp_capacity = scaled_nonzero(production.dns_udp_capacity(), scale_percent, 2)?;
479    let tcp_capacity = scaled_nonzero(production.dns_tcp_capacity(), scale_percent, 2)?;
480    let http_connection_capacity =
481        scaled_nonzero(production.http_connection_capacity(), scale_percent, 2)?;
482    let http_request_capacity =
483        scaled_nonzero(production.http_request_capacity(), scale_percent, 2)?;
484    let http2_streams_per_connection = nonzero(
485        production.http2_streams_per_connection(),
486        "DNS HTTP/2 streams per connection",
487    )?;
488    let http_accept_rate = if scale_percent == 100 {
489        production.http_accept_rate_per_second()
490    } else {
491        100
492    };
493    let http_accept_rate = nonzero(http_accept_rate, "DNS HTTP accept rate")?;
494    let http_accept_burst = if scale_percent == 100 {
495        nonzero(production.http_accept_burst(), "DNS HTTP accept burst")?
496    } else {
497        doubled(http_connection_capacity, "DNS HTTP accept burst")?
498    };
499    let operation_timeout = operation_timeout(scale_percent);
500    let udp_rate = scaled_nonzero(1_000, scale_percent, 100)?;
501    let config = DnsLaneConfig::new(
502        udp_capacity,
503        udp_rate,
504        tcp_capacity,
505        http_connection_capacity,
506        http_request_capacity,
507        http2_streams_per_connection,
508        doubled(udp_capacity, "DNS UDP offered load")?,
509        doubled(tcp_capacity, "DNS TCP offered load")?,
510        doubled(http_connection_capacity, "DNS HTTP connection offered load")?,
511        doubled(http_request_capacity, "DNS HTTP request offered load")?,
512        http_accept_rate,
513        http_accept_burst,
514        lane_timing(timing)?,
515        operation_timeout,
516    )?;
517    let (phases, phase_receiver) = PhaseReporter::new();
518    let (outcome, samples, final_state, elapsed) = monitor_lane(
519        timing.sample_interval(),
520        phase_receiver,
521        run_dns_lane(config, phases),
522    )
523    .await?;
524    let samples_file = "dns-samples.ndjson";
525    write_samples(&artifact_dir.join(samples_file), &samples)?;
526    let resources = resource_summary(&samples)?;
527    match outcome {
528        Ok(outcome) => {
529            let udp_admission_outcomes = u64::try_from(outcome.udp_completed)
530                .map_err(|_| "DNS UDP completion count is out of range")?
531                .checked_add(outcome.udp_rejections)
532                .ok_or(CanaryError::ArithmeticOverflow)?;
533            let expected_udp_rejections = outcome
534                .udp_offered
535                .checked_sub(udp_capacity.get())
536                .ok_or(CanaryError::ArithmeticOverflow)?;
537            let initial_http_connection_rejections = outcome
538                .http_connection_capacity_rejections
539                .checked_add(outcome.http_connection_rate_rejections)
540                .ok_or(CanaryError::ArithmeticOverflow)?;
541            let initial_http_connection_outcomes =
542                u64::try_from(outcome.http_connections_active_high_water)
543                    .map_err(|_| "DNS HTTP connection high-water is out of range")?
544                    .checked_add(initial_http_connection_rejections)
545                    .ok_or(CanaryError::ArithmeticOverflow)?;
546            let expected_total_http_connection_rejections = initial_http_connection_rejections
547                .checked_add(
548                    u64::try_from(outcome.continuity_http_connection_rejections)
549                        .map_err(|_| "DNS HTTP continuity count is out of range")?,
550                )
551                .ok_or(CanaryError::ArithmeticOverflow)?;
552            let acceptance = AcceptanceInput {
553                peak_cpu_basis_points: resources.peak_cpu_basis_points,
554                visible_memory_bytes: preflight.memory_total_bytes,
555                peak_rss_bytes: resources.peak_rss_bytes,
556                file_descriptor_limit: preflight.file_descriptor_limit,
557                peak_file_descriptors: resources.peak_file_descriptors,
558                shutdown: outcome.shutdown,
559                samples_complete: phase_coverage_complete(timing, &samples)?,
560                admission: AdmissionSample {
561                    maximum: tcp_capacity.get(),
562                    high_water: outcome.tcp_active_high_water,
563                    rejections: outcome.tcp_rejections,
564                    counter_exhausted: false,
565                },
566            };
567            let mut failure = evaluate_acceptance(&acceptance, HeadroomThreshold::thirty_percent())
568                .err()
569                .map(|error| error.to_string());
570            if failure.is_none() && !outcome.recovered {
571                failure = Some("DNS admission did not recover after release".to_owned());
572            }
573            if failure.is_none()
574                && outcome.tcp_rejections
575                    != u64::try_from(
576                        outcome
577                            .tcp_offered
578                            .checked_sub(tcp_capacity.get())
579                            .ok_or(CanaryError::ArithmeticOverflow)?,
580                    )
581                    .map_err(|_| "DNS TCP rejection count is out of range")?
582            {
583                failure = Some("DNS TCP admission outcomes do not conserve attempts".to_owned());
584            }
585            if failure.is_none()
586                && (outcome.udp_completed == 0
587                    || outcome.udp_rejections == 0
588                    || outcome.udp_arrival.attempts != outcome.udp_offered
589                    || outcome.udp_completed != udp_capacity.get()
590                    || outcome.udp_rejections
591                        != u64::try_from(expected_udp_rejections)
592                            .map_err(|_| "DNS UDP rejection count is out of range")?
593                    || udp_admission_outcomes
594                        != u64::try_from(outcome.udp_offered)
595                            .map_err(|_| "DNS UDP offered count is out of range")?
596                    || outcome
597                        .udp_completed
598                        .checked_add(outcome.udp_timed_out)
599                        .is_none_or(|total| total != outcome.udp_offered))
600            {
601                failure = Some(
602                    "DNS UDP admission lacks success, overload, or conserved outcomes".to_owned(),
603                );
604            }
605            if failure.is_none()
606                && (outcome.http_connections_active_high_water != http_connection_capacity.get()
607                    || outcome.http_connection_rejections == 0
608                    || outcome.http_connection_arrival.attempts != outcome.http_connections_offered
609                    || initial_http_connection_outcomes
610                        != u64::try_from(outcome.http_connections_offered)
611                            .map_err(|_| "DNS HTTP offered count is out of range")?
612                    || outcome.http_connection_rejections
613                        != expected_total_http_connection_rejections)
614            {
615                failure = Some(
616                    "DNS HTTP connection admission or arrival accounting was incomplete".to_owned(),
617                );
618            }
619            if failure.is_none()
620                && (outcome.http_requests_active_high_water != http_request_capacity.get()
621                    || outcome.http_request_rejections == 0
622                    || outcome.http_requests_admitted != http_request_capacity.get()
623                    || !outcome.http_request_recovered
624                    || u64::try_from(outcome.http_requests_admitted)
625                        .ok()
626                        .and_then(|admitted| admitted.checked_add(outcome.http_request_rejections))
627                        != u64::try_from(outcome.http_requests_offered).ok())
628            {
629                failure = Some(
630                    "DNS HTTP request admission lacks saturation, conservation, or recovery"
631                        .to_owned(),
632                );
633            }
634            if failure.is_none()
635                && (outcome.continuity_udp_successes == 0
636                    || outcome.continuity_http_connection_rejections == 0
637                    || outcome.continuity_http_request_rejections == 0)
638            {
639                failure =
640                    Some("DNS timed phases lacked continuing success or rejection".to_owned());
641            }
642            if failure.is_none() && outcome.store_background_failures != 0 {
643                failure = Some(format!(
644                    "DNS store reported {} background failures",
645                    outcome.store_background_failures
646                ));
647            }
648            let diagnostic = lane_diagnostic(
649                final_state,
650                elapsed,
651                &samples,
652                failure.as_ref().map(|_| "acceptance"),
653            )?;
654            Ok(LaneReport {
655                lane: "dns",
656                accepted: failure.is_none(),
657                failure,
658                resources: Some(resources),
659                outcome: Some(dns_outcome_json(&outcome)?),
660                samples_file: samples_file.to_owned(),
661                diagnostic,
662            })
663        }
664        Err(error) => {
665            let diagnostic = lane_diagnostic(final_state, elapsed, &samples, Some("workload"))?;
666            Ok(LaneReport {

679async fn execute_relay_lane(
680    scale_percent: u8,
681    timing: DurationProfile,
682    preflight: &PreflightReport,
683    artifact_dir: &Path,
684) -> Result<LaneReport, Box<dyn Error>> {
685    let production = WorkloadProfile::production_twice()?;
686    let pending_capacity = scaled_nonzero(production.relay_pending_capacity(), scale_percent, 2)?;
687    let session_capacity = scaled_nonzero(production.relay_session_capacity(), scale_percent, 2)?;
688    let sessions_per_identity = nonzero(
689        production
690            .relay_sessions_per_identity()
691            .min(session_capacity.get()),
692        "relay sessions per identity",
693    )?;
694    let (fill_rate, overload_rate, accept_burst) = if scale_percent == 100 {
695        (
696            nonzero(production.relay_accept_rate_per_second(), "relay fill rate")?,
697            nonzero(
698                production
699                    .relay_accept_rate_per_second()
700                    .checked_mul(2)
701                    .ok_or(CanaryError::ArithmeticOverflow)?,
702                "relay overload rate",
703            )?,
704            nonzero(production.relay_accept_burst(), "relay accept burst")?,
705        )
706    } else {
707        (
708            nonzero(100, "relay fill rate")?,
709            nonzero(200, "relay overload rate")?,
710            nonzero(production.relay_accept_burst(), "relay accept burst")?,
711        )
712    };
713    let operation_timeout = operation_timeout(scale_percent);
714    let config = RelayLaneConfig::new(
715        pending_capacity,
716        session_capacity,
717        sessions_per_identity,
718        doubled(pending_capacity, "relay pending offered load")?,
719        doubled(session_capacity, "relay session offered load")?,
720        fill_rate,
721        overload_rate,
722        accept_burst,
723        lane_timing(timing)?,
724        operation_timeout,
725    )?;
726    let (phases, phase_receiver) = PhaseReporter::new();
727    let (outcome, samples, final_state, elapsed) = monitor_lane(
728        timing.sample_interval(),
729        phase_receiver,
730        run_relay_lane(config, phases),
731    )
732    .await?;
733    let samples_file = "relay-samples.ndjson";
734    write_samples(&artifact_dir.join(samples_file), &samples)?;
735    let resources = resource_summary(&samples)?;
736    match outcome {
737        Ok(outcome) => {
738            let acceptance = AcceptanceInput {
739                peak_cpu_basis_points: resources.peak_cpu_basis_points,
740                visible_memory_bytes: preflight.memory_total_bytes,
741                peak_rss_bytes: resources.peak_rss_bytes,
742                file_descriptor_limit: preflight.file_descriptor_limit,
743                peak_file_descriptors: resources.peak_file_descriptors,
744                shutdown: outcome.shutdown,
745                samples_complete: phase_coverage_complete(timing, &samples)?,
746                admission: AdmissionSample {
747                    maximum: session_capacity.get(),
748                    high_water: outcome.session_high_water,
749                    rejections: u64::try_from(outcome.sessions_rejected)
750                        .map_err(|_| "relay rejection count is out of range")?,
751                    counter_exhausted: false,
752                },
753            };
754            let mut failure = evaluate_acceptance(&acceptance, HeadroomThreshold::thirty_percent())
755                .err()
756                .map(|error| error.to_string());
757            if failure.is_none() && outcome.pending_rejections == 0 {
758                failure = Some("relay pending admission did not reject overload".to_owned());
759            }
760            if failure.is_none() && outcome.endpoint_session_rejections == 0 {
761                failure = Some("relay per-identity admission did not reject overload".to_owned());
762            }
763            if failure.is_none() && outcome.global_session_rejections == 0 {
764                failure = Some("relay global admission did not reject overload".to_owned());
765            }
766            if failure.is_none()
767                && (outcome.fill_arrival.attempts != outcome.sessions_accepted
768                    || outcome
769                        .identity_overload_arrival
770                        .attempts
771                        .checked_add(outcome.overload_arrival.attempts)
772                        != Some(outcome.sessions_rejected))
773            {
774                failure = Some("relay arrival campaigns do not conserve attempts".to_owned());
775            }
776            if failure.is_none()
777                && (outcome.rejection_client_outcomes.total()? != outcome.sessions_rejected
778                    || outcome.continuity_client_outcomes.total()? != outcome.continuity_rejections
779                    || outcome.rejection_client_outcomes.timed_out != 0
780                    || outcome.continuity_client_outcomes.timed_out != 0)
781            {
782                failure = Some(
783                    "relay client-visible outcome classes do not conserve attempts".to_owned(),
784                );
785            }
786            if failure.is_none() && !outcome.recovered {
787                failure = Some("relay admission did not recover after release".to_owned());
788            }
789            if failure.is_none()
790                && (outcome.continuity_successes == 0 || outcome.continuity_rejections == 0)
791            {
792                failure =
793                    Some("relay timed phases lacked continuing success or rejection".to_owned());
794            }
795            if failure.is_none()
796                && outcome
797                    .endpoint_session_rejections
798                    .checked_add(outcome.global_session_rejections)
799                    .and_then(|value| value.checked_add(outcome.session_pending_rejections))
800                    .and_then(|value| value.checked_add(outcome.rate_rejections))
801                    != u64::try_from(
802                        outcome
803                            .sessions_rejected
804                            .checked_add(outcome.continuity_rejections)
805                            .ok_or(CanaryError::ArithmeticOverflow)?,
806                    )
807                    .ok()
808            {
809                failure = Some("relay session status classes do not conserve attempts".to_owned());
810            }
811            let diagnostic = lane_diagnostic(
812                final_state,
813                elapsed,
814                &samples,
815                failure.as_ref().map(|_| "acceptance"),
816            )?;
817            Ok(LaneReport {
818                lane: "relay",
819                accepted: failure.is_none(),
820                failure,
821                resources: Some(resources),
822                outcome: Some(relay_outcome_json(&outcome)?),
823                samples_file: samples_file.to_owned(),
824                diagnostic,
825            })
826        }
827        Err(error) => {
828            let diagnostic = lane_diagnostic(final_state, elapsed, &samples, Some("workload"))?;
829            Ok(LaneReport {
830                lane: "relay",
831                accepted: false,
832                failure: Some(error.to_string()),
833                resources: Some(resources),
834                outcome: None,
835                samples_file: samples_file.to_owned(),
836                diagnostic,
837            })
838        }
839    }
840}

935async fn monitor_lane<T, E>(
936    sample_interval: Duration,
937    phase_receiver: watch::Receiver<LaneState>,
938    future: impl Future<Output = Result<T, E>>,
939) -> Result<(Result<T, E>, Vec<ResourceSample>, LaneState, Duration), Box<dyn Error>> {
940    let started = Instant::now();
941    let final_state = phase_receiver.clone();
942    let (stop_tx, stop_rx) = watch::channel(false);
943    let mut sampler = tokio::spawn(sample_resources(sample_interval, stop_rx, phase_receiver));
944    tokio::select! {
945        outcome = future => {
946            stop_tx.send(true).map_err(|_| "resource sampler stopped before cancellation")?;
947            let samples = sampler
948                .await
949                .map_err(|error| format!("resource sampler task failed: {error}"))?
950                .map_err(io::Error::other)?;
951            let state = *final_state.borrow();
952            Ok((outcome, samples, state, started.elapsed()))
953        }
954        result = &mut sampler => {
955            let samples = result
956                .map_err(|error| format!("resource sampler task failed: {error}"))?
957                .map_err(io::Error::other)?;
958            Err(format!(
959                "resource sampler stopped before the workload after {} samples",
960                samples.len()
961            )
962            .into())
963        }
964    }
965}

967async fn sample_resources(
968    interval: Duration,
969    mut stop: watch::Receiver<bool>,
970    phase: watch::Receiver<LaneState>,
971) -> Result<Vec<ResourceSample>, String> {
972    let started = Instant::now();
973    let mut previous_cpu =
974        parse_cpu_ticks(&fs::read_to_string("/proc/stat").map_err(|error| error.to_string())?)
975            .map_err(|error| error.to_string())?;
976    let mut samples = Vec::new();
977    let mut ticker = resource_sample_ticker(interval)?;
978    loop {
979        tokio::select! {
980            changed = stop.changed() => {
981                changed.map_err(|error| error.to_string())?;
982                if *stop.borrow() {
983                    if samples.is_empty() {
984                        let (sample, _) = collect_resource_sample(
985                            started,
986                            previous_cpu,
987                            phase.borrow().phase.as_str(),
988                        )?;
989                        samples.push(sample);
990                    }
991                    break;
992                }
993            }
994            _ = ticker.tick() => {
995                if samples.len() >= MAX_RESOURCE_SAMPLES {
996                    return Err(format!(
997                        "resource sample count exceeds maximum {MAX_RESOURCE_SAMPLES}"
998                    ));
999                }
1000                let (sample, current_cpu) = collect_resource_sample(
1001                    started,
1002                    previous_cpu,
1003                    phase.borrow().phase.as_str(),
1004                )?;
1005                samples.push(sample);
1006                previous_cpu = current_cpu;
1007            }
1008        }
1009    }
1010    Ok(samples)
1011}

1013fn resource_sample_ticker(interval: Duration) -> Result<tokio::time::Interval, String> {
1014    let first_sample = tokio::time::Instant::now()
1015        .checked_add(interval)
1016        .ok_or_else(|| "resource sample deadline overflowed".to_owned())?;
1017    let mut ticker = tokio::time::interval_at(first_sample, interval);
1018    ticker.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
1019    Ok(ticker)
1020}

1022fn collect_resource_sample(
1023    started: Instant,
1024    previous_cpu: iroh_bench::canary::CpuTicks,
1025    phase: &'static str,
1026) -> Result<(ResourceSample, iroh_bench::canary::CpuTicks), String> {
1027    let current_cpu =
1028        parse_cpu_ticks(&fs::read_to_string("/proc/stat").map_err(|error| error.to_string())?)
1029            .map_err(|error| error.to_string())?;
1030    let memory =
1031        parse_meminfo(&fs::read_to_string("/proc/meminfo").map_err(|error| error.to_string())?)
1032            .map_err(|error| error.to_string())?;
1033    let process = parse_process_status(
1034        &fs::read_to_string("/proc/self/status").map_err(|error| error.to_string())?,
1035    )
1036    .map_err(|error| error.to_string())?;
1037    let elapsed_millis = u64::try_from(started.elapsed().as_millis())
1038        .map_err(|_| "sample elapsed time is out of range".to_owned())?;
1039    Ok((
1040        ResourceSample {
1041            elapsed_millis,
1042            phase,
1043            cpu_basis_points: cpu_usage_basis_points(previous_cpu, current_cpu)
1044                .map_err(|error| error.to_string())?,
1045            memory_total_bytes: memory.total_bytes(),
1046            memory_available_bytes: memory.available_bytes(),
1047            swap_used_bytes: memory.swap_used_bytes(),
1048            process_rss_bytes: process.rss_bytes(),
1049            process_file_descriptors: count_open_file_descriptors()
1050                .map_err(|error| error.to_string())?,
1051            process_threads: process.threads(),
1052        },
1053        current_cpu,
1054    ))
1055}

1110fn phase_coverage_complete(
1111    timing: DurationProfile,
1112    samples: &[ResourceSample],
1113) -> Result<bool, CanaryError> {
1114    let interval_nanos = timing.sample_interval().as_nanos();
1115    let required = [
1116        (LanePhase::Warmup, timing.warmup()),
1117        (LanePhase::Measurement, timing.measurement()),
1118        (LanePhase::Cooldown, timing.cooldown()),
1119    ];
1120    for (phase, duration) in required {
1121        let expected = duration.as_nanos() / interval_nanos;
1122        let expected = usize::try_from(expected).map_err(|_| CanaryError::ArithmeticOverflow)?;
1123        let minimum = expected.saturating_sub(1);
1124        if phase_sample_count(samples, phase) < minimum {
1125            return Ok(false);
1126        }
1127    }
1128    let expected_total =
1129        usize::try_from(timing.minimum_samples()).map_err(|_| CanaryError::ArithmeticOverflow)?;
1130    Ok(!samples.is_empty() && samples.len() >= expected_total.saturating_sub(3))
1131}

1133fn endpoint_queue_headroom(outcome: &EndpointLaneOutcome) -> Result<bool, Box<dyn Error>> {
1134    let packet_byte_capacity = usize::try_from(noq::DEFAULT_MAX_PACKET_BYTES_PER_ENDPOINT)
1135        .map_err(|_| "Noq packet byte capacity is out of range")?;
1136    for stats in [outcome.client_noq, outcome.server_noq] {
1137        if stats.packet_event_rejections != 0
1138            || stats.packet_byte_rejections != 0
1139            || stats.connection_rejections != 0
1140            || stats.control_event_rejections != 0
1141            || !within_seventy_percent(
1142                stats.packet_events_per_connection_high_water,
1143                noq::DEFAULT_MAX_PACKET_EVENTS_PER_CONNECTION,
1144            )?
1145            || !within_seventy_percent(stats.packet_bytes_high_water, packet_byte_capacity)?
1146            || !within_seventy_percent(
1147                stats.control_events_high_water,
1148                noq::DEFAULT_MAX_CONTROL_EVENTS_PER_ENDPOINT,
1149            )?
1150        {
1151            return Ok(false);
1152        }
1153    }
1154    Ok(true)
1155}

1157fn endpoint_task_headroom(outcome: &EndpointLaneOutcome) -> Result<bool, CanaryError> {
1158    for tasks in [outcome.client_tasks, outcome.server_tasks] {
1159        if tasks.counter_exhausted
1160            || tasks.rejections != 0
1161            || tasks.maximum == 0
1162            || !within_seventy_percent(tasks.current, tasks.maximum)?
1163            || !within_seventy_percent(tasks.high_water, tasks.maximum)?
1164        {
1165            return Ok(false);
1166        }
1167    }
1168    Ok(true)
1169}
1170
1171fn within_seventy_percent(used: usize, total: usize) -> Result<bool, CanaryError> {
1172    let used = used
1173        .checked_mul(10_000)
1174        .ok_or(CanaryError::ArithmeticOverflow)?;
1175    let maximum = total
1176        .checked_mul(7_000)
1177        .ok_or(CanaryError::ArithmeticOverflow)?;
1178    Ok(used <= maximum)
1179}

1242        source_revision: source_revision(),

1252        release_build: is_release_build(),

1256        warmup_seconds: timing.warmup().as_secs(),
1257        measurement_seconds: timing.measurement().as_secs(),

1260        accepted: complete && lanes.iter().all(|lane| lane.accepted),
1261        lanes: lanes

1492async fn observe_preflight(
1493    host_profile: HostProfile,
1494    baseline: Duration,
1495    artifact_dir: &Path,
1496) -> Result<(PreflightReport, Result<(), CanaryError>), Box<dyn Error>> {
1497    let before_cpu = parse_cpu_ticks(&fs::read_to_string("/proc/stat")?)?;
1498    let before_processes = read_processes()?;
1499
1500    tokio::time::sleep(baseline).await;
1501
1502    let after_cpu = parse_cpu_ticks(&fs::read_to_string("/proc/stat")?)?;
1503    let memory = parse_meminfo(&fs::read_to_string("/proc/meminfo")?)?;
1504    let file_descriptor_limit = parse_open_file_limit(&fs::read_to_string("/proc/self/limits")?)?;
1505    let free_storage_bytes = read_storage_available_bytes(artifact_dir)?;
1506    let after_processes = read_processes()?;
1507    let baseline_cpu_basis_points = cpu_usage_basis_points(before_cpu, after_cpu)?;
1508    let total_cpu_delta = after_cpu
1509        .total()
1510        .checked_sub(before_cpu.total())
1511        .ok_or(CanaryError::InvalidCpuDelta)?;
1512    let self_pid = std::process::id();
1513    let mut competitors = Vec::new();
1514    for (pid, after) in &after_processes {
1515        if *pid == self_pid {
1516            continue;
1517        }
1518        let cpu_basis_points = before_processes
1519            .get(pid)
1520            .and_then(|before| after.cpu_ticks.checked_sub(before.cpu_ticks))
1521            .and_then(|delta| delta.checked_mul(10_000))
1522            .map(|scaled| scaled / total_cpu_delta)
1523            .and_then(|value| u16::try_from(value).ok())
1524            .unwrap_or(0);
1525        competitors.push(CompetitorReport {
1526            pid: *pid,
1527            name: after.name.clone(),
1528            cpu_basis_points,
1529            rss_bytes: after.rss_bytes,
1530        });
1531    }
1532    competitors.sort_by(|left, right| {
1533        right
1534            .rss_bytes
1535            .cmp(&left.rss_bytes)
1536            .then_with(|| right.cpu_basis_points.cmp(&left.cpu_basis_points))
1537            .then_with(|| left.pid.cmp(&right.pid))
1538    });
1539    competitors.truncate(16);
1540
1541    let largest_competitor_cpu_basis_points = after_processes
1542        .iter()
1543        .filter(|(pid, _)| **pid != self_pid)
1544        .filter_map(|(pid, after)| {
1545            let before = before_processes.get(pid)?;
1546            let delta = after.cpu_ticks.checked_sub(before.cpu_ticks)?;
1547            let scaled = delta.checked_mul(10_000)?;
1548            u16::try_from(scaled / total_cpu_delta).ok()
1549        })
1550        .max()
1551        .unwrap_or(0);
1552    let largest_competitor_rss_bytes = after_processes
1553        .iter()
1554        .filter(|(pid, _)| **pid != self_pid)
1555        .map(|(_, process)| process.rss_bytes)
1556        .max()
1557        .unwrap_or(0);
1558    let observation = HostObservation {
1559        cpu_cores: std::thread::available_parallelism()?.get(),
1560        memory_total_bytes: memory.total_bytes(),
1561        memory_available_bytes: memory.available_bytes(),
1562        file_descriptor_limit,
1563        free_storage_bytes,
1564        baseline_cpu_basis_points,
1565        largest_competitor_cpu_basis_points,
1566        largest_competitor_rss_bytes,
1567    };
1568    let result = require_production_platform(std::env::consts::OS, std::env::consts::ARCH)
1569        .and_then(|()| evaluate_host_preflight(&observation, host_profile.requirements()));
1570    let report = PreflightReport {
1571        schema_version: 2,
1572        recorded_unix_seconds: unix_seconds()?,
1573        source_revision: source_revision(),
1574        host_profile: host_profile.as_str(),
1575        accepted: result.is_ok(),
1576        failure: result.as_ref().err().map(ToString::to_string),
1577        operating_system: std::env::consts::OS,
1578        architecture: std::env::consts::ARCH,
1579        kernel_release: read_bounded_line("/proc/sys/kernel/osrelease", "kernel release")?,
1580        cpu_model: read_cpu_model()?,
1581        load_averages: read_load_averages()?,
1582        baseline_seconds: baseline.as_secs(),
1583        cpu_cores: observation.cpu_cores,
1584        memory_total_bytes: observation.memory_total_bytes,
1585        memory_available_bytes: observation.memory_available_bytes,
1586        swap_used_bytes: memory.swap_used_bytes(),
1587        file_descriptor_limit: observation.file_descriptor_limit,
1588        free_storage_bytes: observation.free_storage_bytes,
1589        baseline_cpu_basis_points: observation.baseline_cpu_basis_points,
1590        largest_competitor_cpu_basis_points,
1591        largest_competitor_rss_bytes,
1592        competitors,
1593    };
1594    Ok((report, result))
1595}

1633fn read_processes() -> Result<BTreeMap<u32, ProcessReading>, Box<dyn Error>> {
1634    let mut processes = BTreeMap::new();
1635    for entry in fs::read_dir("/proc")? {
1636        let entry = entry?;
1637        let Some(pid) = entry
1638            .file_name()
1639            .to_str()
1640            .and_then(|name| name.parse::<u32>().ok())
1641        else {
1642            continue;
1643        };
1644        if processes.len() >= MAX_PROCESSES {
1645            return Err(format!("process count exceeds maximum {MAX_PROCESSES}").into());
1646        }
1647        let stat = match fs::read_to_string(entry.path().join("stat")) {
1648            Ok(stat) => stat,
1649            Err(_) => continue,
1650        };
1651        let status = match fs::read_to_string(entry.path().join("status")) {
1652            Ok(status) => status,
1653            Err(_) => continue,
1654        };
1655        let cpu_ticks = match parse_process_cpu_ticks(&stat) {
1656            Ok(ticks) => ticks,
1657            Err(_) => continue,
1658        };
1659        let process = match parse_process_status(&status) {
1660            Ok(process) => process,
1661            Err(_) => continue,
1662        };
1663        let name = status
1664            .lines()
1665            .find_map(|line| line.strip_prefix("Name:\t"))
1666            .unwrap_or("unknown")
1667            .to_owned();
1668        processes.insert(
1669            pid,
1670            ProcessReading {
1671                name,
1672                cpu_ticks,
1673                rss_bytes: process.rss_bytes(),
1674            },
1675        );
1676    }
1677    Ok(processes)
1678}

1760                }

1772fn finalize_artifacts(artifact_dir: &Path) -> Result<String, Box<dyn Error>> {
1773    let mut paths = Vec::new();
1774    for entry in fs::read_dir(artifact_dir)? {
1775        let entry = entry?;
1776        if paths.len() >= MAX_ARTIFACT_FILES {
1777            return Err(format!("artifact file count exceeds maximum {MAX_ARTIFACT_FILES}").into());
1778        }
1779        let metadata = fs::symlink_metadata(entry.path())?;
1780        if !metadata.file_type().is_file() {
1781            return Err(format!(
1782                "artifact entry is not a regular file: {}",
1783                entry.path().display()
1784            )
1785            .into());
1786        }
1787        paths.push(entry.path());
1788    }
1789    paths.sort();
1790
1791    let mut files = Vec::with_capacity(paths.len());
1792    for path in &paths {
1793        let (bytes, sha256) = sha256_file(path)?;
1794        let name = path
1795            .file_name()
1796            .and_then(|name| name.to_str())
1797            .ok_or("artifact filename is not valid UTF-8")?;
1798        files.push(ArtifactDigest {
1799            path: name.to_owned(),
1800            bytes,
1801            sha256,
1802        });
1803    }
1804    let manifest = ArtifactManifest {
1805        schema_version: 1,
1806        source_revision: source_revision(),
1807        files,
1808    };
1809    let manifest_path = artifact_dir.join("manifest.json");
1810    write_json_new(&manifest_path, &manifest)?;
1811    let (_, manifest_digest) = sha256_file(&manifest_path)?;
1812    let digest_path = artifact_dir.join("manifest.sha256");
1813    let mut digest_file = OpenOptions::new()
1814        .write(true)
1815        .create_new(true)
1816        .open(&digest_path)?;
1817    writeln!(digest_file, "{manifest_digest}  manifest.json")?;
1818    digest_file.flush()?;
1819
1820    for path in paths.iter().chain([&manifest_path, &digest_path]) {
1821        let mut permissions = fs::metadata(path)?.permissions();
1822        permissions.set_readonly(true);
1823        fs::set_permissions(path, permissions)?;
1824    }
1825    let mut directory_permissions = fs::metadata(artifact_dir)?.permissions();
1826    directory_permissions.set_readonly(true);
1827    fs::set_permissions(artifact_dir, directory_permissions)?;
1828    Ok(manifest_digest)
1829}

1831fn sha256_file(path: &Path) -> Result<(u64, String), Box<dyn Error>> {
1832    let mut file = File::open(path)?;
1833    let mut digest = Sha256::new();
1834    let mut bytes = 0u64;
1835    let mut buffer = [0u8; 64 * 1024];
1836    loop {
1837        let read = file.read(&mut buffer)?;
1838        if read == 0 {
1839            break;
1840        }
1841        bytes = bytes
1842            .checked_add(u64::try_from(read)?)
1843            .ok_or("artifact byte count overflowed")?;
1844        digest.update(&buffer[..read]);
1845    }
1846    Ok((bytes, format!("{:x}", digest.finalize())))
1847}
