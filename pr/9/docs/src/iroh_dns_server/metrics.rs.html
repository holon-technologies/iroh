3use iroh_metrics::{Counter, Gauge, MetricsGroup};
4
5/// Counters exposed by iroh-dns-server.
6#[derive(Debug, Default, MetricsGroup)]
7#[metrics(name = "dns_server")]
8#[non_exhaustive]
9pub struct Metrics {
10    /// Number of pkarr relay puts that updated the stored packet.
11    pub pkarr_publish_update: Counter,
12    /// Number of pkarr relay puts that did not change the stored packet.
13    pub pkarr_publish_noop: Counter,
14    /// Total number of DNS requests across all transports.
15    pub dns_requests: Counter,
16    /// Number of DNS requests received over UDP.
17    pub dns_requests_udp: Counter,
18    /// Number of DNS requests received over HTTPS (DoH).
19    pub dns_requests_https: Counter,
20    /// Current number of admitted UDP DNS request tasks.
21    pub dns_udp_requests_active: Gauge,
22    /// Number of UDP DNS requests dropped because request capacity was full.
23    pub dns_udp_requests_rejected: Counter,
24    /// Current number of admitted DNS TCP connection tasks.
25    pub dns_tcp_connections_active: Gauge,
26    /// Number of DNS TCP connections closed because connection capacity was full.
27    pub dns_tcp_connections_rejected: Counter,
28    /// Number of DNS lookups that returned at least one answer.
29    pub dns_lookup_success: Counter,
30    /// Number of DNS lookups that returned no answers.
31    pub dns_lookup_notfound: Counter,
32    /// Number of DNS lookups that failed with an error.
33    pub dns_lookup_error: Counter,
34    /// Number of HTTP requests served.
35    pub http_requests: Counter,
36    /// Number of HTTP requests that returned a 2xx status code.
37    pub http_requests_success: Counter,
38    /// Number of HTTP requests that returned a non-2xx status code.
39    pub http_requests_error: Counter,
40    /// Cumulative duration of all HTTP requests, in milliseconds.
41    pub http_requests_duration_ms: Counter,
42    /// Current number of admitted HTTP and HTTPS connections.
43    pub http_connections_active: Gauge,
44    /// Number of HTTP(S) connections rejected because capacity was full.
45    pub http_connections_rejected_capacity: Counter,
46    /// Number of HTTP(S) connections rejected by the global accept rate.
47    pub http_connections_rejected_rate: Counter,
48    /// Current number of admitted HTTP requests.
49    pub http_requests_active: Gauge,
50    /// Number of HTTP requests rejected because request capacity was full.
51    pub http_requests_rejected_capacity: Counter,
52    /// Number of HTTP requests rejected by the per-IP rate policy.
53    pub http_requests_rejected_rate: Counter,
54    /// Current bounded per-IP rate-limit entry count.
55    pub http_rate_limit_entries: Gauge,
56    /// Number of signed packets newly inserted into the store.
57    pub store_packets_inserted: Counter,
58    /// Number of signed packets removed from the store.
59    ///
60    /// Currently always at 0 because the removal API is not used.
61    pub store_packets_removed: Counter,
62    /// Number of times an existing signed packet was replaced by a newer one.
63    pub store_packets_updated: Counter,
64    /// Number of signed packets removed by the eviction task.
65    pub store_packets_expired: Counter,
66    /// Number of corrupt persistent rows detected.
67    pub store_corrupt_rows: Counter,
68    /// Number of packet-store actor or eviction failures.
69    pub store_background_failures: Counter,
70    /// Current number of zones in the main cache
71    pub cache_zones: Gauge,
72    /// Current number of zones in the DHT cache
73    pub cache_zones_dht: Gauge,
74}

76impl hickory_server::server::AdmissionObserver for Metrics {
77    fn admitted(&self, kind: hickory_server::server::AdmissionKind) {
78        match kind {
79            hickory_server::server::AdmissionKind::UdpRequest => {
80                self.dns_udp_requests_active.inc();
81            }
82            hickory_server::server::AdmissionKind::TcpConnection => {
83                self.dns_tcp_connections_active.inc();
84            }
85        }
86    }
87
88    fn released(&self, kind: hickory_server::server::AdmissionKind) {
89        match kind {
90            hickory_server::server::AdmissionKind::UdpRequest => {
91                self.dns_udp_requests_active.dec();
92            }
93            hickory_server::server::AdmissionKind::TcpConnection => {
94                self.dns_tcp_connections_active.dec();
95            }
96        }
97    }
98
99    fn rejected(&self, rejection: hickory_server::server::AdmissionRejection) {
100        match rejection {
101            hickory_server::server::AdmissionRejection::UdpRequestCapacity => {
102                self.dns_udp_requests_rejected.inc();
103            }
104            hickory_server::server::AdmissionRejection::TcpConnectionCapacity => {
105                self.dns_tcp_connections_rejected.inc();
106            }
107        }
108    }
109}
