5/// A running Relay + QAD server.
6///
7/// This is a full Relay server, including QAD, Relay and various associated HTTP services.
8///
9/// Dropping this will stop the server.
10#[derive(Debug)]
11pub struct Server {
12    /// The address of the HTTP server, if configured.
13    http_addr: Option<SocketAddr>,
14    /// The address of the HTTPS server, if the relay server is using TLS.
15    ///
16    /// If the Relay server is not using TLS then it is served from the
17    /// [`Server::http_addr`].
18    https_addr: Option<SocketAddr>,
19    /// The address of the QUIC server, if configured.
20    quic_addr: Option<SocketAddr>,
21    /// Handle to the relay server.
22    relay_handle: Option<http_server::ServerHandle>,
23    /// Handle to the relay service for runtime control.
24    relay_service: Option<http_server::RelayService>,
25    /// Handle to the quic server.
26    quic_handle: Option<QuicServerHandle>,
27    /// The main task running the server.
28    supervisor: AbortOnDropHandle<Result<(), SupervisorError>>,
29    metrics: RelayMetrics,
30    metrics_server: Option<metrics_server::MetricsServer>,
31}

33/// Server spawn errors
34#[allow(missing_docs)]
35#[stack_error(derive, add_meta, std_sources)]
36#[non_exhaustive]
37pub enum SpawnError {
38    #[error("Unable to get local address")]
39    LocalAddr { source: std::io::Error },
40    #[error("Failed to bind QAD listener")]
41    QuicSpawn { source: QuicSpawnError },
42    #[error("Failed to parse TLS header")]
43    TlsHeaderParse { source: InvalidHeaderValue },
44    #[error("Failed to bind TcpListener")]
45    BindTlsListener { source: std::io::Error },
46    #[error("Failed to build ACME client TLS config")]
47    AcmeClientTlsConfig {
48        #[error(std_err)]
49        source: std::io::Error,
50    },
51    #[error("No local address")]
52    NoLocalAddr { source: std::io::Error },
53    #[error("Failed to bind server socket to {addr}")]
54    BindTcpListener {
55        source: std::io::Error,
56        addr: SocketAddr,
57    },
58    #[error("Error starting metrics server")]
59    Metrics {
60        #[error(std_err)]
61        source: std::io::Error,
62    },
63    #[error("Invalid relay admission policy")]
64    AdmissionPolicy { source: AdmissionPolicyError },
65}

86    /// Starts the server.
87    pub async fn spawn(config: ServerConfig) -> Result<Self, SpawnError> {
88        let mut tasks = JoinSet::new();
89
90        let relay_config = config
91            .relay
92            .map(|relay| {
93                AdmissionPolicy::try_from(&relay.limits)
94                    .map(|policy| (relay, policy))
95                    .map_err(|err| e!(SpawnError::AdmissionPolicy, err))
96            })
97            .transpose()?;
98
99        let metrics = RelayMetrics::default();
100
101        #[cfg(not(feature = "metrics"))]
102        let metrics_server = None;
103
104        #[cfg(feature = "metrics")]
105        let metrics_server = if let Some(addr) = config.metrics_addr {
106            debug!("Starting metrics server");
107            let mut registry = iroh_metrics::Registry::default();
108            registry.register_all(&metrics);
109            let server = metrics_server::MetricsServer::spawn(addr, Arc::new(registry))
110                .await
111                .map_err(|err| e!(SpawnError::Metrics, err))?;
112            Some(server)
113        } else {
114            None
115        };
116
117        let (relay_server, http_addr, tls_config) = match relay_config {
118            Some((relay_config, admission_policy)) => {
119                debug!("Starting Relay server");
120                let captive_portal_capacity = admission_policy.max_pending_establishments;
121                let mut headers = HeaderMap::new();
122                for (name, value) in TLS_HEADERS.iter() {
123                    headers.insert(
124                        *name,
125                        value
126                            .parse()
127                            .map_err(|err| e!(SpawnError::TlsHeaderParse, err))?,
128                    );
129                }
130                let relay_bind_addr = match relay_config.tls {
131                    Some(ref tls) => tls.https_bind_addr,
132                    None => relay_config.http_bind_addr,
133                };
134                let key_cache_capacity = relay_config
135                    .key_cache_capacity
136                    .unwrap_or(DEFAULT_KEY_CACHE_CAPACITY);
137                let mut builder = http_server::ServerBuilder::new(relay_bind_addr)
138                    .admission_policy(admission_policy)
139                    .metrics(metrics.server.clone())
140                    .headers(headers)
141                    .key_cache_capacity(key_cache_capacity)
142                    .access(relay_config.access)
143                    .request_handler(Method::GET, "/", Box::new(root_handler))
144                    .request_handler(Method::GET, "/index.html", Box::new(root_handler))
145                    .request_handler(Method::GET, RELAY_PROBE_PATH, Box::new(probe_handler))
146                    .request_handler(Method::GET, "/robots.txt", Box::new(robots_handler))
147                    .request_handler(Method::GET, "/healthz", Box::new(healthz_handler));
148                if let Some(cfg) = relay_config.limits.client_rx {
149                    builder = builder.client_rx_ratelimit(cfg);
150                }
151                let (http_addr, tls_config) = match relay_config.tls {
152                    Some(tls_config) => {
153                        let server_tls_config = match tls_config.cert {
154                            #[cfg(feature = "server-acme")]
155                            CertConfig::LetsEncrypt {
156                                acme_config,
157                                server_config_builder,
158                            } => {
159                                let cache = acme_config.cache_path.map(BoundedAcmeCache::new);
160                                let crypto_provider =
161                                    server_config_builder.crypto_provider().clone();
162                                let client_tls_config = acme_config
163                                    .tls_config
164                                    .client_config(crypto_provider.clone())
165                                    .map_err(|err| e!(SpawnError::AcmeClientTlsConfig, err))?;
166                                let config = tokio_rustls_acme::AcmeConfig::new_with_client_config(
167                                    acme_config.domains,
168                                    Arc::new(client_tls_config),
169                                )
170                                .contact(acme_config.contact)
171                                .directory(acme_config.directory_url)
172                                .cache_option(cache);
173                                let mut state = config.state();
174                                let resolver = state.resolver().clone();
175                                let challenge_config =
176                                    state.challenge_rustls_config_with_provider(crypto_provider);
177                                let server_config =
178                                    server_config_builder.with_cert_resolver(resolver);
179                                let acceptor =
180                                    http_server::TlsAcceptor::LetsEncrypt { challenge_config };
181                                tasks.spawn(
182                                    async move {
183                                        while let Some(event) = state.next().await {
184                                            match event {
185                                                Ok(ok) => debug!("acme event: {ok:?}"),
186                                                Err(err) => error!("error: {err:?}"),
187                                            }
188                                        }
189                                        Err(e!(SupervisorError::AcmeEventStreamFinished))
190                                    }
191                                    .instrument(info_span!("acme")),
192                                );
193                                http_server::TlsConfig {
194                                    config: Arc::new(server_config),
195                                    acceptor,
196                                }
197                            }
198                            CertConfig::Manual { server_config } => {
199                                let server_config = Arc::new(server_config);
200                                let acceptor =
201                                    tokio_rustls::TlsAcceptor::from(server_config.clone());
202                                let acceptor = http_server::TlsAcceptor::Manual(acceptor);
203                                http_server::TlsConfig {
204                                    config: server_config,
205                                    acceptor,
206                                }
207                            }
208                        };
209                        builder = builder.tls_config(Some(server_tls_config.clone()));
210
211                        // Some services always need to be served over HTTP without TLS.  Run
212                        // these standalone.
213                        let http_listener = TcpListener::bind(&relay_config.http_bind_addr)
214                            .await
215                            .map_err(|err| e!(SpawnError::BindTlsListener, err))?;
216                        let http_addr = http_listener
217                            .local_addr()
218                            .map_err(|err| e!(SpawnError::NoLocalAddr, err))?;
219                        let captive_metrics = metrics.server.clone();
220                        tasks.spawn(
221                            async move {
222                                run_captive_portal_service(
223                                    http_listener,
224                                    captive_portal_capacity,
225                                    captive_metrics,
226                                )
227                                .await;
228                                Ok(())
229                            }
230                            .instrument(info_span!("http-service", addr = %http_addr)),
231                        );
232                        (Some(http_addr), Some(server_tls_config))
233                    }
234                    None => {
235                        // If running Relay without TLS add the plain HTTP server directly
236                        // to the Relay server.
237                        builder = builder.request_handler(
238                            Method::GET,
239                            "/generate_204",
240                            Box::new(serve_no_content_handler),
241                        );
242                        (None, None)
243                    }
244                };
245                let relay_server = builder.spawn().await?;
246                (Some(relay_server), http_addr, tls_config)
247            }
248            None => (None, None, None),
249        };
250        // If http_addr is Some then relay_server is serving HTTPS.  If http_addr is None
251        // relay_server is serving HTTP, including the /generate_204 service.
252        let relay_addr = relay_server.as_ref().map(|srv| srv.addr());
253        let relay_handle = relay_server.as_ref().map(|srv| srv.handle());
254        let relay_service = relay_server.as_ref().map(|srv| srv.service().clone());
255
256        let quic_server = match config.quic {
257            Some(quic_config) => {
258                debug!("Starting QUIC server {}", quic_config.bind_addr);
259                let server_config = quic_config
260                    .server_config
261                    .or(tls_config.map(|config| (*config.config).clone()))
262                    .ok_or_else(|| {
263                        e!(SpawnError::QuicSpawn, e!(QuicSpawnError::TlsNotConfigured))
264                    })?;
265                Some(
266                    QuicServer::spawn(
267                        quic_config.bind_addr,
268                        server_config,
269                        quic_config.max_connections,
270                        metrics.server.clone(),
271                    )
272                    .map_err(|err| e!(SpawnError::QuicSpawn, err))?,
273                )
274            }
275            None => None,
276        };
277        let quic_addr = quic_server.as_ref().map(|srv| srv.bind_addr());
278        let quic_handle = quic_server.as_ref().map(|srv| srv.handle());
279
280        let task = tokio::spawn(relay_supervisor(tasks, relay_server, quic_server));
281
282        Ok(Self {
283            http_addr: http_addr.or(relay_addr),
284            https_addr: http_addr.and(relay_addr),
285            quic_addr,
286            relay_handle,
287            relay_service,
288            quic_handle,
289            supervisor: AbortOnDropHandle::new(task),
290            metrics,
291            metrics_server,
292        })
293    }

295    /// Requests graceful shutdown.
296    ///
297    /// Returns once all server tasks have stopped.
298    pub async fn shutdown(self) -> Result<(), SupervisorError> {
299        // Only the Relay server and QUIC server need shutting down, the supervisor will abort the tasks in
300        // the JoinSet when the server terminates.
301        if let Some(handle) = self.relay_handle {
302            handle.shutdown();
303        }
304        if let Some(handle) = self.quic_handle {
305            handle.shutdown();
306        }
307        if let Some(server) = self.metrics_server {
308            server.shutdown().await;
309        }
310        self.supervisor.await?
311    }
312
313    /// Waits for the server's supervisor task to finish.
314    ///
315    /// Returns the exit result of the supervisor task. Unlike [`Self::shutdown`], this does
316    /// *not* request shutdown, it only waits for the server to terminate on its own (for
317    /// example, after an internal error or because the supervisor was aborted from
318    /// elsewhere). The outer [`JoinError`] is only produced if the supervisor task itself
319    /// panics or is aborted.
320    pub async fn join(&mut self) -> Result<Result<(), SupervisorError>, JoinError> {
321        (&mut self.supervisor).await
322    }

374/// Supervisor for the relay server tasks.
375///
376/// As soon as one of the tasks exits, all other tasks are stopped and the server stops.
377/// The supervisor finishes once all tasks are finished.
378#[instrument(skip_all)]
379async fn relay_supervisor(
380    mut tasks: JoinSet<Result<(), SupervisorError>>,
381    mut relay_http_server: Option<http_server::Server>,
382    mut quic_server: Option<QuicServer>,
383) -> Result<(), SupervisorError> {
384    let quic_enabled = quic_server.is_some();
385    let mut quic_fut = match quic_server {
386        Some(ref mut server) => n0_future::Either::Left(server.task_handle()),
387        None => n0_future::Either::Right(n0_future::future::pending()),
388    };
389    let relay_enabled = relay_http_server.is_some();
390    let mut relay_fut = match relay_http_server {
391        Some(ref mut server) => n0_future::Either::Left(server.task_handle()),
392        None => n0_future::Either::Right(n0_future::future::pending()),
393    };
394    let res = tokio::select! {
395        biased;
396        Some(ret) = tasks.join_next() => ret,
397        ret = &mut quic_fut, if quic_enabled => ret.map(Ok),
398        ret = &mut relay_fut, if relay_enabled => ret.map(Ok),
399        else => Ok(Err(e!(SupervisorError::NoRelayServicesEnabled))),
400    };
401    let ret = match res {
402        Ok(Ok(())) => {
403            debug!("Task exited");
404            Ok(())
405        }
406        Ok(Err(err)) => {
407            error!(%err, "Task failed");
408            Err(err)
409        }
410        Err(err) => {
411            if let Ok(panic) = err.try_into_panic() {
412                error!("Task panicked");
413                std::panic::resume_unwind(panic);
414            }
415            debug!("Task cancelled");
416            Err(e!(SupervisorError::TaskCancelled))
417        }
418    };
419
420    // Ensure the HTTP server terminated, there is no harm in calling this after it is
421    // already shut down.
422    if let Some(server) = relay_http_server {
423        server.shutdown();
424    }
425
426    // Ensure the QUIC server is closed
427    if let Some(server) = quic_server {
428        server.shutdown().await;
429    }
430
431    // Stop all remaining tasks
432    tasks.shutdown().await;
433
434    ret
