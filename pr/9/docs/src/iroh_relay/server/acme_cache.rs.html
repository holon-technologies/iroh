12/// Maximum size of one cached ACME account or certificate.
13const MAX_ACME_CACHE_FILE_BYTES: usize = 1024 * 1024;
14const MAX_ACME_CACHE_FILE_READ_BYTES: u64 = 1024 * 1024 + 1;
15
16/// Filesystem-backed ACME cache with an explicit per-file input and output limit.
17#[derive(Clone, Debug)]
18pub struct BoundedAcmeCache {
19    root: PathBuf,
20}
21
22impl BoundedAcmeCache {
23    /// Creates a bounded cache rooted at `directory`.
24    pub fn new(directory: impl Into<PathBuf>) -> Self {
25        Self {
26            root: directory.into(),
27        }
28    }
29
30    fn cert_path(&self, domains: &[String], directory_url: &str) -> PathBuf {
31        self.root
32            .join(cache_file_name("cached_cert_", domains, directory_url))
33    }
34
35    fn account_path(&self, contact: &[String], directory_url: &str) -> PathBuf {
36        self.root
37            .join(cache_file_name("cached_account_", contact, directory_url))
38    }
39
40    async fn read_if_exists(&self, path: &Path) -> io::Result<Option<Vec<u8>>> {
41        let file = match tokio::fs::File::open(path).await {
42            Ok(file) => file,
43            Err(error) if error.kind() == io::ErrorKind::NotFound => return Ok(None),
44            Err(error) => return Err(error),
45        };
46        let capacity = usize::try_from(
47            file.metadata()
48                .await?
49                .len()
50                .min(MAX_ACME_CACHE_FILE_READ_BYTES),
51        )
52        .map_err(|_| io::Error::new(io::ErrorKind::InvalidData, "ACME file length is invalid"))?;
53        let mut reader = file.take(MAX_ACME_CACHE_FILE_READ_BYTES);
54        let mut bytes = Vec::with_capacity(capacity);
55        reader.read_to_end(&mut bytes).await?;
56        if bytes.len() > MAX_ACME_CACHE_FILE_BYTES {
57            return Err(oversized_error(path, bytes.len()));
58        }
59        Ok(Some(bytes))
60    }
61
62    async fn write(&self, path: &Path, bytes: &[u8]) -> io::Result<()> {
63        if bytes.len() > MAX_ACME_CACHE_FILE_BYTES {
64            return Err(oversized_error(path, bytes.len()));
65        }
66        tokio::fs::create_dir_all(&self.root).await?;
67        tokio::fs::write(path, bytes).await
68    }
69}

71#[async_trait]
72impl CertCache for BoundedAcmeCache {
73    type EC = io::Error;
74
75    async fn load_cert(
76        &self,
77        domains: &[String],
78        directory_url: &str,
79    ) -> Result<Option<Vec<u8>>, Self::EC> {
80        self.read_if_exists(&self.cert_path(domains, directory_url))
81            .await
82    }
83
84    async fn store_cert(
85        &self,
86        domains: &[String],
87        directory_url: &str,
88        cert: &[u8],
89    ) -> Result<(), Self::EC> {
90        self.write(&self.cert_path(domains, directory_url), cert)
91            .await
92    }
93}

119fn cache_file_name(prefix: &str, identifiers: &[String], directory_url: &str) -> String {
120    let mut digest = Sha256::new();
121    for identifier in identifiers {
122        digest.update(identifier.as_bytes());
123        digest.update([0]);
124    }
125    digest.update(directory_url.as_bytes());
126    format!(
127        "{prefix}{}",
128        BASE64URL_NOPAD.encode(digest.finalize().as_ref())
129    )
130}
131
132fn oversized_error(path: &Path, actual: usize) -> io::Error {
133    io::Error::new(
134        io::ErrorKind::InvalidData,
135        format!(
136            "ACME cache file {} is {actual} bytes and exceeds {MAX_ACME_CACHE_FILE_BYTES} bytes",
137            path.display()
138        ),
139    )
140}

143mod tests {
144    use tokio_rustls_acme::caches::DirCache;

148    #[tokio::test]
149    async fn cache_uses_upstream_file_names_and_accepts_the_exact_limit() {
150        let directory = tempfile::tempdir().expect("temporary ACME cache");
151        let cache = BoundedAcmeCache::new(directory.path().to_path_buf());
152        let upstream = DirCache::new(directory.path().to_path_buf());
153        let domains = ["example.com".to_owned()];
154        let contact = ["mailto:test@example.com".to_owned()];
155        let directory_url = "https://acme.invalid";
156        let exact_limit = vec![7_u8; MAX_ACME_CACHE_FILE_BYTES];
157
158        upstream
159            .store_cert(&domains, directory_url, &exact_limit)
160            .await
161            .expect("store through upstream cache");
162        assert_eq!(
163            cache
164                .load_cert(&domains, directory_url)
165                .await
166                .expect("load through bounded cache"),
167            Some(exact_limit.clone())
168        );
169
170        cache
171            .store_account(&contact, directory_url, &exact_limit)
172            .await
173            .expect("store through bounded cache");
174        assert_eq!(
175            upstream
176                .load_account(&contact, directory_url)
177                .await
178                .expect("load through upstream cache"),
179            Some(exact_limit)
180        );
181    }

183    #[tokio::test]
184    async fn cache_rejects_the_first_byte_over_the_limit_on_store_and_load() {
185        let directory = tempfile::tempdir().expect("temporary ACME cache");
186        let cache = BoundedAcmeCache::new(directory.path().to_path_buf());
187        let oversized = vec![0_u8; MAX_ACME_CACHE_FILE_BYTES + 1];
188
189        let store_error = cache
190            .store_cert(
191                &["example.com".to_owned()],
192                "https://acme.invalid",
193                &oversized,
194            )
195            .await
196            .expect_err("oversized certificate cache write must be rejected");
197        assert_eq!(store_error.kind(), std::io::ErrorKind::InvalidData);
198
199        let account_path = cache.account_path(
200            &["mailto:test@example.com".to_owned()],
201            "https://acme.invalid",
202        );
203        std::fs::write(&account_path, oversized).expect("oversized cache fixture");
204        let load_error = cache
205            .load_account(
206                &["mailto:test@example.com".to_owned()],
207                "https://acme.invalid",
208            )
209            .await
210            .expect_err("oversized account cache read must be rejected");
211        assert_eq!(load_error.kind(), std::io::ErrorKind::InvalidData);
212    }
