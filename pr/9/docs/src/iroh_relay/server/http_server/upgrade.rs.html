3/// WebSocket GUID needed for accepting websocket connections, see RFC 6455 (<https://www.rfc-editor.org/rfc/rfc6455>) section 1.3
4const SEC_WEBSOCKET_ACCEPT_GUID: &[u8] = b"258EAFA5-E914-47DA-95CA-C5AB0DC85B11";
5
6/// Timeout for a connection to finish the TLS and WebSocket upgrade handshakes.
7///
8/// The connection is aborted if the connection does not complete the TLS handshake
9/// and establishes relay protocol WebSocket stream within this timeout.
10pub(super) const ESTABLISH_TIMEOUT: Duration = Duration::from_secs(30);
11
12/// Derives the accept key for WebSocket handshake according to RFC 6455.
13/// Takes the client's Sec-WebSocket-Key value and returns the calculated accept key.
14pub(super) fn derive_accept_key(client_key: &HeaderValue) -> String {
15    use sha1::Digest;
16
17    let mut sha1 = sha1::Sha1::new();
18    sha1.update(client_key.as_bytes());
19    sha1.update(SEC_WEBSOCKET_ACCEPT_GUID);
20    data_encoding::BASE64.encode(&sha1.finalize())
21}

23#[stack_error(derive, add_meta)]
24pub(super) enum RelayUpgradeReqError {
25    #[error("missing header: {header}")]
26    MissingHeader { header: http::HeaderName },
27    #[error("invalid header value for {header}: {details}")]
28    InvalidHeader {
29        header: http::HeaderName,
30        details: String,
31    },
32    #[error(
33        "invalid header value for {SEC_WEBSOCKET_VERSION}: unsupported websocket version, only supporting {SUPPORTED_WEBSOCKET_VERSION}"
34    )]
35    UnsupportedWebsocketVersion,
36    #[error(
37        "invalid header value for {SEC_WEBSOCKET_PROTOCOL}: unsupported relay version: we support {we_support} but you only provide {you_support}"
38    )]
39    UnsupportedRelayVersion {
40        we_support: String,
41        you_support: String,
42    },
43}

45impl RelayServiceWithNotify {
46    pub(super) fn build_response(&self) -> http::response::Builder {
47        let mut res = Response::builder();
48        for (key, value) in self.service.0.headers.iter() {
49            res = res.header(key, value);
50        }
51        res
52    }
53
54    /// Upgrades the HTTP connection to the relay protocol, runs relay client.
55    pub(super) fn handle_relay_ws_upgrade(
56        &self,
57        mut req: Request<Incoming>,
58    ) -> Result<Response<BytesBody>, RelayUpgradeReqError> {
59        fn expect_header(
60            req: &Request<Incoming>,
61            header: http::HeaderName,
62        ) -> Result<&HeaderValue, RelayUpgradeReqError> {
63            req.headers()
64                .get(&header)
65                .ok_or_else(|| e!(RelayUpgradeReqError::MissingHeader { header }))
66        }
67
68        let upgrade_header = expect_header(&req, UPGRADE)?;
69        ensure!(
70            upgrade_header == HeaderValue::from_static(WEBSOCKET_UPGRADE_PROTOCOL),
71            RelayUpgradeReqError::InvalidHeader {
72                header: UPGRADE,
73                details: format!("value must be {WEBSOCKET_UPGRADE_PROTOCOL}")
74            }
75        );
76
77        let key = expect_header(&req, SEC_WEBSOCKET_KEY)?.clone();
78        let version = expect_header(&req, SEC_WEBSOCKET_VERSION)?.clone();
79
80        ensure!(
81            version.as_bytes() == SUPPORTED_WEBSOCKET_VERSION.as_bytes(),
82            RelayUpgradeReqError::UnsupportedWebsocketVersion
83        );
84
85        let subprotocols = expect_header(&req, SEC_WEBSOCKET_PROTOCOL)?
86            .to_str()
87            .ok()
88            .ok_or_else(|| {
89                e!(RelayUpgradeReqError::InvalidHeader {
90                    header: SEC_WEBSOCKET_PROTOCOL,
91                    details: "header value is not ascii".to_string()
92                })
93            })?;
94        let protocol_version = subprotocols
95            .split(",")
96            .map(|s| s.trim())
97            .filter_map(ProtocolVersion::match_from_str)
98            .max()
99            .ok_or_else(|| {
100                e!(RelayUpgradeReqError::UnsupportedRelayVersion {
101                    we_support: ProtocolVersion::all_joined(),
102                    you_support: subprotocols.to_string()
103                })
104            })?;
105
106        // Setup a future that will eventually receive the upgraded
107        // connection and talk a new protocol, and spawn the future
108        // into the runtime.
109        //
110        // Note: This can't possibly be fulfilled until the 101 response
111        // is returned below, so it's better to spawn this future instead
112        // waiting for it to complete to then return a response.
113        tokio::task::spawn({
114            let this = self.clone();
115            async move {
116                match hyper::upgrade::on(&mut req).await {
117                    Ok(upgraded) => {
118                        let (parts, _) = req.into_parts();
119                        if let Err(err) = this
120                            .service
121                            .0
122                            .relay_connection_handler(upgraded, parts, protocol_version)
123                            .await
124                        {
125                            warn!("error accepting upgraded connection: {err:#}",);
126                        } else {
127                            // We have passed the connection to the relay protocol handler,
128                            // thus we trigger the on_establish notification so that timeouts
129                            // on the upper layer will be cleared.
130                            release_establishment_lease(&this.establishment_lease);
131                            this.on_establish.notify_waiters();
132                            debug!("upgraded connection completed");
133                        };
134                    }
135                    Err(err) => warn!("upgrade error: {err:#}"),
136                }
137            }
138            .instrument(info_span!("handler"))
139        });
140
141        // Now return a 101 Response saying we agree to the upgrade to the
142        // websocket upgrade protocol
143        Ok(self
144            .build_response()
145            .status(StatusCode::SWITCHING_PROTOCOLS)
146            .header(
147                UPGRADE,
148                HeaderValue::from_static(WEBSOCKET_UPGRADE_PROTOCOL),
149            )
150            .header(SEC_WEBSOCKET_ACCEPT, derive_accept_key(&key))
151            .header(SEC_WEBSOCKET_PROTOCOL, protocol_version.to_header_value())
152            .header(CONNECTION, "upgrade")
153            .body(body_full("switching to websocket protocol"))
154            .expect("valid body"))
155    }
