23/// Combines [`RelayService`] with a notification token.
24///
25/// This struct implements [`Service`]. Note that the service has to be called with hyper's `io`
26/// argument set to [`MaybeTlsStream`] wrapped by [`hyper_util::rt::TokioIo`], otherwise handling
27/// WebSocket requests at `/relay` will fail at runtime with [`ConnectionHandlerError::DowncastUpgrade`].
28///
29/// The notification token is triggered once the relay connection is fully established. It can be used
30/// to cancel a timeout aborting the TCP connection if no upgrade request is received in some time.
31///
32/// ## Example
33///
34/// ```no_run
35/// # use std::sync::Arc;
36/// # use http::HeaderMap;
37/// # use hyper::server::conn::http1;
38/// # use hyper_util::rt::TokioIo;
39/// # use tokio::{net::TcpListener, sync::Notify};
40/// # use iroh_relay::{
41/// #     KeyCache,
42/// #     server::{
43/// #         AllowAll, Metrics,
44/// #         http_server::{Handlers, RelayService, RelayServiceWithNotify},
45/// #         streams::MaybeTlsStream
46/// #     },
47/// # };
48/// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
49/// let service = RelayService::new(
50///     Handlers::default(),
51///     HeaderMap::new(),
52///     None,
53///     KeyCache::new(1024),
54///     Arc::new(AllowAll),
55///     Arc::new(Metrics::default()),
56/// );
57/// let service = RelayServiceWithNotify::new(service, Arc::new(Notify::new()));
58///
59/// let listener = TcpListener::bind("127.0.0.1:0").await?;
60/// let (stream, _peer) = listener.accept().await?;
61/// // Wrap the TCP stream in `MaybeTlsStream`, otherwise the relay WebSocket handler will error at runtime
62/// // for all WebSocket requests to `/relay`.
63/// let stream = MaybeTlsStream::Plain(stream);
64/// http1::Builder::new()
65///     .serve_connection(TokioIo::new(stream), service)
66///     .with_upgrades()
67///     .await?;
68/// # Ok(())
69/// # }
70/// ```
71#[derive(Debug, Clone)]
72pub struct RelayServiceWithNotify {
73    pub(super) service: RelayService,
74    pub(super) on_establish: Arc<Notify>,
75    pub(super) establishment_lease: Arc<Mutex<Option<EstablishmentLease>>>,
76}

104impl Service<Request<Incoming>> for RelayServiceWithNotify {
105    type Response = Response<BytesBody>;
106    type Error = HyperError;
107    type Future = std::future::Ready<Result<Self::Response, Self::Error>>;
108
109    fn call(&self, req: Request<Incoming>) -> Self::Future {
110        // Create a client if the request hits the relay endpoint.
111        if matches!(
112            (req.method(), req.uri().path()),
113            (&hyper::Method::GET, RELAY_PATH)
114        ) {
115            let response = match self.handle_relay_ws_upgrade(req) {
116                Ok(response) => Ok(response),
117                // It's convention to send back the version(s) we *do* support
118                Err(e @ RelayUpgradeReqError::UnsupportedWebsocketVersion { .. }) => self
119                    .build_response()
120                    .status(StatusCode::BAD_REQUEST)
121                    .header(SEC_WEBSOCKET_VERSION, SUPPORTED_WEBSOCKET_VERSION)
122                    .body(body_full(e.to_string())),
123                Err(e) => self
124                    .build_response()
125                    .status(StatusCode::BAD_REQUEST)
126                    .body(body_full(e.to_string())),
127            }
128            .map_err(Into::into);
129            return std::future::ready(response);
130        }
131        // Otherwise handle the relay connection as normal.
132
133        // Check all other possible endpoints.
134        let uri = req.uri().clone();
135        if let Some(handler) = self
136            .service
137            .0
138            .handlers
139            .get(&(req.method().clone(), uri.path()))
140        {
141            let response = handler(req, self.service.0.default_response());
142            return std::future::ready(response);
143        }
144
145        // Otherwise return 404
146        let response = self
147            .service
148            .0
149            .not_found_fn(req, self.service.0.default_response());
150        std::future::ready(response)
151    }
152}

154impl Inner {
155    fn default_response(&self) -> ResponseBuilder {
156        let mut response = Response::builder();
157        for (key, value) in self.headers.iter() {
158            response = response.header(key.clone(), value.clone());
159        }
160        response
161    }
162
163    fn not_found_fn(
164        &self,
165        _req: Request<Incoming>,
166        mut res: ResponseBuilder,
167    ) -> HyperResult<Response<BytesBody>> {
168        for (k, v) in self.headers.iter() {
169            res = res.header(k.clone(), v.clone());
170        }
171        let body = body_full("Not Found");
172        let r = res.status(StatusCode::NOT_FOUND).body(body)?;
173        HyperResult::Ok(r)
174    }
175
176    /// The server HTTP handler to do HTTP upgrades.
177    ///
178    /// This handler runs while doing the connection upgrade handshake.  Once the connection
179    /// is upgraded it sends the stream to the relay server which takes it over.  After
180    /// having sent off the connection this handler returns.
181    pub(super) async fn relay_connection_handler(
182        &self,
183        upgraded: Upgraded,
184        request_parts: http::request::Parts,
185        protocol_version: ProtocolVersion,
186    ) -> Result<(), ConnectionHandlerError> {
187        debug!("relay_connection upgraded");
188        let (io, read_buf) = downcast_upgrade(upgraded)?;
189        if !read_buf.is_empty() {
190            return Err(e!(ConnectionHandlerError::BufferNotEmpty { buf: read_buf }));
191        }
192
193        self.accept(io, request_parts, protocol_version).await?;
194        Ok(())
195    }
196
197    /// Adds a new connection to the server and serves it.
198    ///
199    /// Will error if it takes too long (10 sec) to write or read to the connection, if there is
200    /// some read or write error to the connection,  if the server is meant to verify clients,
201    /// and is unable to verify this one, or if there is some issue communicating with the server.
202    ///
203    /// The provided [`AsyncRead`] and [`AsyncWrite`] must be already connected to the connection.
204    ///
205    /// [`AsyncRead`]: tokio::io::AsyncRead
206    /// [`AsyncWrite`]: tokio::io::AsyncWrite
207    pub(super) async fn accept(
208        &self,
209        io: MaybeTlsStream,
210        request_parts: http::request::Parts,
211        protocol_version: ProtocolVersion,
212    ) -> Result<(), AcceptError> {
213        trace!("accept: start");
214
215        // Set the socket to NO_DELAY.
216        io.disable_nagle();
217
218        let io = RateLimited::from_watcher(io, self.rate_limit.subscribe(), self.metrics.clone())
219            .map_err(|err| e!(AcceptError::RateLimitingMisconfigured, err))?;
220
221        // Create a server builder with default config
222        let websocket = tokio_websockets::ServerBuilder::new()
223            .limits(tokio_websockets::Limits::default().max_payload_len(Some(MAX_FRAME_SIZE)))
224            // Serve will create a WebSocketStream on an already upgraded connection
225            .serve(io);
226
227        let io = WsBytesFramed { io: websocket };
228
229        self.accept_framed(io, request_parts, protocol_version)
230            .await
231    }
232
233    /// Authenticates, authorizes, and registers one transport-independent relay session.
234    async fn accept_framed<S>(
235        &self,
236        mut io: S,
237        request_parts: http::request::Parts,
238        protocol_version: ProtocolVersion,
239    ) -> Result<(), AcceptError>
240    where
241        S: BytesStreamSink + crate::ExportKeyingMaterial + Send + 'static,
242    {
243        let client_auth_header = request_parts.headers.get(CLIENT_AUTH_HEADER).cloned();
244        let challenge = self.clients.next_auth_challenge();
245        let authentication =
246            handshake::serverside_with_challenge(&mut io, client_auth_header, challenge).await?;
247
248        trace!(?authentication.mechanism, "accept: verified authentication");
249
250        let request =
251            ClientRequest::new(authentication.client_key, protocol_version, request_parts);
252
253        // Authorize the request against the configured `AccessControl`.
254        let guard = authentication
255            .authorize_with(&request, &self.access, &mut io)
256            .await?;
257
258        trace!("accept: verified authorization");
259
260        let io = RelayedStream::new(io, self.key_cache.clone());
261
262        trace!("accept: build client conn");
263        let mut client_conn_builder = Config::new(guard, io, protocol_version);
264        client_conn_builder.write_timeout = self.write_timeout;
265        trace!(endpoint_id = %request.endpoint_id().fmt_short(), "create client");
266
267        // build and register client, starting up read & write loops for the client
268        // connection
269        match self
270            .clients
271            .register(client_conn_builder, self.metrics.clone())
272        {
273            Ok(()) => {}
274            Err(RegisterError::GlobalSessionFull { .. }) => {
275                return Err(e!(AcceptError::GlobalSessionFull));
276            }
277            Err(RegisterError::EndpointSessionFull { .. }) => {
278                return Err(e!(AcceptError::EndpointSessionFull));
279            }
280            Err(RegisterError::Runtime { source, .. }) => {
281                return Err(e!(AcceptError::Runtime, source));
282            }
283        }
284        Ok(())
285    }
286}

392    /// Opens one production relay client/server session over an in-memory byte pipe.
393    ///
394    /// DNS, TCP, TLS, and HTTP upgrade mechanics are bypassed. The normal WebSocket framing,
395    /// challenge authentication, authorization, server client actor, registry, and routing code
396    /// remain in the call path. This API is reserved for repository simulation infrastructure.
397    #[doc(hidden)]
398    #[cfg(feature = "test-utils")]
399    pub async fn connect_in_memory(
400        &self,
401        builder: &crate::client::ClientBuilder,
402        protocol_version: ProtocolVersion,
403        byte_capacity: usize,
404    ) -> Result<crate::client::Client, InMemoryConnectError> {
405        let request_parts = builder
406            .in_memory_request_parts()
407            .map_err(|error| e!(InMemoryConnectError::Client, error))?;
408        let (client_io, server_io) = tokio::io::duplex(byte_capacity.max(1));
409        let server_io = RateLimited::from_watcher(
410            MaybeTlsStream::Test(server_io),
411            self.0.rate_limit.subscribe(),
412            self.0.metrics.clone(),
413        )
414        .map_err(|error| e!(AcceptError::RateLimitingMisconfigured, error))
415        .map_err(|error| e!(InMemoryConnectError::Server, error))?;
416        let websocket = tokio_websockets::ServerBuilder::new()
417            .limits(tokio_websockets::Limits::default().max_payload_len(Some(MAX_FRAME_SIZE)))
418            .serve(server_io);
419        let framed = WsBytesFramed { io: websocket };
420        let server = async {
421            self.0
422                .accept_framed(framed, request_parts, protocol_version)
423                .await
424                .map_err(|error| e!(InMemoryConnectError::Server, error))
425        };
426        let client = async {
427            builder
428                .connect_in_memory(client_io, protocol_version)
429                .await
430                .map_err(|error| e!(InMemoryConnectError::Client, error))
431        };
432        let ((), client) = tokio::try_join!(server, client)?;
433        Ok(client)
434    }

444    /// Handle the incoming connection.
445    ///
446    /// If a `tls_config` is given, will serve the connection using HTTPS, otherwise HTTP.
447    ///
448    /// If the connection did not fully upgrade to a relay WebSocket connection after
449    /// `establish_timeout`, the connection is aborted.
450    ///
451    /// # Example
452    ///
453    /// ```no_run
454    /// # use std::{sync::Arc, time::Duration};
455    /// # use tokio::net::TcpStream;
456    /// # use http::HeaderMap;
457    /// # use iroh_relay::server::http_server::{Handlers, RelayService, TlsConfig};
458    /// # use iroh_relay::{KeyCache, server::{AllowAll, Metrics}};
459    /// # use webpki_types::{CertificateDer, PrivateKeyDer};
460    /// # async fn example(stream: TcpStream) -> Result<(), Box<dyn std::error::Error>> {
461    /// // Create a relay service
462    /// let handlers = Handlers::default();
463    /// let headers = HeaderMap::new();
464    /// let key_cache = KeyCache::new(1024);
465    /// let metrics = Arc::new(Metrics::default());
466    /// let relay_service = RelayService::new(
467    ///     handlers,
468    ///     headers,
469    ///     None, // No rate limiting
470    ///     key_cache,
471    ///     Arc::new(AllowAll),
472    ///     metrics,
473    /// );
474    ///
475    /// // Generate a self-signed certificate for HTTPS
476    /// let cert = rcgen::generate_simple_self_signed(vec!["localhost".to_string()])?;
477    /// let cert_der = cert.cert.der().to_vec();
478    /// let private_key_der = cert.signing_key.serialize_der();
479    /// let cert_chain = vec![CertificateDer::from(cert_der)];
480    /// let private_key = PrivateKeyDer::try_from(private_key_der)?;
481    ///
482    /// // Serve with HTTPS
483    /// let server_config = Arc::new(
484    ///     rustls::ServerConfig::builder()
485    ///         .with_no_client_auth()
486    ///         .with_single_cert(cert_chain, private_key)?,
487    /// );
488    /// let tls_config = TlsConfig::new(server_config);
489    /// relay_service
490    ///     .clone()
491    ///     .handle_connection(stream, Some(tls_config), Duration::from_secs(30))
492    ///     .await;
493    ///
494    /// // Or serve with plain HTTP
495    /// # let stream = TcpStream::connect("127.0.0.1:0").await?;
496    /// relay_service
497    ///     .handle_connection(stream, None, Duration::from_secs(30))
498    ///     .await;
499    /// # Ok(())
500    /// # }
501    /// ```
502    pub async fn handle_connection(
503        self,
504        stream: TcpStream,
505        tls_config: Option<TlsConfig>,
506        establish_timeout: Duration,
507    ) {
508        self.handle_connection_inner(stream, tls_config, establish_timeout, None)
509            .await;
510    }
511
512    pub(super) async fn handle_connection_with_lease(
513        self,
514        stream: TcpStream,
515        tls_config: Option<TlsConfig>,
516        establish_timeout: Duration,
517        lease: EstablishmentLease,
518    ) {
519        self.handle_connection_inner(stream, tls_config, establish_timeout, Some(lease))
520            .await;
521    }
522
523    async fn handle_connection_inner(
524        self,
525        stream: TcpStream,
526        tls_config: Option<TlsConfig>,
527        establish_timeout: Duration,
528        establishment_lease: Option<EstablishmentLease>,
529    ) {
530        let metrics = self.0.metrics.clone();
531        metrics.http_connections.inc();
532        // We create a notification token to be triggered once the connection is fully established
533        // and passed to the relay server.
534        let on_establish = Arc::new(Notify::new());
535        let establishment_lease = Arc::new(Mutex::new(establishment_lease));
536        let service = RelayServiceWithNotify::new_with_establishment_lease(
537            self,
538            on_establish.clone(),
539            establishment_lease.clone(),
540        );
541
542        // This is the main connection future, driving the connection to completion.
543        let serve_fut = async move {
544            match tls_config {
545                Some(tls_config) => {
546                    debug!("HTTPS: serve connection");
547                    service.tls_serve_connection(stream, tls_config).await
548                }
549                None => {
550                    debug!("HTTP: serve connection");
551                    let stream = MaybeTlsStream::Plain(stream);
552                    service.serve_connection(stream).await
553                }
554            }
555        };
556
557        // We set a timeout for the connection to limit lingering connections during establishment.
558        // The timeout is cleared once the connection has completed the TLS and WebSocket
559        // handshakes and has been passed over to the relay protocol handler.
560        // If the timeout expires before that happens, the connection is aborted.
561        let res = clearable_timeout(
562            establish_timeout,
563            on_establish,
564            establishment_lease,
565            serve_fut,
566        )
567        .await
568        .map_err(|_elapsed| e!(ServeConnectionError::EstablishTimeout))
569        .flatten();
570
571        metrics.http_connections_closed.inc();
572
573        if let Err(error) = res {
574            match error {
575                ServeConnectionError::ManualAccept { source, .. }
576                    if source.kind() == std::io::ErrorKind::UnexpectedEof =>
577                {
578                    debug!(reason=?source, "peer disconnected");
579                }
580                #[cfg(feature = "server-acme")]
581                ServeConnectionError::LetsEncryptAccept { source, .. }
582                    if source.kind() == std::io::ErrorKind::UnexpectedEof =>
583                {
584                    debug!(reason=?source, "peer disconnected");
585                }
586                // From hyper: <https://github.com/hyperium/hyper/commit/271bba16672ff54a44e043c5cc1ae6b9345bb172>
587                // `hyper::Error::IncompleteMessage` is hyper's equivalent of UnexpectedEof
588                ServeConnectionError::Https { source, .. }
589                | ServeConnectionError::Http { source, .. }
590                    if source.is_incomplete_message() =>
591                {
592                    debug!(reason=?source, "peer disconnected");
593                }
594                _ => {
595                    metrics.http_connections_errored.inc();
596                    error!(?error, "failed to handle connection");
597                }
598            }
599        }
600    }

626impl RelayServiceWithNotify {
627    /// Serves a TLS connection.
628    async fn tls_serve_connection(
629        self,
630        stream: TcpStream,
631        tls_config: TlsConfig,
632    ) -> Result<(), ServeConnectionError> {
633        let TlsConfig { acceptor, config } = tls_config;
634        #[cfg(not(feature = "server-acme"))]
635        let _ = config;
636        let stream = match acceptor {
637            #[cfg(feature = "server-acme")]
638            TlsAcceptor::LetsEncrypt { challenge_config } => {
639                let start_handshake =
640                    tokio_rustls::LazyConfigAcceptor::new(Default::default(), stream)
641                        .await
642                        .map_err(|err| e!(ServeConnectionError::LetsEncryptAccept, err))?;
643                if tokio_rustls_acme::is_tls_alpn_challenge(&start_handshake.client_hello()) {
644                    info!("TLS[acme]: received TLS-ALPN-01 validation request");
645                    start_handshake
646                        .into_stream(challenge_config)
647                        .await
648                        .map_err(|err| e!(ServeConnectionError::TlsHandshake, err))?;
649                    return Ok(());
650                }
651                debug!("TLS[acme]: start handshake");
652                let tls_stream = start_handshake
653                    .into_stream(config)
654                    .await
655                    .map_err(|err| e!(ServeConnectionError::TlsHandshake, err))?;
656                MaybeTlsStream::Tls(tls_stream)
657            }
658            TlsAcceptor::Manual(a) => {
659                debug!("TLS[manual]: accept");
660                let tls_stream = a
661                    .accept(stream)
662                    .await
663                    .map_err(|err| e!(ServeConnectionError::ManualAccept, err))?;
664                MaybeTlsStream::Tls(tls_stream)
665            }
666        };
667        self.serve_connection(stream).await
668    }
669
670    /// Wrapper for the actual http connection (with upgrades)
671    async fn serve_connection(self, io: MaybeTlsStream) -> Result<(), ServeConnectionError> {
672        hyper::server::conn::http1::Builder::new()
673            .serve_connection(hyper_util::rt::TokioIo::new(io), self)
674            .with_upgrades()
675            .await
676            .map_err(|err| e!(ServeConnectionError::ServeConnection, err))
677    }
678}

681#[derive(Default)]
682pub struct Handlers(HashMap<(Method, &'static str), HyperHandler>);
683
684impl std::fmt::Debug for Handlers {
685    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
686        let s = self.0.keys().fold(String::new(), |curr, next| {
687            let (method, uri) = next;
688            format!("{curr}\n({method},{uri}): Box<Fn(ResponseBuilder) -> Result<Response<Body>> + Send + Sync + 'static>")
689        });
690        write!(f, "HashMap<{s}>")
691    }
692}
