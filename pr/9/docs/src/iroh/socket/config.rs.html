63/// Configuration for a [`noq::Endpoint`] that cannot be changed at runtime.
64#[derive(derive_more::Debug)]
65pub(crate) struct StaticConfig {

85impl StaticConfig {
86    /// Create a [`noq_proto::ServerConfig`] with the specified ALPN protocols.
87    pub(crate) fn create_server_config(
88        &self,
89        alpn_protocols: Vec<Vec<u8>>,
90    ) -> noq_proto::ServerConfig {
91        let mut quic_server_config = self.server_config.clone();
92        quic_server_config.set_alpn_protocols(alpn_protocols);
93        let mut inner =
94            noq::ServerConfig::new(Arc::new(quic_server_config), self.token_key.clone());
95        inner.transport_config(self.transport_config.to_inner_arc());
96        inner.max_incoming(self.limits.max_connections().get());
97        #[cfg(not(wasm_browser))]
98        inner.time_source(Arc::new(crate::runtime::NoqWallClock::new(
99            self.runtime_context.wall_clock(),
100        )));
101        inner
102    }
103
104    /// Create a [`noq_proto::ClientConfig`] with the specified ALPN protocols.
105    pub(crate) fn create_client_config(
106        &self,
107        alpn_protocols: Vec<Vec<u8>>,
108        transport_config: Arc<noq::TransportConfig>,
109    ) -> noq_proto::ClientConfig {
110        let mut quic_client_config = self.client_config.clone();
111        quic_client_config.set_alpn_protocols(alpn_protocols);
112        let mut inner = noq::ClientConfig::new(Arc::new(quic_client_config));
113        inner.transport_config(transport_config);
114        inner.token_store(self.token_store.clone());
115        #[cfg(not(wasm_browser))]
116        if let Some(provider) = &self.simulation_initial_dst_cid_provider {
117            inner.initial_dst_cid_provider(provider.clone());
118        }
119        inner
120    }
121}

126pub enum BindError {

128    Sockets { source: io::Error },
129    #[error("Failed to create internal QUIC endpoint")]
130    CreateQuicEndpoint { source: io::Error },
131    #[error("Failed to create netmon monitor")]

134    InvalidTransportConfig,
135    #[error("Invalid CA root configuration")]
136    InvalidCaRootConfig { source: io::Error },
137    #[error("Failed to create an address lookup service")]
138    AddressLookup {
139        #[error(from)]
140        source: crate::address_lookup::AddressLookupBuilderError,
141    },
142    #[error("Missing or incompatible rustls crypto provider configured")]
143    InvalidCryptoProvider,
144    #[error("Error constructing TLS configuration")]
145    TlsConfigError {
146        #[error(from)]
147        source: tls::TlsConfigError,
148    },
149    #[error("Failed to create QUIC address-discovery client")]
150    CreateQuicClient {
151        source: iroh_relay::quic::QuicClientBuildError,
152    },
153    #[error("Invalid deterministic runtime context")]
154    RuntimeContext { source: AnyError },
155    #[error("Endpoint task capacity cannot cover its configured connection and actor limits")]
