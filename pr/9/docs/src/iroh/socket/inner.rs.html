52impl Drop for EndpointInner {
53    fn drop(&mut self) {
54        if self.sock.is_closed() {
55            return;
56        }
57        tracing::error!(
58            "Endpoint dropped without calling `Endpoint::close`. Aborting ungracefully."
59        );
60        self.abort();
61    }
62}

64/// This coordinates the shutdown of the [`Socket`] and all its tasks.
65///
66/// It also tightly binds to the [`EndpointInner`] and [`Actor`] closing as that is where
67/// most of the logic lives.
68#[derive(Debug)]
69struct ShutdownState {
70    /// Token that is cancelled at the moment [`crate::Endpoint::close`] is called.
71    ///
72    /// Currently cancelled from [`EndpointInner::close`].
73    at_close_start: CancellationToken,
74    /// Token that is cancelled once the [`noq::Endpoint`] is drained.
75    ///
76    /// Only 100ms after this is cancelled will the [`Actor`] task be cancelled, it should
77    /// have exited already by then as it is considered an error if it was still running.
78    at_endpoint_closed: CancellationToken,
79    /// Set if the endpoint is closed and all tasks are stopped.
80    ///
81    /// This is only set once both [`Self::at_close_start`] and [`Self::at_endpoint_closed`]
82    /// are cancelled **and** the [`Actor`] task is no longer running.
83    closed: AtomicBool,
84}
85
86impl Default for ShutdownState {
87    fn default() -> Self {
88        Self {
89            at_close_start: CancellationToken::new(),
90            at_endpoint_closed: CancellationToken::new(),
91            closed: AtomicBool::new(false),
92        }
93    }
94}

96impl ShutdownState {
97    /// Whether the endpoint has started closing, or is already closed.
98    ///
99    /// This is true once [`crate::Endpoint::close`] is called, and remains true forever
100    /// after. Tasks might still be shutting down.
101    fn is_closing(&self) -> bool {
102        self.at_close_start.is_cancelled()
103    }
104
105    /// Whether the endpoint is fully closed and all tasks stopped.
106    ///
107    /// The endpoint will be drained, all transports and sockets will be closed.
108    fn is_closed(&self) -> bool {
109        self.closed.load(Ordering::Relaxed)
110    }
111}

113/// Iroh connectivity layer.
114///
115/// This is responsible for routing packets to endpoints based on endpoint IDs, it will initially
116/// route packets via a relay and transparently try and establish an endpoint-to-endpoint
117/// connection and upgrade to it.  It will also keep looking for better connections as the
118/// network details of both endpoints change.
119///
120/// It is usually only necessary to use a single [`Socket`] instance in an application, it
121/// means any QUIC endpoints on top will be sharing as much information about endpoints as
122/// possible.
123#[derive(Debug)]
124pub(crate) struct Socket {

131    /// `RemoteStateActor`s must go through the socket actor channel.

134    // - Shutdown Management

142    /// If the last net_report report, reports IPv6 to be available.
143    pub(super) ipv6_reported: Arc<AtomicBool>,

145    pub(super) mapped_addrs: MappedAddrs,
146
147    /// Local addresses
148    local_addrs_watch: LocalAddrsWatch,
149    home_relay_watch: HomeRelayWatcher,
150    /// Currently bound IP addresses of all sockets

363    /// Process datagrams received from all the transports.
364    ///
365    /// All the `bufs` and `metas` should have initialized packets in them.
366    ///
367    /// This fixes up the datagrams to use the correct [`MultipathMappedAddr`] and extracts
368    /// DISCO packets, processing them inside the socket.
369    ///
370    /// [`MultipathMappedAddr`]: mapped_addrs::MultipathMappedAddr
371    pub(super) fn process_datagrams(
372        &self,
373        bufs: &mut [io::IoSliceMut<'_>],
374        metas: &mut [noq_udp::RecvMeta],
375        recv_infos: &[transports::RecvInfo],
376    ) {
377        assert_eq!(bufs.len(), metas.len(), "non matching bufs & metas");
378        assert_eq!(
379            bufs.len(),
380            recv_infos.len(),
381            "non matching bufs & recv_infos"
382        );
383
384        // zip is slow :(
385        for i in 0..metas.len() {
386            let noq_meta = &mut metas[i];
387            let recv_info = &recv_infos[i];
388            let remote_addr = recv_info.remote();
389            let datagram_count = if noq_meta.stride == 0 {
390                if noq_meta.len > 0 {
391                    warn!(
392                        src = ?remote_addr,
393                        len = noq_meta.len,
394                        "received datagram with stride=0 but len>0",
395                    );
396                    // fix the weird len
397                    noq_meta.len = 0;
398                }
399                // one empty datagram
400                1
401            } else {
402                noq_meta.len.div_ceil(noq_meta.stride)
403            };
404            self.metrics
405                .socket
406                .recv_datagrams
407                .inc_by(datagram_count as _);
408            if noq_meta.len > noq_meta.stride {
409                trace!(
410                    src = ?remote_addr,
411                    len = noq_meta.len,
412                    stride = %noq_meta.stride,
413                    datagram_count,
414                    "GRO datagram received",
415                );
416                self.metrics.socket.recv_gro_datagrams.inc();
417            } else {
418                trace!(src = ?remote_addr, len = noq_meta.len, "datagram received");
419            }
420            match remote_addr {
421                transports::Addr::Ip(SocketAddr::V4(..)) => {
422                    self.metrics.socket.recv_data_ipv4.inc_by(noq_meta.len as _);
423                }
424                transports::Addr::Ip(SocketAddr::V6(..)) => {
425                    self.metrics.socket.recv_data_ipv6.inc_by(noq_meta.len as _);
426                }
427                transports::Addr::Relay(src_url, src_node) => {
428                    self.metrics
429                        .socket
430                        .recv_data_relay
431                        .inc_by(noq_meta.len as _);
432
433                    // Fill in the correct mapped address
434                    let mapped_addr = self
435                        .mapped_addrs
436                        .relay_addrs
437                        .get(&(src_url.clone(), *src_node));
438                    noq_meta.addr = mapped_addr.private_socket_addr();
439                }
440                transports::Addr::Custom(remote) => {
441                    self.metrics
442                        .socket
443                        .recv_data_custom
444                        .inc_by(noq_meta.len as _);
445                    // Fill in the correct mapped address
446                    let mapped_addr = self.mapped_addrs.custom_addrs.get(remote);
447                    noq_meta.addr = mapped_addr.private_socket_addr();
448                    if let Some(local) = recv_info.local() {
449                        let local_mapped = self.mapped_addrs.custom_addrs.get(local);
450                        noq_meta.dst_ip = Some(local_mapped.private_socket_addr().ip());
451                    }
452                }
453            }
454        }
455    }

457    /// Publishes our address to an address lookup service, if configured.
458    ///
459    /// Called whenever our addresses or home relay endpoint changes.
460    pub(super) fn publish_my_addr(&self) {
461        let relay_url = self.my_relay();
462        let mut addrs: Vec<_> = self
463            .direct_addrs
464            .sockaddrs()
465            .map(TransportAddr::Ip)
466            .collect();
467
468        let user_data = self
469            .address_lookup_user_data
470            .read()
471            .expect("lock poisened")
472            .clone();
473        if relay_url.is_none() && addrs.is_empty() && user_data.is_none() {
474            // do not bother publishing if we don't have any information
475            return;
476        }
477        if let Some(url) = relay_url {
478            addrs.push(TransportAddr::Relay(url));
479        }
480
481        let mut data = EndpointData::new(addrs);
482        data.set_user_data(user_data);
483        self.address_lookup.publish(&data);
484    }

502    /// Creates a [`EndpointInner`].
503    pub(crate) async fn bind(opts: Options) -> Result<Self, BindError> {
504        // Use the current span as the main span for all tasks spawned in this endpoint.
505        // `EndpointInner::bind` is not public and only called from `crate::endpoint::Builder::bind`,
506        // which instruments the call with a span created for this purpose.
507        let span = tracing::Span::current();
508
509        let Options {
510            secret_key,
511            transports: transport_configs,
512            address_lookup_user_data,
513            #[cfg(not(wasm_browser))]
514            dns_resolver,
515            #[cfg(not(wasm_browser))]
516            runtime_context,
517            #[cfg(not(wasm_browser))]
518            ip_socket_factory,
519            #[cfg(not(wasm_browser))]
520            network_monitor,
521            #[cfg(not(wasm_browser))]
522            simulation_port_mapper,
523            #[cfg(not(wasm_browser))]
524            simulation_relay_connector,
525            #[cfg(not(wasm_browser))]
526            simulation_preferred_relay,
527            #[cfg(not(wasm_browser))]
528            simulation_reset_key,
529            proxy_url,
530            server_config,
531            tls_config,
532            metrics,
533            hooks,
534            path_selector,
535            portmapper_config,
536            net_report_config,
537            static_config,
538            configured_addrs,
539            limits,
540        } = opts;
541
542        #[cfg(not(wasm_browser))]
543        let runtime = Arc::new(Runtime::new_with_limits(
544            secret_key.public(),
545            runtime_context.clone(),
546            limits,
547        ));
548        #[cfg(wasm_browser)]
549        let runtime = Arc::new(Runtime::new_with_limits(secret_key.public(), limits));
550        let connection_admission =
551            crate::endpoint::limits::AdmissionLedger::new(limits.max_connections());
552
553        let address_lookup = address_lookup::AddressLookupServices::default();
554        #[cfg(not(wasm_browser))]
555        let port_mapper = portmapper::create_client(&portmapper_config, simulation_port_mapper);
556        #[cfg(wasm_browser)]
557        let port_mapper = portmapper::create_client(&portmapper_config);
558
559        let relay_transport_configs: Vec<_> = transport_configs
560            .iter()
561            .filter(|t| matches!(t, TransportConfig::Relay { .. }))
562            .collect();
563
564        // Currently we only support a single relay transport
565        if relay_transport_configs.len() > 1 {
566            bail!(BindError::InvalidTransportConfig);
567        }
568        let relay_map = relay_transport_configs
569            .iter()
570            .filter_map(|t| {
571                #[allow(irrefutable_let_patterns)]
572                if let TransportConfig::Relay { relay_map, .. } = t {
573                    Some(relay_map.clone())
574                } else {
575                    None
576                }
577            })
578            .next()
579            .unwrap_or_else(RelayMap::empty);
580
581        let ipv6_reported = Arc::new(AtomicBool::new(false));
582
583        #[cfg(not(wasm_browser))]
584        let has_simulation_relay_connector = simulation_relay_connector.is_some();
585
586        let relay_actor_config = RelayActorConfig {
587            my_relay: HomeRelayWatch::default(),
588            secret_key: secret_key.clone(),
589            #[cfg(not(wasm_browser))]
590            dns_resolver: dns_resolver.clone(),
591            proxy_url: proxy_url.clone(),
592            ipv6_reported: ipv6_reported.clone(),
593            tls_config: tls_config.clone(),
594            metrics: metrics.socket.clone(),
595            relay_map: relay_map.clone(),
596            #[cfg(not(wasm_browser))]
597            relay_connector: simulation_relay_connector,
598            #[cfg(not(wasm_browser))]
599            initial_relay: simulation_preferred_relay,
600            limits,
601        };
602
603        let shutdown_state = ShutdownState::default();
604        let shutdown_token = shutdown_state.at_endpoint_closed.child_token();
605
606        let transports = Transports::bind(
607            &transport_configs,
608            relay_actor_config,
609            &metrics,
610            shutdown_token.child_token(),
611            #[cfg(not(wasm_browser))]
612            runtime.clone(),
613            #[cfg(not(wasm_browser))]
614            ip_socket_factory,
615        )
616        .map_err(|err| e!(BindError::Sockets, err))?;
617
618        if let Some(v4_port) = transports.local_addrs().into_iter().find_map(|t| {
619            if let transports::Addr::Ip(SocketAddr::V4(addr)) = t {
620                Some(addr.port())
621            } else {
622                None
623            }
624        }) {
625            // NOTE: we can end up with a zero port if `netwatch::UdpSocket::socket_addr` fails
626            match v4_port.try_into() {
627                Ok(non_zero_port) => {
628                    port_mapper.update_local_port(non_zero_port);
629                }
630                Err(_zero_port) => debug!("Skipping port mapping with zero local port"),
631            }
632        }
633
634        let (actor_sender, actor_receiver) = mpsc::channel(256);
635
636        #[cfg(not(wasm_browser))]
637        let has_ipv6_transport = transports
638            .ip_bind_addrs()
639            .into_iter()
640            .any(|addr| addr.is_ipv6());
641
642        #[cfg(not(wasm_browser))]
643        let has_ip_transports = !transports.ip_bind_addrs().is_empty();
644
645        let direct_addrs = DiscoveredDirectAddrs::default();
646
647        let remote_map = {
648            RemoteMap::new(
649                metrics.socket.clone(),

688            Some(key) => Blake3HmacKey::from_key(key),
689            None => Blake3HmacKey::new(&mut rand::rng()),

707        // For performance reasons and to not rewrite buffers we pass non-QUIC UDP packets straight
708        // through to noq. We set the first byte of the packet to zero, which makes noq ignore

775            #[cfg(not(wasm_browser))]

786            port_mapper,

799            .context()
800            .decisions()
801            .stream(&format!("endpoint/{}/socket/re-stun-period", runtime.id()))
802            .map_err(|err| e!(BindError::RuntimeContext, anyerr!(err)))?;

813            sock: sock.clone(),

833                actor_receiver,
834                shutdown_token.child_token(),
835                local_addrs_watch,
836            )
837            .instrument(info_span!(parent: span, "actor"));

841                iroh_runtime::TaskKind::SocketActor,
842                "socket-actor",

844            )
845            .map_err(|err| e!(BindError::RuntimeContext, anyerr!(err)))?;

851        Ok(EndpointInner {
852            sock,
853            actor_sender,
854            actor_task,
855            endpoint,
856            runtime,
857            connection_admission,
858            static_config,
859        })

867    /// Closes the iroh endpoint.
868    ///
869    /// Only the first close does anything. Any later closes return nil.  Polling the socket
870    /// ([`noq::AsyncUdpSocket::poll_recv`]) will return [`Poll::Pending`] indefinitely
871    /// after this call.
872    ///
873    /// [`Poll::Pending`]: std::task::Poll::Pending
874    #[instrument(skip_all, parent = self.sock.span.clone())]
875    pub(crate) async fn close(&self) {
876        if self.sock.is_closed() || self.sock.is_closing() {
877            return;
878        }
879        trace!("socket closing...");
880
881        // Cancel at_close_start token, which cancels running netreports.
882        self.sock.shutdown.at_close_start.cancel();
883
884        // Remove address lookup services
885        self.sock.address_lookup().clear();
886
887        // Initiate closing all connections, and refuse future connections.
888        self.noq_endpoint().close(0u16.into(), b"");
889
890        // In the history of this code, this call had been
891        // - removed: https://github.com/n0-computer/iroh/pull/1753
892        // - then added back in: https://github.com/n0-computer/iroh/pull/2227/files#diff-ba27e40e2986a3919b20f6b412ad4fe63154af648610ea5d9ed0b5d5b0e2d780R573
893        // - then removed again: https://github.com/n0-computer/iroh/pull/3165
894        // and finally added back in together with this comment.
895        // So before removing this call, please consider carefully.
896        // Among other things, this call tries its best to make sure that any queued close frames
897        // (e.g. via the call to `endpoint.close(...)` above), are flushed out to the sockets
898        // *and acknowledged* (or time out with the "probe timeout" of usually 3 seconds).
899        // This allows the other endpoints for these connections to be notified to release
900        // their resources, or - depending on the protocol - that all data was received.
901        // With the current noq API, this is the only way to ensure protocol code can use
902        // connection close codes, and close the endpoint properly.
903        // If this call is skipped, then connections that protocols close just shortly before the
904        // call to `Endpoint::close` will in most cases cause connection time-outs on remote ends.
905        trace!("wait_all_draining start");
906        self.noq_endpoint().wait_all_draining().await;
907        trace!("wait_all_draining done");
908
909        // Start cancellation of all actors.
910        self.sock.shutdown.at_endpoint_closed.cancel();
911
912        // MutexGuard is not held across await points
913        let task = self.actor_task.lock().expect("poisoned").take();
914        if let Some(task) = task {
915            // give the tasks a moment to shutdown cleanly
916            #[cfg(wasm_browser)]
917            let shutdown_done = time::timeout(Duration::from_millis(100), async move {
918                if let Err(err) = task.await {
919                    warn!("unexpected error in task shutdown: {:?}", err);
920                }
921            })
922            .await;
923            #[cfg(wasm_browser)]
924            match shutdown_done {
925                Ok(_) => trace!("tasks finished in time, shutdown complete"),
926                Err(time::Elapsed { .. }) => {
927                    // Dropping the task will abort it
928                    warn!("tasks didn't finish in time, aborting");
929                }
930            }
931            #[cfg(not(wasm_browser))]
932            match crate::runtime::RuntimeTimeout::after(
933                self.runtime.context().clock(),
934                Duration::from_millis(100),
935                async move {
936                    if let Err(err) = task.join().await {
937                        warn!("unexpected error in task shutdown: {:?}", err);
938                    }
939                },
940            ) {
941                Ok(timeout) => match timeout.await {
942                    Ok(()) => trace!("tasks finished in time, shutdown complete"),
943                    Err(iroh_runtime::TimeoutError::Elapsed) => {
944                        warn!("tasks didn't finish in time, aborting");
945                    }
946                    Err(iroh_runtime::TimeoutError::Clock(error)) => {
947                        self.runtime.latch_failure(error.to_string());
948                    }
949                },
950                Err(error) => self.runtime.latch_failure(error.to_string()),
951            }
952        }
953
954        // Waits for the EndpointDriver and all ConnectionDrivers to shut down
955        // Expects that the `noq::Endpoint` has been closed before this call,
956        // otherwise, the runtime will never shutdown.
957        self.runtime.shutdown().await;
958
959        self.sock.shutdown.closed.store(true, Ordering::SeqCst);
960
961        trace!("socket closed");
962    }

964    /// Aborts the endpoint ungracefully:
965    ///
966    /// - Calls cancellation token that stops running net reports
967    /// - Removes all address lookup services
968    /// - Calls cancellation token that stops all the Socket actors
969    /// - Aborts the runtime
970    /// - Drops the actor task
971    /// - Sets the `Socket::is_closed` state to true
972    ///
973    /// This does not wait for any current connections or tasks to close gracefully.
974    ///
975    /// This should only be called in the `iroh::Endpoint` `Drop` impl when the
976    /// `iroh::Endpoint` is dropped without first calling `Endpoint::close`.
977    #[instrument(skip_all, parent = self.sock.span.clone())]
978    pub(crate) fn abort(&self) {
979        if self.sock.is_closed() || self.sock.is_closing() {
980            return;
981        }
982        trace!("socket aborting...");
983
984        // Cancel at_close_start token, which cancels running netreports.
985        self.sock.shutdown.at_close_start.cancel();
986
987        self.sock.address_lookup().clear();
988
989        // Cancel all actors.
990        self.sock.shutdown.at_endpoint_closed.cancel();
991
992        // Aborts all tasks, not waiting for any to close gracefully.
993        self.runtime.abort();
994
995        self.actor_task.lock().expect("poisoned").take();
996
997        self.sock.shutdown.closed.store(true, Ordering::SeqCst);
998        trace!("socket closed");
999    }

1069    /// Resolves an [`EndpointAddr`] to an [`EndpointIdMappedAddr`] to connect to via [`EndpointInner::endpoint`].
1070    ///
1071    /// This starts a `RemoteStateActor` for the remote if not running already, and then checks
1072    /// if the actor has any known paths to the remote. If not, it starts address lookup and waits for
1073    /// at least one result to arrive.
1074    ///
1075    /// Returns `Ok(Ok(EndpointIdMappedAddr))` if there is a known path or Address Lookup produced
1076    /// at least one result. This does not mean there is a working path, only that we have at least
1077    /// one transport address we can try to connect to.
1078    ///
1079    /// Returns `Ok(Err(address_lookup_error))` if there are no known paths to the remote and Address Lookup
1080    /// failed or produced no results. This means that we don't have any transport address for
1081    /// the remote, thus there is no point in trying to connect over the noq endpoint.
1082    ///
1083    /// Returns `Err(RemoteStateActorStoppedError)` if the `RemoteStateActor` for the remote has stopped,
1084    /// which may never happen and thus is a bug if it does.
1085    pub(crate) async fn resolve_remote(
1086        &self,
1087        addr: EndpointAddr,
1088    ) -> Result<Result<EndpointIdMappedAddr, AddressLookupFailed>, RemoteStateActorStoppedError>
1089    {
1090        let (tx, rx) = oneshot::channel();
1091        let remote_id = addr.id;
1092        self.actor_sender
1093            .send(ActorMessage::ResolveRemote(addr, tx))
1094            .await
1095            .ok();
1096        let reply = rx.await.map_err(|_| RemoteStateActorStoppedError::new())?;
1097        match reply {
1098            Ok(()) => Ok(Ok(self.mapped_addrs.endpoint_addrs.get(&remote_id))),
1099            Err(err) => Ok(Err(err)),
1100        }
1101    }

1116    /// Registers the connection in the `RemoteStateActor`.
1117    ///
1118    /// The actor is responsible for holepunching and opening additional paths to this
1119    /// connection.
1120    ///
1121    /// Returns a future that resolves to a [`PathStateReceiver`] for the new connection.
1122    ///
1123    /// The returned future is `'static`, so it can be stored without being lifetime-bound to `&self`.
1124    pub(crate) fn register_connection(
1125        &self,
1126        remote: EndpointId,
1127        conn: noq::Connection,
1128    ) -> impl Future<Output = Result<PathStateReceiver, RemoteStateRegistrationError>> + Send + 'static
1129    {
1130        let (tx, rx) = oneshot::channel();
1131        let sender = self.actor_sender.clone();
1132        async move {
1133            sender
1134                .send(ActorMessage::AddConnection(remote, conn, tx))
1135                .await
1136                .map_err(|_| e!(RemoteStateRegistrationError::ActorStopped))?;
1137            rx.await
1138                .map_err(|_| e!(RemoteStateRegistrationError::ActorStopped))?
1139                .map_err(|source| e!(RemoteStateRegistrationError::Admission, source))
1140        }
1141    }
