3/// Builder for [`Endpoint`].
4///
5/// By default the endpoint will generate a new random [`SecretKey`], which will result in a
6/// new [`EndpointId`].
7///
8/// To create the [`Endpoint`] call [`Builder::bind`].
9#[derive(Debug)]
10pub struct Builder {
11    secret_key: Option<SecretKey>,
12    alpn_protocols: Vec<Vec<u8>>,
13    transport_config: QuicTransportConfig,
14    keylog: bool,
15    address_lookup: Vec<Box<dyn DynAddressLookupBuilder>>,
16    address_lookup_user_data: Option<UserData>,
17    /// Default address filter applied to all address lookup services added via
18    /// [`Builder::address_lookup`].
19    addr_filter: Option<AddrFilter>,
20    proxy_url: Option<Url>,
21    ca_tls_config: Option<CaTlsConfig>,
22    #[cfg(not(wasm_browser))]
23    dns_resolver: Option<DnsResolver>,
24    transports: Vec<TransportConfig>,
25    max_tls_tickets: usize,
26    hooks: EndpointHooksList,
27    path_selector: Arc<dyn PathSelector>,
28    portmapper_config: PortmapperConfig,
29    net_report_config: NetReportConfig,
30    crypto_provider: Option<Arc<rustls::crypto::CryptoProvider>>,
31    configured_addrs: BTreeSet<SocketAddr>,
32    #[cfg(not(wasm_browser))]
33    pub(super) simulation_environment: Option<crate::simulation::SimulationEnvironment>,
34    limits: EndpointLimits,
35}

37impl From<RelayMode> for Option<TransportConfig> {
38    fn from(mode: RelayMode) -> Self {
39        match mode {
40            RelayMode::Disabled => None,
41            RelayMode::Default => Some(TransportConfig::Relay {
42                relay_map: mode.relay_map(),
43                is_user_defined: true,
44            }),
45            RelayMode::Staging => Some(TransportConfig::Relay {
46                relay_map: mode.relay_map(),
47                is_user_defined: true,
48            }),
49            RelayMode::Custom(relay_map) => Some(TransportConfig::Relay {
50                relay_map,
51                is_user_defined: true,
52            }),
53        }
54    }

111    /// Binds the endpoint.
112    pub async fn bind(self) -> Result<Endpoint, BindError> {
113        if let Err(error) = self.limits.validate() {
114            tracing::warn!(?error, "invalid endpoint capacity limits");
115            return Err(e!(BindError::InvalidEndpointLimits));
116        }
117        let secret_key = self.secret_key.unwrap_or_else(SecretKey::generate);
118
119        #[cfg(not(wasm_browser))]
120        let simulation_environment = self.simulation_environment;
121        #[cfg(not(wasm_browser))]
122        let runtime_context = simulation_environment
123            .as_ref()
124            .map(crate::simulation::SimulationEnvironment::runtime)
125            .unwrap_or_else(|| {
126                Arc::new(iroh_runtime::RuntimeContext::production(Arc::new(
127                    iroh_runtime::NoopTraceSink,
128                )))
129            });
130
131        #[cfg(not(wasm_browser))]
132        let environment_crypto_provider = simulation_environment
133            .as_ref()
134            .and_then(crate::simulation::SimulationEnvironment::crypto_provider);
135        #[cfg(wasm_browser)]
136        let environment_crypto_provider = None;
137        let crypto_provider = environment_crypto_provider
138            .or(self.crypto_provider)
139            .ok_or_else(|| e!(BindError::InvalidCryptoProvider))?;
140
141        #[cfg(not(wasm_browser))]
142        let simulation_crypto = simulation_environment
143            .as_ref()
144            .map(crate::simulation::SimulationEnvironment::crypto);
145        #[cfg(not(wasm_browser))]
146        let token_key = if let Some(material) = simulation_crypto {
147            RustlsTokenKey::from_key(material.token_key(), &crypto_provider)
148                .ok_or_else(|| e!(BindError::InvalidCryptoProvider))?
149        } else {
150            RustlsTokenKey::new(&mut rand::rng(), &crypto_provider)
151                .ok_or_else(|| e!(BindError::InvalidCryptoProvider))?
152        };
153        #[cfg(wasm_browser)]
154        let token_key = RustlsTokenKey::new(&mut rand::rng(), &crypto_provider)
155            .ok_or_else(|| e!(BindError::InvalidCryptoProvider))?;
156        let token_key = Arc::new(token_key);
157
158        let span = info_span!("endpoint", id = %secret_key.public().fmt_short());
159        let _guard = span.enter();
160
161        let tls_config = tls::TlsConfig::new(
162            secret_key.clone(),
163            self.max_tls_tickets,
164            crypto_provider.clone(),
165        );
166        let static_config = StaticConfig {
167            server_config: tls_config.make_server_config(self.keylog)?,
168            client_config: tls_config.make_client_config(self.keylog)?,
169            tls_config,
170            transport_config: self.transport_config.clone(),
171            token_key,
172            token_store: Arc::new(noq::TokenMemoryCache::default()),
173            #[cfg(not(wasm_browser))]
174            runtime_context: runtime_context.clone(),
175            #[cfg(not(wasm_browser))]
176            simulation_initial_dst_cid_provider: simulation_crypto.map(|material| {
177                socket::deterministic_simulation_initial_dst_cid_provider(material.reset_key())
178            }),
179            limits: self.limits,
180        };
181        let server_config = static_config.create_server_config(self.alpn_protocols);
182
183        #[cfg(not(wasm_browser))]
184        let dns_resolver = self.dns_resolver.unwrap_or_default();
185
186        let metrics = EndpointMetrics::default();
187
188        let tls_config = self
189            .ca_tls_config
190            .unwrap_or_default()
191            .client_config(crypto_provider)
192            .map_err(|err| e!(BindError::InvalidCaRootConfig, err))?;
193
194        let sock_opts = socket::Options {
195            transports: self.transports,
196            secret_key,
197            address_lookup_user_data: self.address_lookup_user_data,
198            proxy_url: self.proxy_url,
199            #[cfg(not(wasm_browser))]
200            dns_resolver,
201            #[cfg(not(wasm_browser))]
202            runtime_context,
203            #[cfg(not(wasm_browser))]
204            ip_socket_factory: simulation_environment
205                .as_ref()
206                .map(crate::simulation::SimulationEnvironment::ip_sockets)
207                .unwrap_or_else(|| Arc::new(crate::simulation::OsIpSocketFactory)),
208            #[cfg(not(wasm_browser))]
209            network_monitor: simulation_environment
210                .as_ref()
211                .map(crate::simulation::SimulationEnvironment::network_monitor),
212            #[cfg(not(wasm_browser))]
213            simulation_port_mapper: simulation_environment
214                .as_ref()
215                .and_then(crate::simulation::SimulationEnvironment::port_mapper),
216            #[cfg(not(wasm_browser))]
217            simulation_relay_connector: simulation_environment
218                .as_ref()
219                .and_then(crate::simulation::SimulationEnvironment::relay_connector),
220            #[cfg(not(wasm_browser))]
221            simulation_preferred_relay: simulation_environment
222                .as_ref()
223                .and_then(crate::simulation::SimulationEnvironment::preferred_relay),
224            #[cfg(not(wasm_browser))]
225            simulation_reset_key: simulation_crypto.map(|material| material.reset_key()),
226            server_config,
227            tls_config,
228            metrics,
229            hooks: self.hooks,
230            path_selector: self.path_selector,
231            portmapper_config: self.portmapper_config,
232            net_report_config: self.net_report_config,
233            static_config,
234            configured_addrs: self.configured_addrs,
235            limits: self.limits,
236        };
237
238        let inner = socket::EndpointInner::bind(sock_opts)
239            .instrument(Span::current())
240            .await?;
241        debug!(
242            id = %inner.static_config.tls_config.secret_key.public(),
243            iroh_version = %env!("CARGO_PKG_VERSION"),
244            "iroh endpoint bound"
245        );
246
247        let ep = Endpoint {
248            inner: Arc::new(inner),
249        };
250
251        // Add Address Lookup mechanisms
252        let address_lookup = ep.address_lookup().expect("just created the endpoint");
253        if let Some(filter) = self.addr_filter {
254            address_lookup.set_addr_filter(filter);
255        }
256        for create_service in self.address_lookup {
257            let service = create_service.into_address_lookup(&ep)?;
258            address_lookup.add_boxed(service);
259        }
260
261        Ok(ep)
262    }

278    /// Binds an IP socket at the provided socket address.
279    ///
280    /// This is an advanced API to tightly control the sockets used by the endpoint. Most
281    /// uses do not need to explicitly bind sockets.
282    ///
283    /// # Warning
284    ///
285    /// - The builder always comes pre-configured with an IPv4 socket to be bound on the
286    ///   *unspecified* address: `0.0.0.0`. This is the equivalent of using `INADDR_ANY`
287    ///   special bind address and results in a socket listening on *all* interfaces
288    ///   available.
289    ///
290    /// - Likewise the builder always comes pre-configured with an IPv6 socket to be bound
291    ///   on the *unspecified* address: `[::]`. This bind is allowed to fail however.
292    ///
293    /// - Adding a bind address removes the pre-configured unspecified bind address for this
294    ///   address family. Use [`Self::bind_addr_with_opts`] to bind additional addresses without
295    ///   replacing the default bind address.
296    ///
297    /// - This should be called at most once for each address family: once for IPv4 and/or
298    ///   once for IPv6. Calling it multiple times for the same address family will result
299    ///   in undefined routing behaviour. To bind multiple sockets of the same address
300    ///   family, use [`Self::bind_addr_with_opts`].
301    ///
302    /// # Description
303    ///
304    /// Requests a socket to be bound on a specific address, with an implied netmask of
305    /// `/0`. This allows restricting binding to only one network interface for a given
306    /// address family.
307    ///
308    /// If the port specified is already in use, binding the endpoint will fail. Using
309    /// port `0` in the socket address assigns a random free port.
310    ///
311    /// # Example
312    ///
313    /// ```no_run
314    /// # #[cfg(with_crypto_provider)]
315    /// # {
316    /// # #[tokio::main]
317    /// # async fn main() -> n0_error::Result<()> {
318    /// # use iroh::{Endpoint, endpoint::presets};
319    /// let endpoint = Endpoint::builder(presets::N0)
320    ///     .clear_ip_transports()
321    ///     .bind_addr("127.0.0.1:0")?
322    ///     .bind_addr("[::1]:0")?
323    ///     .bind()
324    ///     .await?;
325    /// # Ok(()) }
326    /// # }
327    /// ```
328    #[cfg(not(wasm_browser))]
329    pub fn bind_addr<A>(self, addr: A) -> Result<Self, InvalidSocketAddr>
330    where
331        A: ToSocketAddr,
332        <A as ToSocketAddr>::Err: Into<InvalidSocketAddr>,
333    {
334        self.bind_addr_with_opts(addr, BindOpts::default())
335    }

337    /// Binds an IP socket at the provided socket address.
338    ///
339    /// This is an advanced API to tightly control the sockets used by the endpoint. Most
340    /// uses do not need to explicitly bind sockets.
341    ///
342    /// # Warning
343    ///
344    /// - The builder always comes pre-configured with an IPv4 socket to be bound on the
345    ///   *unspecified* address: `0.0.0.0`. This is the equivalent of using `INADDR_ANY`
346    ///   special bind address and results in a socket listening on *all* interfaces
347    ///   available.
348    ///
349    /// - Likewise the builder always comes pre-configured with an IPv6 socket to be bound
350    ///   on the *unspecified* address: `[::]`. This bind is allowed to fail however.
351    ///
352    /// # Description
353    ///
354    /// Requests a socket to be bound on a specific address. This allows restricting binding
355    /// to only one network interface for a given address family.
356    ///
357    /// [`BindOpts::set_prefix_len`] **should** be used to configure the netmask of the
358    /// network interface. This allows outgoing datagrams that start a new network flow to
359    /// be sent over the socket which is attached to the subnet of the destination
360    /// address. If multiple sockets are bound the standard routing-table semantics are
361    /// used: the socket attached to the subnet with the longest prefix matching the
362    /// destination is used. Practically this means the smallest subnets are at the top of
363    /// the routing table, and the first subnet containing the destination address is
364    /// chosen.
365    ///
366    /// If no socket is bound to a subnet that contains the destination address, the notion
367    /// of "default route" is used. At most one socket per address family may be marked as
368    /// the default route using [`BindOpts::set_is_default_route`], and this will be used
369    /// for destinations not contained by the subnets of the bound sockets. This network is
370    /// expected to have a default gateway configured. A socket with a prefix length of `/0`
371    /// will be set as a "default route" implicitly, unless [`BindOpts::set_is_default_route`]
372    /// is set to `false` explicitly.
373    ///
374    /// Be aware that using a subnet with a prefix length of `/0` will always contain all
375    /// destination addresses. It is valid to configure this, but no more than one such
376    /// socket should be bound or the routing will be non-deterministic.
377    ///
378    /// To use a subnet with a non-zero prefix length as the default route in addition to
379    /// being routed when its prefix matches, use [`BindOpts::set_is_default_route`].
380    /// Subnets with a prefix length of zero are always marked as default routes.
381    ///
382    /// Finally note that most outgoing datagrams are part of an existing network flow. That
383    /// is, they are in response to an incoming datagram. In this case the outgoing datagram
384    /// will be sent over the same socket as the incoming datagram was received on, and the
385    /// routing with the prefix length and default route as described above does not apply.
386    ///
387    /// Using port `0` in the socket address assigns a random free port.
388    ///
389    /// If the port specified is already in use, binding the endpoint will fail, unless
390    /// [`BindOpts::set_is_required`] is set to `false`
391    ///
392    /// # Example
393    /// ```no_run
394    /// # #[cfg(with_crypto_provider)]
395    /// # {
396    /// # #[tokio::main]
397    /// # async fn main() -> n0_error::Result<()> {
398    /// # use iroh::{Endpoint, endpoint::{BindOpts, presets}};
399    /// let endpoint = Endpoint::builder(presets::N0)
400    ///     .clear_ip_transports()
401    ///     .bind_addr_with_opts("127.0.0.1:0", BindOpts::default().set_prefix_len(24))?
402    ///     .bind_addr_with_opts("[::1]:0", BindOpts::default().set_prefix_len(48))?
403    ///     .bind()
404    ///     .await?;
405    /// # Ok(()) }
406    /// # }
407    /// ```
408    #[cfg(not(wasm_browser))]
409    pub fn bind_addr_with_opts<A>(
410        mut self,
411        addr: A,
412        opts: BindOpts,
413    ) -> Result<Self, InvalidSocketAddr>
414    where
415        A: ToSocketAddr,
416        <A as ToSocketAddr>::Err: Into<InvalidSocketAddr>,
417    {
418        let addr = addr.to_socket_addr().map_err(Into::into)?;
419        match addr {
420            SocketAddr::V4(addr) => {
421                if self
422                    .transports
423                    .iter()
424                    .any(|t| t.is_ipv4_default() && t.is_user_defined())
425                {
426                    bail!(InvalidSocketAddr::DuplicateDefaultAddr);
427                }
428
429                let ip_net = Ipv4Net::new(*addr.ip(), opts.prefix_len())
430                    .map_err(|_| e!(InvalidSocketAddr::InvalidPrefixLength))?;
431                self.transports.push(TransportConfig::Ip {
432                    config: IpConfig::V4 {
433                        ip_net,
434                        port: addr.port(),
435                        is_required: opts.is_required(),
436                        is_default: opts.is_default_route(),
437                    },
438                    is_user_defined: true,
439                });
440            }
441            SocketAddr::V6(addr) => {
442                if self
443                    .transports
444                    .iter()
445                    .any(|t| t.is_ipv6_default() && t.is_user_defined())
446                {
447                    bail!(InvalidSocketAddr::DuplicateDefaultAddr);
448                }
449
450                let ip_net = Ipv6Net::new(*addr.ip(), opts.prefix_len())
451                    .map_err(|_| e!(InvalidSocketAddr::InvalidPrefixLength))?;
452                self.transports.push(TransportConfig::Ip {
453                    config: IpConfig::V6 {
454                        ip_net,
455                        scope_id: addr.scope_id(),
456                        port: addr.port(),
457                        is_required: opts.is_required(),
458                        is_default: opts.is_default_route(),
459                    },
460                    is_user_defined: true,
461                });
462            }
463        }
464        Ok(self)
465    }

517    /// Sets the relay servers to assist in establishing connectivity.
518    ///
519    /// Relay servers are used to establish initial connection with another iroh endpoint.
520    /// They also perform various functions related to hole punching, see the [crate docs]
521    /// for more details.
522    ///
523    /// By default the [number 0] relay servers are used, see [`RelayMode::Default`].
524    ///
525    /// When using [RelayMode::Custom], the provided `relay_map` must contain at least one
526    /// configured relay endpoint.  If an invalid RelayMap is provided [`bind`]
527    /// will result in an error.
528    ///
529    /// [`bind`]: Builder::bind
530    /// [crate docs]: crate
531    /// [number 0]: https://n0.computer
532    pub fn relay_mode(mut self, relay_mode: RelayMode) -> Self {
533        let transport: Option<_> = relay_mode.into();
534        match transport {
535            Some(transport) => {
536                if let Some(og) = self
537                    .transports
538                    .iter_mut()
539                    .find(|t| matches!(t, TransportConfig::Relay { .. }))
540                {
541                    *og = transport;
542                } else {
543                    self.transports.push(transport);
544                }
545            }
546            None => {
547                self.transports
548                    .retain(|t| !matches!(t, TransportConfig::Relay { .. }));
549            }
550        }
551        self
552    }

808    /// It is not covered by semantic versioning guarantees and may change in any release
809    /// without a major version bump.
810    ///
811    /// </div>
812    ///
813    /// [`PathSelector`]: socket::remote_map::PathSelector
814    #[cfg(feature = "unstable-custom-transports")]
815    pub fn path_selector(mut self, selector: Arc<dyn PathSelector>) -> Self {
816        self.path_selector = selector;
