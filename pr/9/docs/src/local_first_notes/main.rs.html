19const ECHO_ALPN: &[u8] = b"/holon/local-first/notes-echo/1";
20const NOTE_KEY: &[u8] = b"notes/demo.md";
21const MAX_ECHO_BYTES: usize = 4 * 1024;
22const SYNC_TIMEOUT: Duration = Duration::from_secs(15);
23
24#[tokio::main]
25async fn main() -> Result<()> {
26    let mut args = std::env::args_os().skip(1);
27    let root = PathBuf::from(
28        args.next()
29            .context("usage: local-first-notes DATA_ROOT [TEXT]")?,
30    );
31    let note = args
32        .next()
33        .map(|text| text.to_string_lossy().into_owned())
34        .unwrap_or_else(|| "hello from a local-first Iroh application".to_owned());
35    ensure!(
36        args.next().is_none(),
37        "usage: local-first-notes DATA_ROOT [TEXT]"
38    );
39    let node_a_root = root.join("node-a");
40    let node_b_root = root.join("node-b");
41
42    let node_a = StandardBundle::persistent(&node_a_root)
43        .local_only()
44        .start()
45        .await?;
46    let node_b = StandardBundle::persistent(&node_b_root)
47        .local_only()
48        .protocol(ECHO_ALPN, Echo)?
49        .start()
50        .await?;
51    echo(&node_a, &node_b).await?;
52
53    let author_a = node_a.docs().author_default().await?;
54    let author_b = node_b.docs().author_default().await?;
55    let doc_a = node_a.docs().create().await?;
56    let namespace = doc_a.id();
57    let note = Bytes::from(note);
58    let tag = node_a.blobs().blobs().add_bytes(note.clone()).await?;
59    doc_a
60        .set_hash(
61            author_a,
62            NOTE_KEY,
63            tag.hash,
64            u64::try_from(note.len()).context("note is too large")?,
65        )
66        .await?;
67    let ticket = doc_a
68        .share(ShareMode::Read, AddrInfoOptions::RelayAndAddresses)
69        .await?;
70    let invitation = ticket.to_string();
71    ensure!(invitation.len() <= 64 * 1024, "invitation is too large");
72    let doc_b = node_b
73        .docs()
74        .import(invitation.parse::<iroh_docs::DocTicket>()?)
75        .await?;
76    let received = read_when_ready(&node_b, &doc_b, author_a, tag.hash).await?;
77    ensure!(received == note, "nodes disagree about note bytes");
78    ensure!(
79        doc_b
80            .set_bytes(
81                author_b,
82                Bytes::from_static(b"notes/not-allowed.md"),
83                Bytes::from_static(b"write"),
84            )
85            .await
86            .is_err(),
87        "read-only invitation allowed a write"
88    );
89
90    println!("node A: {}", node_a.endpoint_id());
91    println!("node B: {}", node_b.endpoint_id());
92    println!("document: {namespace}");
93    println!("read-only invitation: {invitation}");
94    println!("node B read: {}", String::from_utf8_lossy(&received));
95
96    node_b.shutdown().await?;
97    node_a.shutdown().await?;
98    drop(node_b);
99    drop(node_a);
100
101    let restarted_a = StandardBundle::persistent(node_a_root)
102        .local_only()
103        .start()
104        .await?;
105    let reopened = restarted_a
106        .docs()
107        .open(namespace)
108        .await?
109        .context("document was not durable")?;
110    ensure!(
111        read_when_ready(&restarted_a, &reopened, author_a, tag.hash).await? == note,
112        "note was not durable"
113    );
114    println!("restart retained identity and note state");
115    restarted_a.shutdown().await?;
116    Ok(())
117}

119async fn read_when_ready(
120    application: &Application,
121    doc: &Doc,
122    author: AuthorId,
123    hash: iroh_blobs::Hash,
124) -> Result<Bytes> {
125    tokio::time::timeout(SYNC_TIMEOUT, async {
126        loop {
127            if let Some(entry) = doc.get_exact(author, NOTE_KEY, false).await?
128                && entry.content_hash() == hash
129            {
130                let mut reader = application.blobs().blobs().reader(hash);
131                let mut bytes = Vec::new();
132                if reader.read_to_end(&mut bytes).await.is_ok() {
133                    ensure!(iroh_blobs::Hash::new(&bytes) == hash, "hash mismatch");
134                    return Ok(Bytes::from(bytes));
135                }
136            }
137            tokio::time::sleep(Duration::from_millis(20)).await;
138        }
139    })
140    .await
141    .context("note synchronization timed out")?
142}

144async fn echo(client: &Application, server: &Application) -> Result<()> {
145    let connection = client
146        .endpoint()
147        .connect(server.endpoint().addr(), ECHO_ALPN)
148        .await?;
149    let (mut send, mut recv) = connection.open_bi().await?;
150    send.write_all(b"custom protocol works").await?;
151    send.finish()?;
152    ensure!(
153        recv.read_to_end(MAX_ECHO_BYTES).await? == b"custom protocol works",
154        "echo mismatch"
155    );
156    connection.close(0_u32.into(), b"complete");
157    Ok(())
158}
159
160#[derive(Clone, Debug)]
161struct Echo;
162
163impl ProtocolHandler for Echo {
164    async fn accept(&self, connection: Connection) -> Result<(), AcceptError> {
165        let (mut send, mut recv) = connection.accept_bi().await?;
166        let request = recv
167            .read_to_end(MAX_ECHO_BYTES)
168            .await
169            .map_err(AcceptError::from_err)?;
170        send.write_all(&request)
171            .await
172            .map_err(AcceptError::from_err)?;
173        send.finish().map_err(AcceptError::from_err)?;
174        connection.closed().await;
175        Ok(())
176    }
177}
