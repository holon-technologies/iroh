6/// TLS certificate configuration.
7#[derive(Debug)]
8#[non_exhaustive]
9pub enum CertConfig {
10    /// Use Let's Encrypt.
11    #[cfg(feature = "server-acme")]
12    LetsEncrypt {
13        /// Configuration for the ACME client.
14        acme_config: AcmeConfig,
15        /// Builder for the [`rustls::ServerConfig`].
16        ///
17        /// The ACME resolver will be injected when starting the server.
18        server_config_builder: rustls::ConfigBuilder<rustls::ServerConfig, WantsServerCert>,
19    },
20    /// Use a TLS key and certificate chain that can be reloaded.
21    Manual {
22        /// The [`rustls::ServerConfig`] to use.
23        ///
24        /// This needs to have the certificates or a certificate loader, it will be used by the server as-is.
25        server_config: rustls::ServerConfig,
26    },
27}
28
29/// Configuration for the ACME client.
30#[cfg(feature = "server-acme")]
31#[derive(Debug, Clone)]
32pub struct AcmeConfig {
33    pub(crate) directory_url: String,
34    pub(crate) domains: Vec<String>,
35    pub(crate) contact: Vec<String>,
36    pub(crate) cache_path: Option<PathBuf>,
37    pub(crate) tls_config: CaTlsConfig,
38}
39
40#[cfg(feature = "server-acme")]
41impl AcmeConfig {
42    /// Creates a new [`AcmeConfig`] with a ACME directory URL.
43    pub fn new(directory_url: String) -> Self {
44        Self {
45            directory_url,
46            domains: Vec::new(),
47            contact: Vec::new(),
48            cache_path: None,
49            tls_config: CaTlsConfig::default(),
50        }
51    }
52
53    /// Creates a new [`AcmeConfig`] with the Let's Encrypt directory URL.
54    pub fn letsencrypt(production: bool) -> Self {
55        let url = if production {
56            LETS_ENCRYPT_PRODUCTION_DIRECTORY
57        } else {
58            LETS_ENCRYPT_STAGING_DIRECTORY
59        };
60        Self::new(url.to_string())
61    }
62
63    /// Provides the list of domains for which certificates should be obtained.
64    pub fn domains(mut self, domains: Vec<String>) -> Self {
65        self.domains = domains;
66        self
67    }
68
69    /// Provides a list of contacts for the account.
70    ///
71    /// Note that email addresses must include a `mailto:` prefix.
72    pub fn contact(mut self, contact: Vec<String>) -> Self {
73        self.contact = contact;
74        self
75    }
76
77    /// Sets the directory where to cache certificates.
78    ///
79    /// If not called certificates will not be cached.
80    pub fn cache_path(mut self, path: PathBuf) -> Self {
81        self.cache_path = Some(path);
82        self
83    }
84
85    /// Sets the [`CaTlsConfig`] used to verify the ACME server's TLS certificate.
86    ///
87    /// Defaults to [`CaTlsConfig::embedded`]. Set a config with extra roots when targeting
88    /// an ACME server whose certificate is not signed by a publicly trusted CA, such as a
89    /// local test server.
90    pub fn tls_config(mut self, tls_config: CaTlsConfig) -> Self {
91        self.tls_config = tls_config;
92        self
93    }
94}
