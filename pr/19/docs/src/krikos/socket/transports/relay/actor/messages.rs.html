47/// Connection state of the home relay.
48///
49/// Published via [`HomeRelayWatch`] so that [`Endpoint::online`] and the public
50/// [`Endpoint::home_relay_status`] watcher can observe the connection state.
51/// This type is `pub(crate)`; the public surface lives on
52/// [`crate::endpoint::RelayStatus`], and this enum is intentionally free to
53/// evolve without affecting the public API.
54///
55/// [`Endpoint::online`]: crate::Endpoint::online
56/// [`Endpoint::home_relay_status`]: crate::Endpoint::home_relay_status
57#[derive(Debug, Clone)]
58pub(crate) enum RelayConnectionState {
59    /// Dialing or performing the relay handshake.
60    Connecting,
61    /// Connected and handshaked.
62    Connected,
63    /// Not connected. Either the connection was lost after having been
64    /// established, or an attempt to connect failed.
65    ///
66    /// `last_error` carries the most recent connection error, if any. The
67    /// initial transition into this state (before any attempt has produced
68    /// an error) carries `None`.
69    ///
70    /// The `Arc` is compared by pointer identity: each new failure produces
71    /// a fresh allocation, so the watcher fires on every new error.
72    Disconnected { last_error: Option<Arc<AnyError>> },
73}
74
75impl RelayConnectionState {
76    pub(crate) fn is_connected(&self) -> bool {
77        matches!(self, Self::Connected)
78    }
79
80    pub(crate) fn last_error(&self) -> Option<&Arc<AnyError>> {
81        match self {
82            Self::Disconnected { last_error } => last_error.as_ref(),
83            _ => None,
84        }
85    }
86}
87
88impl PartialEq for RelayConnectionState {
89    fn eq(&self, other: &Self) -> bool {
90        match (self, other) {
91            (Self::Connecting, Self::Connecting) | (Self::Connected, Self::Connected) => true,
92            (Self::Disconnected { last_error: a }, Self::Disconnected { last_error: b }) => {
93                match (a, b) {
94                    (None, None) => true,
95                    (Some(a), Some(b)) => Arc::ptr_eq(a, b),
96                    _ => false,
97                }
98            }
99            _ => false,
100        }
101    }
102}

106/// Shared watchable for the home relay URL and connection status.
107///
108/// Owned by [`RelayActor`] and cloned into each [`ActiveRelayActor`].
109///
110/// # Write discipline
111///
112/// The [`RelayActor`] writes the URL (via [`Self::set`] and [`Self::clear`]).
113/// Each [`ActiveRelayActor`] updates only the status (via [`Self::set_status`]),
114/// which guards against stale writes: if another relay has become home since this actor
115/// was designated, the write is silently dropped.
116#[derive(Debug, Clone)]
117pub(crate) struct HomeRelayWatch {
118    inner: Watchable<Option<RelayStatus>>,
119}
120
121impl Default for HomeRelayWatch {
122    fn default() -> Self {
123        Self {
124            inner: Watchable::new(None),
125        }
126    }
127}
128
129impl HomeRelayWatch {
130    /// Set the home relay URL and status. Used by [`RelayActor`] on relay changes.
131    pub(super) fn set(&self, url: RelayUrl, state: RelayConnectionState) {
132        let _ = self.inner.set(Some(RelayStatus::new(url, state)));
133    }
134
135    /// Clear the home relay (no preferred relay). Used by [`RelayActor`].
136    pub(super) fn clear(&self) {
137        let _ = self.inner.set(None);
138    }
139
140    /// Update the status, but only if `url` is still the current home relay.
141    ///
142    /// This is the only write method [`ActiveRelayActor`] should use. It prevents a
143    /// demoted actor from overwriting a newer home relay's status: the [`RelayActor`]
144    /// updates the URL in the watchable *before* sending `SetHomeRelay(false)`, so by
145    /// the time the old actor tries to write, the URL no longer matches.
146    pub(super) fn set_status(&self, url: &RelayUrl, state: RelayConnectionState) {
147        if self.inner.get().as_ref().map(RelayStatus::url) == Some(url) {
148            let _ = self.inner.set(Some(RelayStatus::new(url.clone(), state)));
149        }
150    }
151
152    pub(super) fn get(&self) -> Option<RelayStatus> {
153        self.inner.get()
154    }
155
156    pub(crate) fn watch(&self) -> n0_watcher::Direct<Option<RelayStatus>> {
157        self.inner.watch()
158    }
159}
