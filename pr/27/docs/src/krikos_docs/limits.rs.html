1//! Resource limits enforced at document protocol and actor boundaries.
2
3use std::time::Duration;
4
5/// Largest accepted encoded sync frame.
6pub const MAX_SYNC_MESSAGE_SIZE: usize = 4 * 1024 * 1024;
7/// Maximum reconciliation parts accepted in one sync message.
8pub const MAX_SYNC_MESSAGE_PARTS: usize = 1024;
9/// Maximum signed entries accepted in one sync message.
10pub const MAX_ENTRIES_PER_SYNC_MESSAGE: usize = 2048;
11/// Maximum encoded bytes accepted for one document ticket.
12pub const MAX_TICKET_BYTES: usize = 512 * 1024;
13/// Maximum peers accepted in one document ticket or start-sync request.
14pub const MAX_PEERS_PER_DOCUMENT: usize = 256;
15/// Maximum simultaneously active document gossip subscriptions.
16pub const MAX_ACTIVE_DOCUMENTS: usize = 1024;
17/// Maximum simultaneous inbound and outbound reconciliation sessions.
18pub const MAX_LIVE_SYNC_SESSIONS: usize = 256;
19/// Maximum subscribers retained for one document.
20pub const MAX_SUBSCRIBERS_PER_DOCUMENT: usize = 256;
21/// Maximum unresolved or downloading content hashes retained globally.
22pub const MAX_PENDING_CONTENT_HASHES: usize = 16_384;
23/// Maximum content hashes retained for one document.
24pub const MAX_PENDING_CONTENT_HASHES_PER_DOCUMENT: usize = 4096;
25
26/// Capacity of the storage actor's command queue.
27pub const STORE_ACTION_QUEUE_CAPACITY: usize = 1024;
28/// Capacity of the live-sync actor's command queue.
29pub const LIVE_ACTOR_QUEUE_CAPACITY: usize = 64;
30/// Capacity of each public event subscription queue.
31pub const SUBSCRIPTION_QUEUE_CAPACITY: usize = 256;
32/// Capacity of the internal replica-event queue.
33pub const REPLICA_EVENT_QUEUE_CAPACITY: usize = 1024;
34/// Capacity of the local RPC actor queue.
35pub const RPC_ACTOR_QUEUE_CAPACITY: usize = 64;
36/// Capacity of the GC protection response queue.
37pub const GC_PROTECTION_QUEUE_CAPACITY: usize = 64;
38
39/// Maximum age of an open database write transaction.
40pub const MAX_DATABASE_COMMIT_DELAY: Duration = Duration::from_millis(500);
41/// Time allowed for graceful engine shutdown.
42pub const SHUTDOWN_TIMEOUT: Duration = Duration::from_secs(10);
43
44/// A document protocol input exceeded a named resource limit.
45#[derive(Debug, Clone, Copy, PartialEq, Eq, thiserror::Error)]
46#[error("{resource} contains {actual} items or bytes, maximum is {maximum}")]
47pub struct LimitError {
48    /// Name of the bounded resource.
49    pub resource: &'static str,
50    /// Observed size.
51    pub actual: usize,
52    /// Maximum accepted size.
53    pub maximum: usize,
54}
55
56impl LimitError {
57    pub(crate) fn new(resource: &'static str, actual: usize, maximum: usize) -> Self {
58        Self {
59            resource,
60            actual,
61            maximum,
62        }
63    }
64}
65
66const _: () = {
67    assert!(MAX_SYNC_MESSAGE_SIZE > 0);
68    assert!(MAX_SYNC_MESSAGE_PARTS > 0);
69    assert!(MAX_ENTRIES_PER_SYNC_MESSAGE >= MAX_SYNC_MESSAGE_PARTS);
70    assert!(MAX_TICKET_BYTES >= MAX_SYNC_MESSAGE_SIZE / 8);
71    assert!(MAX_PEERS_PER_DOCUMENT > 0);
72    assert!(MAX_ACTIVE_DOCUMENTS >= MAX_LIVE_SYNC_SESSIONS);
73    assert!(MAX_SUBSCRIBERS_PER_DOCUMENT > 0);
74    assert!(MAX_PENDING_CONTENT_HASHES >= MAX_PENDING_CONTENT_HASHES_PER_DOCUMENT);
75    assert!(STORE_ACTION_QUEUE_CAPACITY > 0);
76    assert!(LIVE_ACTOR_QUEUE_CAPACITY > 0);
77    assert!(SUBSCRIPTION_QUEUE_CAPACITY > 0);
78    assert!(REPLICA_EVENT_QUEUE_CAPACITY > 0);
79    assert!(RPC_ACTOR_QUEUE_CAPACITY > 0);
80    assert!(GC_PROTECTION_QUEUE_CAPACITY > 0);
81};
