65/// Evidence class explaining why a provider-log leaf must remain available after compaction.
66#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
67pub enum ProviderRetentionClass {
68    /// Current checkpoint tip, or every branch while the account remains forked.
69    CheckpointLineage,
70    /// Controller removal or retirement tombstone.
71    ControllerTombstone,
72    /// Device revocation or replacement tombstone.
73    DeviceTombstone,
74    /// Evidence belonging to an unresolved account fork.
75    UnresolvedFork,
76    /// Pending, dual, retired, or aborted cryptographic migration evidence.
77    CryptoMigration,
78    /// Pending or completed recovery lineage.
79    Recovery,
80    /// Provider signing-key or log-generation rotation evidence.
81    ProviderRotation,
82    /// Known signed provider equivocation evidence.
83    Equivocation,
84}
85
86impl ProviderRetentionClass {
87    pub(crate) const fn code(self) -> u16 {
88        match self {
89            Self::CheckpointLineage => 1,
90            Self::ControllerTombstone => 2,
91            Self::DeviceTombstone => 3,
92            Self::UnresolvedFork => 4,
93            Self::CryptoMigration => 5,
94            Self::Recovery => 6,
95            Self::ProviderRotation => 7,
96            Self::Equivocation => 8,
97        }
98    }
99
100    #[cfg(feature = "provider-store")]
101    pub(crate) fn from_code(code: u16) -> Result<Self, IdentityError> {
102        match code {
103            1 => Ok(Self::CheckpointLineage),
104            2 => Ok(Self::ControllerTombstone),
105            3 => Ok(Self::DeviceTombstone),
106            4 => Ok(Self::UnresolvedFork),
107            5 => Ok(Self::CryptoMigration),
108            6 => Ok(Self::Recovery),
109            7 => Ok(Self::ProviderRotation),
110            8 => Ok(Self::Equivocation),
111            _ => Err(IdentityError::StorageCorruption),
112        }
113    }
114}

152/// Complete sorted retention inventory for one exact source tree size.
153#[derive(Debug, Clone, PartialEq, Eq)]

155    tree_size: u64,
156    items: Vec<ProviderRetentionItem>,
157    audit_artifacts: Vec<ProviderAuditArtifact>,
158}
159
160impl ProviderRetentionInventory {
161    /// Validate, sort, and deduplicate the caller's complete retained-evidence inventory.
162    pub fn new(tree_size: u64, items: Vec<ProviderRetentionItem>) -> Result<Self, IdentityError> {
163        Self::with_audit_artifacts(tree_size, items, Vec::new())
164    }
165
166    /// Validate all leaf reasons and sorted non-leaf rollback/equivocation artifacts.
167    pub fn with_audit_artifacts(
168        tree_size: u64,
169        mut items: Vec<ProviderRetentionItem>,
170        mut audit_artifacts: Vec<ProviderAuditArtifact>,
171    ) -> Result<Self, IdentityError> {
172        if tree_size > MAX_MERKLE_LOG_LEAVES as u64 {
173            return Err(IdentityError::limit(
174                "provider compaction tree size",
175                usize::try_from(tree_size).unwrap_or(usize::MAX),
176                MAX_MERKLE_LOG_LEAVES,
177            ));
178        }
179        items.sort_unstable();
180        if items.windows(2).any(|pair| pair[0] == pair[1]) {
181            return Err(IdentityError::DuplicateElement {
182                resource: "provider retention inventory",
183            });
184        }
185        if items.iter().any(|item| item.leaf_index >= tree_size) {
186            return Err(IdentityError::InvalidRelationship {
187                resource: "provider retention leaf/tree size",
188            });
189        }
190        if audit_artifacts.len() > MAX_PROVIDER_AUDIT_ARTIFACTS {
191            return Err(IdentityError::limit(
192                "provider retained audit artifacts",
193                audit_artifacts.len(),
194                MAX_PROVIDER_AUDIT_ARTIFACTS,
195            ));
196        }
197        audit_artifacts.sort_unstable_by_key(|artifact| (artifact.sequence(), artifact.kind()));
198        if audit_artifacts.windows(2).any(|pair| {
199            pair[0].sequence() == pair[1].sequence() && pair[0].kind() == pair[1].kind()
200        }) {
201            return Err(IdentityError::DuplicateElement {
202                resource: "provider retained audit artifacts",
203            });
204        }
205        Ok(Self {
206            tree_size,
207            items,
208            audit_artifacts,
209        })
210    }
211
212    /// Exact source tree size governed by this inventory.
213    pub const fn tree_size(&self) -> u64 {
214        self.tree_size
215    }
216
217    /// Sorted unique retained leaf/reason pairs.
218    pub fn items(&self) -> &[ProviderRetentionItem] {
219        &self.items
220    }
221
222    /// Sorted exact rollback/equivocation evidence retained outside the provider leaf tree.
223    pub fn audit_artifacts(&self) -> &[ProviderAuditArtifact] {
224        &self.audit_artifacts
225    }
226}

348    pub(super) fn validate_wire(&self) -> Result<(), IdentityError> {
349        let source_tree_size =
350            usize::try_from(self.source_tree_size).map_err(|_| IdentityError::LimitExceeded {
351                resource: "provider compaction source tree size",
352                actual: usize::MAX,
353                maximum: MAX_MERKLE_LOG_LEAVES,
354            })?;
355        if self.format_version != 1 {
356            return Err(IdentityError::UnsupportedVersion {
357                version: self.format_version,
358            });
359        }
360        if source_tree_size > MAX_MERKLE_LOG_LEAVES {
361            return Err(IdentityError::limit(
362                "provider compaction source tree size",
363                source_tree_size,
364                MAX_MERKLE_LOG_LEAVES,
365            ));
366        }
367        let digests = [
368            self.source_tree_root,
369            self.archive_commitment,
370            self.generation_commitment,
371            self.audit_commitment,
372            self.audit_artifact_commitment,
373            self.inventory_commitment,
374            self.retained_evidence_commitment,
375        ];
376        if digests
377            .iter()
378            .any(|digest| digest.algorithm() != HashAlgorithm::Blake3_256)
379            || self.archive_commitment
380                != provider_recovery_commitment(
381                    self.generation_commitment,
382                    self.audit_commitment,
383                    self.audit_artifact_commitment,
384                )?
385        {
386            return Err(IdentityError::InvalidProof);
387        }
388        let mut previous_end = 0_u64;
389        for range in &self.retained_ranges {
390            if range.start >= range.end_exclusive
391                || range.end_exclusive > self.source_tree_size
392                || range.start < previous_end
393            {
394                return Err(IdentityError::InvalidRelationship {
395                    resource: "provider compaction retained ranges",
396                });
397            }
398            previous_end = range.end_exclusive;
399        }
400        Ok(())
401    }
402
403    pub(super) fn retained_leaf_indices(&self) -> Result<Vec<u64>, IdentityError> {
404        let mut indices = Vec::new();
405        for range in &self.retained_ranges {
406            let length = range
407                .end_exclusive
408                .checked_sub(range.start)
409                .ok_or(IdentityError::StorageCorruption)?;
410            let length = usize::try_from(length).map_err(|_| IdentityError::StorageCorruption)?;
411            if indices.len().saturating_add(length) > MAX_MERKLE_LOG_LEAVES {
412                return Err(IdentityError::StorageCorruption);
413            }
414            indices.extend(range.start..range.end_exclusive);
415        }
416        Ok(indices)
417    }

433    pub(super) fn validate_generation(
434        &self,
435        provider_id: ProviderId,
436        log_id: ProviderLogId,
437        key_version: ProviderKeyVersion,
438        leaf_hashes: &[Digest],
439    ) -> Result<(), IdentityError> {
440        let source_len =
441            usize::try_from(self.source_tree_size).map_err(|_| IdentityError::StorageCorruption)?;
442        if source_len > leaf_hashes.len() {
443            return Err(IdentityError::StorageCorruption);
444        }
445        let source_root = crate::merkle::AppendOnlyMerkleLog::from_leaf_hashes(
446            leaf_hashes[..source_len].to_vec(),
447        )?
448        .root()?;
449        if self.format_version != 1
450            || self.provider_id != provider_id
451            || self.log_id != log_id
452            || self.key_version != key_version
453            || self.source_tree_root != source_root
454            || self.archive_commitment
455                != provider_recovery_commitment(
456                    self.generation_commitment,
457                    self.audit_commitment,
458                    self.audit_artifact_commitment,
459                )?
460            || self.archive_commitment.algorithm() != HashAlgorithm::Blake3_256
461            || self.generation_commitment.algorithm() != HashAlgorithm::Blake3_256
462            || self.audit_commitment.algorithm() != HashAlgorithm::Blake3_256
463            || self.audit_artifact_commitment.algorithm() != HashAlgorithm::Blake3_256
464            || self.inventory_commitment.algorithm() != HashAlgorithm::Blake3_256
465            || self.retained_evidence_commitment.algorithm() != HashAlgorithm::Blake3_256
466        {
467            return Err(IdentityError::StorageCorruption);
468        }
469        let mut previous_end = 0;
470        for range in &self.retained_ranges {
471            if range.start >= range.end_exclusive
472                || range.end_exclusive > self.source_tree_size
473                || range.start < previous_end
474            {
475                return Err(IdentityError::StorageCorruption);
476            }
477            previous_end = range.end_exclusive;
478        }
479        Ok(())
480    }

483        &self,
484        inventory: &ProviderRetentionInventory,
485        retained_evidence_commitment: Digest,
486    ) -> Result<(), IdentityError> {
487        if self.inventory_commitment != inventory_commitment(inventory)?

493        }
494        if self.retained_ranges != retained_ranges(inventory)? {
495            return Err(IdentityError::StorageCorruption);
496        }
497        Ok(())
498    }

582/// Derive the semantic minimum leaf/reason inventory from authenticated generation state.
583///
584/// The unique current checkpoint tip remains locally queryable; a fork retains every known branch.
585/// Additional destructive/migration/recovery/rotation classes are derived from verified checkpoint
586/// events, while complete ancestry remains in the exact recovery archive. Callers may retain more,
587/// but cannot omit or relabel these mandatory items.
588pub fn derive_provider_retention_inventory(
589    source: &ProviderRecoveryExport,
590) -> Result<ProviderRetentionInventory, IdentityError> {
591    source.validate()?;
592    let generation = source.generation();
593    let source_store = MemoryProviderStore::restore_generation(generation.clone())?;
594    let snapshot = source_store.snapshot()?;
595    let checkpoint_index =
596        rebuild_checkpoint_index(&generation.entries, &generation.checkpoint_bundles)?;
597    let mut required_lineage = std::collections::BTreeSet::new();
598    for index in &checkpoint_index {
599        if index.forked {
600            for bundle in &generation.checkpoint_bundles {
601                let checkpoint = bundle.verified_checkpoint();
602                if checkpoint.checkpoint().body().account_id() == index.account_id {
603                    required_lineage.insert(checkpoint.checkpoint_id());
604                }
605            }
606        } else if let Some(checkpoint_id) = index.current_checkpoint_id {
607            // The current tip remains locally queryable. Its older ancestry is recoverable from
608            // the exact verified archive and deliberately releasable after sealing.
609            required_lineage.insert(checkpoint_id);
610        }
611    }
612    let mut items = Vec::new();
613    for (index, entry) in generation.entries.iter().enumerate() {
614        let leaf_index = u64::try_from(index).map_err(|_| IdentityError::ArithmeticOverflow {
615            resource: "provider mandatory retention leaf index",
616        })?;
617        match entry.subject() {
618            ProviderLogSubject::EventIntent(_) => {
619                items.push(ProviderRetentionItem::new(
620                    leaf_index,
621                    ProviderRetentionClass::Recovery,
622                )?);
623            }
624            ProviderLogSubject::Checkpoint(checkpoint_id) => {
625                if required_lineage.contains(&checkpoint_id) {
626                    items.push(ProviderRetentionItem::new(
627                        leaf_index,
628                        ProviderRetentionClass::CheckpointLineage,
629                    )?);
630                }
631                if checkpoint_index
632                    .iter()
633                    .any(|retained| retained.account_id == entry.account_id() && retained.forked)
634                {
635                    items.push(ProviderRetentionItem::new(
636                        leaf_index,
637                        ProviderRetentionClass::UnresolvedFork,
638                    )?);
639                }
640                let bundle = generation
641                    .checkpoint_bundles
642                    .iter()
643                    .find(|bundle| {
644                        let checkpoint = bundle.verified_checkpoint();
645                        checkpoint.checkpoint().body().account_id() == entry.account_id()
646                            && checkpoint.checkpoint_id() == checkpoint_id
647                    })
648                    .ok_or(IdentityError::StorageCorruption)?;
649                for event in bundle.events() {
650                    add_operation_retention_items(
651                        &mut items,
652                        leaf_index,
653                        event.body().operation().kind(),
654                    )?;
655                }
656            }
657        }
658    }
659    items.sort_unstable();
660    items.dedup();
661    ProviderRetentionInventory::with_audit_artifacts(
662        snapshot.tree_size(),
663        items,
664        source.artifacts().to_vec(),
665    )
666}

668fn build_manifest(
669    source: &ProviderRecoveryExport,
670    mirror: &ProviderRecoveryExport,
671    inventory: &ProviderRetentionInventory,
672) -> Result<ProviderCompactionManifest, IdentityError> {
673    source.validate()?;
674    mirror.validate()?;
675    let source_store = MemoryProviderStore::restore_generation(source.generation().clone())?;
676    let mirror_store = MemoryProviderStore::restore_generation(mirror.generation().clone())?;
677    let source_snapshot = source_store.snapshot()?;
678    if source != mirror || source_snapshot != mirror_store.snapshot()? {
679        return Err(IdentityError::InvalidProof);
680    }
681    if inventory.tree_size != source_snapshot.tree_size() {
682        return Err(IdentityError::InvalidRelationship {
683            resource: "provider compaction inventory tree size",
684        });
685    }
686    let mandatory = derive_provider_retention_inventory(source)?;
687    if mandatory
688        .items
689        .iter()
690        .any(|required| !inventory.items.contains(required))
691    {
692        return Err(IdentityError::InvalidRelationship {
693            resource: "provider compaction mandatory retention inventory",
694        });
695    }
696    if inventory.audit_artifacts != mandatory.audit_artifacts {
697        return Err(IdentityError::InvalidRelationship {
698            resource: "provider compaction mandatory audit artifacts",
699        });
700    }
701    for artifact in &inventory.audit_artifacts {
702        artifact.verify(source.generation().provider(), source.generation().log_id())?;
703    }
704    let retained_evidence_commitment =
705        retained_provider_evidence_commitment(source.generation(), inventory)?;
706    Ok(ProviderCompactionManifest {
707        format_version: 1,
708        provider_id: source.generation().provider.id()?,
709        log_id: source.generation().log_id,
710        key_version: source.generation().key_version,
711        source_tree_size: source_snapshot.tree_size(),
712        source_tree_root: source_snapshot.tree_root(),
713        archive_commitment: mirror.recovery_commitment(),
714        generation_commitment: mirror.generation_commitment(),
715        audit_commitment: mirror.audit_commitment(),
716        audit_artifact_commitment: mirror.artifact_commitment(),
717        inventory_commitment: inventory_commitment(inventory)?,
718        retained_evidence_commitment,
719        retained_ranges: retained_ranges(inventory)?,
720    })
721}
722
723fn add_operation_retention_items(
724    items: &mut Vec<ProviderRetentionItem>,
725    leaf_index: u64,
726    operation: OperationKind,
727) -> Result<(), IdentityError> {
728    let mut retain = |class| {
729        items.push(ProviderRetentionItem::new(leaf_index, class)?);
730        Ok::<(), IdentityError>(())
731    };
732    match operation {
733        OperationKind::RemoveController => retain(ProviderRetentionClass::ControllerTombstone)?,
734        OperationKind::RevokeDevice | OperationKind::RotateDeviceKeys => {
735            retain(ProviderRetentionClass::DeviceTombstone)?;
736        }
737        OperationKind::ResolveFork => {
738            retain(ProviderRetentionClass::UnresolvedFork)?;
739            retain(ProviderRetentionClass::ControllerTombstone)?;
740            retain(ProviderRetentionClass::DeviceTombstone)?;
741        }
742        OperationKind::BeginCryptoMigration
743        | OperationKind::ActivateCryptoMigration
744        | OperationKind::RetireCryptoSuite
745        | OperationKind::UpgradeProtocol => retain(ProviderRetentionClass::CryptoMigration)?,
746        OperationKind::ChangeRecoveryPolicy
747        | OperationKind::BeginRecovery
748        | OperationKind::VetoRecovery
749        | OperationKind::CancelRecovery => retain(ProviderRetentionClass::Recovery)?,
750        OperationKind::FinalizeRecovery => {
751            retain(ProviderRetentionClass::Recovery)?;
752            retain(ProviderRetentionClass::ControllerTombstone)?;
753            retain(ProviderRetentionClass::DeviceTombstone)?;
754        }
755        OperationKind::ChangeProviderPolicy => retain(ProviderRetentionClass::ProviderRotation)?,
756        OperationKind::RetireAccount => {
757            retain(ProviderRetentionClass::ControllerTombstone)?;
758            retain(ProviderRetentionClass::DeviceTombstone)?;
759        }
760        OperationKind::AuthorizeDevice
761        | OperationKind::UpdateDeviceAuthorization
762        | OperationKind::UpdateDeviceMetadata
763        | OperationKind::SuspendDevice
764        | OperationKind::ReinstateDevice
765        | OperationKind::AddController
766        | OperationKind::ChangeControlPolicy => {}
767    }
768    Ok(())
769}
770
771fn retained_ranges(
772    inventory: &ProviderRetentionInventory,
773) -> Result<Vec<ProviderRetainedRange>, IdentityError> {
774    let mut indices = inventory
775        .items
776        .iter()
777        .map(|item| item.leaf_index)
778        .collect::<Vec<_>>();
779    indices.sort_unstable();
780    indices.dedup();
781    let mut ranges = Vec::<ProviderRetainedRange>::new();
782    for index in indices {
783        match ranges.last_mut() {
784            Some(range) if range.end_exclusive == index => {
785                range.end_exclusive =
786                    index
787                        .checked_add(1)
788                        .ok_or(IdentityError::ArithmeticOverflow {
789                            resource: "provider retained range",
790                        })?;
791            }
792            _ => ranges.push(ProviderRetainedRange {
793                start: index,
794                end_exclusive: index
795                    .checked_add(1)
796                    .ok_or(IdentityError::ArithmeticOverflow {
797                        resource: "provider retained range",
798                    })?,
799            }),
800        }
801    }
802    Ok(ranges)
803}

805pub(super) fn provider_generation_export_commitment(
806    export: &ProviderGenerationExport,
807) -> Result<Digest, IdentityError> {
808    let checkpoint_bundles = export
809        .checkpoint_bundles
810        .iter()
811        .map(|bundle| {
812            let checkpoint = bundle.verified_checkpoint();
813            ProviderCheckpointBundleCommitmentWire {
814                genesis: bundle.genesis(),
815                prior_checkpoint_id: bundle.prior_checkpoint_id(),
816                events: bundle.events(),
817                checkpoint: checkpoint.checkpoint(),
818                transition_event: checkpoint.transition_event(),
819            }
820        })
821        .collect();
822    provider_commitment(
823        PROVIDER_GENERATION_EXPORT_COMMITMENT_DOMAIN,
824        &ProviderGenerationExportCommitmentWire {
825            format_version: 1,
826            provider: &export.provider,
827            log_id: export.log_id,
828            key_version: export.key_version,
829            entries: &export.entries,
830            leaf_hashes: &export.leaf_hashes,
831            latest_head: export.latest_head.as_ref(),
832            receipts: &export.receipts,
833            checkpoint_bundles,
834            // A newly built manifest commits the exact pre-manifest export, avoiding
835            // self-reference. Later exports bind every already-durable manifest.
836            compaction_manifests: &export.compaction_manifests,
837        },
838    )
839}

841fn inventory_commitment(inventory: &ProviderRetentionInventory) -> Result<Digest, IdentityError> {
842    provider_commitment(
843        PROVIDER_RETENTION_INVENTORY_COMMITMENT_DOMAIN,
844        &ProviderRetentionInventoryCommitmentWire {
845            format_version: 1,
846            tree_size: inventory.tree_size,
847            items: inventory
848                .items
849                .iter()
850                .map(|item| ProviderRetentionItemCommitmentWire {
851                    leaf_index: item.leaf_index,
852                    class_code: item.class.code(),
853                })
854                .collect(),
855            audit_artifact_commitments: inventory
856                .audit_artifacts
857                .iter()
858                .map(ProviderAuditArtifact::commitment)
859                .collect::<Result<Vec<_>, _>>()?,
860        },
861    )
862}
