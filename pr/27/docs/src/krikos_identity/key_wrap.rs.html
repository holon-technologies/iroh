41fn validate_distribution_lifecycle(state: &AccountState) -> Result<(), IdentityError> {
42    match state.lifecycle() {
43        ProjectionLifecycle::Active
44        | ProjectionLifecycle::MigrationPending
45        | ProjectionLifecycle::MigrationDual => Ok(()),
46        ProjectionLifecycle::RecoveryPending => Err(IdentityError::RecoveryPending),
47        ProjectionLifecycle::Forked => Err(IdentityError::AccountForked),
48        ProjectionLifecycle::UpgradePending => Err(IdentityError::ProtocolUpgradeReadOnly),
49        ProjectionLifecycle::Retired => Err(IdentityError::AccountRetired),
50    }
51}

116/// Validated application-group membership at one authoritative account revision.
117///
118/// Construction derives every device authorization and lifecycle from [`AccountState`].
119/// The caller supplies the complete application-defined group membership as device IDs;
120/// the resulting private recipient records cannot be forged or lifecycle-labelled later.
121#[derive(Debug, Clone, PartialEq, Eq)]
122pub struct GroupKeyDistributionSnapshot {
123    crypto_suite_id: CryptoSuiteId,
124    account_id: AccountId,
125    application_id: ApplicationId,
126    group_id: GroupId,
127    authorizing_account_epoch: Epoch,
128    group_key_epoch: GroupKeyEpoch,
129    account_revision: AccountRevision,
130    recipients: Vec<DistributionRecipient>,
131}
132
133impl GroupKeyDistributionSnapshot {
134    /// Validate and capture a complete application-group membership from account post-state.
135    ///
136    /// `expected_recipient_ids` is the application's complete membership decision for this
137    /// exact group and revision. Every named device must exist, be active, and already be
138    /// authorized at the state's exact epoch. Input order is not significant. Active,
139    /// migration-pending, and migration-dual projections are eligible; signature-suite
140    /// migration leaves the fixed v1 key-wrap KEM/KDF/AEAD profile unchanged. Recovery,
141    /// forked, upgraded/read-only, and retired projections fail before recipient processing.
142    pub fn from_post_state(
143        state: &AccountState,
144        application_id: ApplicationId,
145        group_id: GroupId,
146        group_key_epoch: GroupKeyEpoch,
147        mut expected_recipient_ids: Vec<DeviceId>,
148    ) -> Result<Self, IdentityError> {
149        // Candidate and dual migration states intentionally remain eligible: the fixed
150        // v1 X25519/BLAKE3/XChaCha20-Poly1305 wrap suite does not use controller
151        // signature keys and is unchanged throughout the signature-suite migration.
152        validate_distribution_lifecycle(state)?;
153        if expected_recipient_ids.is_empty() {
154            return Err(IdentityError::EmptyCollection {
155                resource: "group key distribution membership",
156            });
157        }
158        if expected_recipient_ids.len() > MAX_DEVICES {
159            return Err(IdentityError::limit(
160                "group key distribution membership",
161                expected_recipient_ids.len(),
162                MAX_DEVICES,
163            ));
164        }
165        expected_recipient_ids.sort_unstable();
166        if expected_recipient_ids
167            .windows(2)
168            .any(|pair| pair[0] == pair[1])
169        {
170            return Err(IdentityError::DuplicateElement {
171                resource: "group key distribution membership",
172            });
173        }
174
175        let mut recipients = Vec::with_capacity(expected_recipient_ids.len());
176        for device_id in expected_recipient_ids {
177            let device = state
178                .devices()
179                .binary_search_by_key(&device_id, |candidate| candidate.id())
180                .ok()
181                .map(|index| &state.devices()[index])
182                .ok_or(IdentityError::DeviceNotAuthorized)?;
183            match device.lifecycle() {
184                ProjectedDeviceLifecycle::Active => {}
185                ProjectedDeviceLifecycle::Suspended => {
186                    return Err(IdentityError::DeviceSuspended);
187                }
188                ProjectedDeviceLifecycle::Revoked => return Err(IdentityError::DeviceRevoked),
189            }
190            if device.authorization_epoch() > state.epoch() {
191                return Err(IdentityError::InvalidEpoch);
192            }
193            recipients.push(DistributionRecipient {
194                device_id,
195                agreement_public_key: device.descriptor().agreement_key(),
196                authorization_epoch: device.authorization_epoch(),
197            });
198        }
199
200        Ok(Self {
201            crypto_suite_id: v1_crypto_suite_id()?,
202            account_id: state.account_id(),
203            application_id,
204            group_id,
205            authorizing_account_epoch: state.epoch(),
206            group_key_epoch,
207            account_revision: state.revision_token(),
208            recipients,
209        })
210    }

215    }

234        self.authorizing_account_epoch
235    }

237    /// New application group-key epoch.
238    pub const fn group_key_epoch(&self) -> GroupKeyEpoch {
239        self.group_key_epoch
240    }
241
242    /// Complete authoritative account revision, including the exact sorted head set.
243    pub const fn account_revision(&self) -> &AccountRevision {
244        &self.account_revision
245    }
246
247    /// Complete sorted application-group membership captured by this snapshot.
248    pub fn expected_recipient_ids(&self) -> impl ExactSizeIterator<Item = DeviceId> + '_ {
249        self.recipients.iter().map(|recipient| recipient.device_id)
250    }
251
252    fn recipient(&self, device_id: DeviceId) -> Result<&DistributionRecipient, IdentityError> {
253        self.recipients
254            .binary_search_by_key(&device_id, |recipient| recipient.device_id)
255            .ok()
256            .map(|index| &self.recipients[index])
257            .ok_or(IdentityError::DeviceNotAuthorized)
258    }
259
260    fn validate_header(&self, header: &GroupKeyWrapHeader) -> Result<(), IdentityError> {
261        validate_v1_crypto_suite(self.crypto_suite_id)?;
262        if header.crypto_suite_id != self.crypto_suite_id {
263            return Err(IdentityError::UnsupportedKeyWrapSuite);
264        }
265        if header.account_id != self.account_id {
266            return Err(IdentityError::AccountMismatch);
267        }
268        if header.authorizing_account_epoch != self.authorizing_account_epoch {
269            return Err(IdentityError::InvalidEpoch);
270        }
271        if header.application_id != self.application_id
272            || header.group_id != self.group_id
273            || header.group_key_epoch != self.group_key_epoch
274        {
275            return Err(IdentityError::InvalidRelationship {
276                resource: "group key wrap distribution context",
277            });
278        }
279        Ok(())
280    }
281
282    fn validate_output(&self, wraps: &RecipientKeyWraps) -> Result<(), IdentityError> {
283        if wraps.as_slice().len() != self.recipients.len()
284            || !wraps
285                .as_slice()
286                .iter()
287                .map(WrappedGroupKey::recipient_device_id)
288                .eq(self.expected_recipient_ids())
289        {
290            return Err(IdentityError::InvalidRelationship {
291                resource: "group key rotation snapshot output",
292            });
293        }
294        for wrap in wraps.as_slice() {
295            self.validate_header(wrap.header())?;
296            let recipient = self.recipient(wrap.recipient_device_id())?;
297            wrap.header()
298                .validate_recipient_binding(recipient.device_id, recipient.agreement_public_key)?;
299        }
300        Ok(())
301    }
302}

304/// Revision-bound local result of one complete application group-key rotation.
305///
306/// This artifact is intentionally not a canonical wire type. Persistence must accept
307/// the complete artifact, revalidate its account revision immediately before an atomic
308/// compare-and-swap, and persist its recipient wraps only as part of that transaction.
309/// No constructor or consuming accessor exposes a bare, unbound rotation result.
310#[derive(Debug)]
311pub struct GroupKeyRotation {
312    crypto_suite_id: CryptoSuiteId,
313    account_revision: AccountRevision,
314    account_id: AccountId,
315    application_id: ApplicationId,
316    group_id: GroupId,
317    authorizing_account_epoch: Epoch,
318    group_key_epoch: GroupKeyEpoch,
319    expected_recipient_ids: Vec<DeviceId>,
320    recipient_key_wraps: RecipientKeyWraps,
321}
322
323impl GroupKeyRotation {
324    fn from_snapshot(
325        snapshot: &GroupKeyDistributionSnapshot,
326        recipient_key_wraps: RecipientKeyWraps,
327    ) -> Result<Self, IdentityError> {
328        snapshot.validate_output(&recipient_key_wraps)?;
329        if snapshot.account_revision.account_id() != snapshot.account_id {
330            return Err(IdentityError::StorageCorruption);
331        }
332        Ok(Self {
333            crypto_suite_id: snapshot.crypto_suite_id,
334            account_revision: snapshot.account_revision.clone(),
335            account_id: snapshot.account_id,
336            application_id: snapshot.application_id,
337            group_id: snapshot.group_id,
338            authorizing_account_epoch: snapshot.authorizing_account_epoch,
339            group_key_epoch: snapshot.group_key_epoch,
340            expected_recipient_ids: snapshot.expected_recipient_ids().collect(),
341            recipient_key_wraps,
342        })
343    }

390    /// Verify that persistence still targets the exact authoritative account revision.
391    ///
392    /// A fork is reported distinctly so callers cannot treat an unresolved multi-head
393    /// projection as an ordinary compare-and-swap race.
394    pub fn validate_current_revision(&self, state: &AccountState) -> Result<(), IdentityError> {
395        if state.account_id() != self.account_id {
396            return Err(IdentityError::AccountMismatch);
397        }
398        validate_distribution_lifecycle(state)?;
399        if state.revision_token() != self.account_revision
400            || state.epoch() != self.authorizing_account_epoch
401        {
402            return Err(IdentityError::StaleRevision);
403        }
404        Ok(())
405    }

518impl GroupKeyWrapHeader {
519    /// Construct a header directly from the current recipient authorization.
520    #[allow(clippy::too_many_arguments)]
521    pub fn new_for_recipient(
522        crypto_suite_id: CryptoSuiteId,
523        account_id: AccountId,
524        application_id: ApplicationId,
525        group_id: GroupId,
526        authorizing_account_epoch: Epoch,
527        group_key_epoch: GroupKeyEpoch,
528        recipient: &DeviceAuthorization,
529        ephemeral_public_key: AgreementPublicKey,
530        nonce: KeyWrapNonce,
531        extensions: Extensions,
532    ) -> Result<Self, IdentityError> {
533        if recipient.authorization_epoch() > authorizing_account_epoch {
534            return Err(IdentityError::InvalidEpoch);
535        }
536        let recipient_key = recipient.descriptor().agreement_key();
537        Self::new_for_binding(
538            crypto_suite_id,
539            account_id,
540            application_id,
541            group_id,
542            authorizing_account_epoch,
543            group_key_epoch,
544            recipient.device_id(),
545            recipient_key,
546            ephemeral_public_key,
547            nonce,
548            extensions,
549        )
550    }
551
552    #[allow(clippy::too_many_arguments)]
553    fn new_for_binding(
554        crypto_suite_id: CryptoSuiteId,
555        account_id: AccountId,
556        application_id: ApplicationId,
557        group_id: GroupId,
558        authorizing_account_epoch: Epoch,
559        group_key_epoch: GroupKeyEpoch,
560        recipient_device_id: DeviceId,
561        recipient_agreement_key: AgreementPublicKey,
562        ephemeral_public_key: AgreementPublicKey,
563        nonce: KeyWrapNonce,
564        extensions: Extensions,
565    ) -> Result<Self, IdentityError> {
566        if ephemeral_public_key == recipient_agreement_key {
567            return Err(IdentityError::InvalidRelationship {
568                resource: "ephemeral and recipient agreement keys",
569            });
570        }
571        let recipient_agreement_key_id =
572            AgreementKeyId::derive(recipient_device_id, recipient_agreement_key)?;
573        Self::from_fields(
574            crypto_suite_id,
575            account_id,
576            application_id,
577            group_id,
578            authorizing_account_epoch,
579            group_key_epoch,
580            recipient_device_id,
581            recipient_agreement_key_id,
582            ephemeral_public_key,
583            nonce,
584            extensions,
585        )
586    }
587
588    #[allow(clippy::too_many_arguments)]
589    fn from_fields(
590        crypto_suite_id: CryptoSuiteId,
591        account_id: AccountId,
592        application_id: ApplicationId,
593        group_id: GroupId,
594        authorizing_account_epoch: Epoch,
595        group_key_epoch: GroupKeyEpoch,
596        recipient_device_id: DeviceId,
597        recipient_agreement_key_id: AgreementKeyId,
598        ephemeral_public_key: AgreementPublicKey,
599        nonce: KeyWrapNonce,
600        extensions: Extensions,
601    ) -> Result<Self, IdentityError> {
602        validate_v1_crypto_suite(crypto_suite_id)?;
603        extensions.validate_critical(&[])?;
604        Ok(Self {
605            protocol_version: ProtocolVersion::V1,
606            crypto_suite_id,
607            account_id,
608            application_id,
609            group_id,
610            authorizing_account_epoch,
611            group_key_epoch,
612            recipient_device_id,
613            recipient_agreement_key_id,
614            ephemeral_public_key,
615            nonce,
616            extensions,
617        })
618    }

676    pub fn validate_recipient(&self, recipient: &DeviceAuthorization) -> Result<(), IdentityError> {
677        self.validate_recipient_binding(
678            recipient.device_id(),
679            recipient.descriptor().agreement_key(),
680        )
681    }
682
683    fn validate_recipient_binding(
684        &self,
685        recipient_device_id: DeviceId,
686        recipient_agreement_key: AgreementPublicKey,
687    ) -> Result<(), IdentityError> {
688        if self.recipient_device_id != recipient_device_id {
689            return Err(IdentityError::InvalidRelationship {
690                resource: "group key wrap recipient device",
691            });
692        }
693        let expected_key_id = AgreementKeyId::derive(recipient_device_id, recipient_agreement_key)?;
694        if self.recipient_agreement_key_id != expected_key_id {
695            return Err(IdentityError::InvalidIdentifier {
696                resource: "recipient agreement key",
697            });
698        }
699        if self.ephemeral_public_key == recipient_agreement_key {
700            return Err(IdentityError::InvalidRelationship {
701                resource: "ephemeral and recipient agreement keys",
702            });
703        }
704        Ok(())
705    }
706
707    fn same_distribution_context(&self, other: &Self) -> bool {
708        self.crypto_suite_id == other.crypto_suite_id
709            && self.account_id == other.account_id
710            && self.application_id == other.application_id
711            && self.group_id == other.group_id
712            && self.authorizing_account_epoch == other.authorizing_account_epoch
713            && self.group_key_epoch == other.group_key_epoch
714            && self.extensions == other.extensions
715    }
716}

775fn derive_wrapping_key(
776    shared_secret: &SharedSecret,
777    ephemeral_public_key: AgreementPublicKey,
778    recipient_public_key: AgreementPublicKey,
779) -> Zeroizing<[u8; GROUP_KEY_BYTES]> {
780    let mut key_material = Zeroizing::new([0_u8; KEY_WRAP_KDF_MATERIAL_BYTES]);
781    key_material[..32].copy_from_slice(shared_secret.as_bytes());
782    key_material[32..64].copy_from_slice(ephemeral_public_key.as_bytes());
783    key_material[64..].copy_from_slice(recipient_public_key.as_bytes());
784    Zeroizing::new(blake3::derive_key(KEY_WRAP_KDF_CONTEXT, &key_material[..]))
785}

788    header: &GroupKeyWrapHeader,
789    outer_extensions: &Extensions,
790) -> Result<Vec<u8>, IdentityError> {
791    encode_wire(&(header, outer_extensions))
792}

794#[cfg(any(feature = "os-rng", test))]
795fn generate_wrap_randomness_with<F, E>(
796    mut fill: F,
797) -> Result<(Zeroizing<[u8; 32]>, [u8; 24]), IdentityError>
798where
799    F: FnMut(&mut [u8]) -> Result<(), E>,
800{
801    let mut ephemeral_secret = Zeroizing::new([0_u8; 32]);
802    fill(&mut ephemeral_secret[..]).map_err(|_| IdentityError::EntropyUnavailable)?;
803    let mut nonce = [0_u8; 24];
804    fill(&mut nonce).map_err(|_| IdentityError::EntropyUnavailable)?;
805    Ok((ephemeral_secret, nonce))
806}

817fn wrap_group_key_with_material(
818    snapshot: &GroupKeyDistributionSnapshot,
819    group_key: &GroupKey,
820    recipient: &DistributionRecipient,
821    ephemeral_secret_bytes: Zeroizing<[u8; 32]>,
822    nonce_bytes: [u8; 24],
823) -> Result<WrappedGroupKey, IdentityError> {
824    validate_v1_crypto_suite(snapshot.crypto_suite_id)?;
825    if recipient.authorization_epoch > snapshot.authorizing_account_epoch {
826        return Err(IdentityError::InvalidEpoch);
827    }
828    let ephemeral_secret = StaticSecret::from(*ephemeral_secret_bytes);
829    let ephemeral_public = PublicKey::from(&ephemeral_secret);
830    let ephemeral_public_key = AgreementPublicKey::x25519(ephemeral_public.to_bytes())?;
831    let recipient_public_key = recipient.agreement_public_key;
832    let recipient_public = PublicKey::from(*recipient_public_key.as_bytes());
833    let shared_secret = ephemeral_secret.diffie_hellman(&recipient_public);
834    validate_contributory(&shared_secret)?;
835    let wrapping_key =
836        derive_wrapping_key(&shared_secret, ephemeral_public_key, recipient_public_key);
837    let nonce = KeyWrapNonce::new(nonce_bytes);
838    let header = GroupKeyWrapHeader::new_for_binding(
839        snapshot.crypto_suite_id,
840        snapshot.account_id,
841        snapshot.application_id,
842        snapshot.group_id,
843        snapshot.authorizing_account_epoch,
844        snapshot.group_key_epoch,
845        recipient.device_id,
846        recipient_public_key,
847        ephemeral_public_key,
848        nonce,
849        Extensions::default(),
850    )?;
851    let outer_extensions = Extensions::default();
852    let associated_data = wrap_associated_data(&header, &outer_extensions)?;
853    let cipher = XChaCha20Poly1305::new(&Key::from(*wrapping_key));
854    let ciphertext = cipher
855        .encrypt(
856            &XNonce::from(nonce_bytes),
857            Payload {
858                msg: group_key.as_bytes(),
859                aad: &associated_data,
860            },
861        )
862        .map_err(|_| IdentityError::KeyWrapAuthenticationFailed)?;
863    WrappedGroupKey::new(header, ciphertext, outer_extensions)
864}

866/// Authenticate and unwrap one exact 32-byte application group key.
867///
868/// Wrong secrets, modified ciphertext, and modified associated data all return
869/// the same non-oracular authentication error after public context validation.
870pub fn unwrap_group_key(
871    snapshot: &GroupKeyDistributionSnapshot,
872    wrapped: &WrappedGroupKey,
873    recipient_secret: &AgreementSecretKey,
874) -> Result<GroupKey, IdentityError> {
875    snapshot.validate_header(wrapped.header())?;
876    let recipient = snapshot.recipient(wrapped.recipient_device_id())?;
877    wrapped
878        .header()
879        .validate_recipient_binding(recipient.device_id, recipient.agreement_public_key)?;
880
881    let ephemeral_public_key = wrapped.header().ephemeral_public_key();
882    let ephemeral_public = PublicKey::from(*ephemeral_public_key.as_bytes());
883    let shared_secret = recipient_secret.0.diffie_hellman(&ephemeral_public);
884    validate_contributory(&shared_secret)?;
885    let recipient_public_key = recipient.agreement_public_key;
886    let wrapping_key =
887        derive_wrapping_key(&shared_secret, ephemeral_public_key, recipient_public_key);
888    let associated_data = wrap_associated_data(wrapped.header(), wrapped.extensions())?;
889    let cipher = XChaCha20Poly1305::new(&Key::from(*wrapping_key));
890    let plaintext = cipher
891        .decrypt(
892            &XNonce::from(*wrapped.header().nonce().as_bytes()),
893            Payload {
894                msg: wrapped.ciphertext(),
895                aad: &associated_data,
896            },
897        )
898        .map(Zeroizing::new)
899        .map_err(|_| IdentityError::KeyWrapAuthenticationFailed)?;
900    let group_key_bytes: [u8; GROUP_KEY_BYTES] = plaintext
901        .as_slice()
902        .try_into()
903        .map_err(|_| IdentityError::KeyWrapAuthenticationFailed)?;
904    Ok(GroupKey::new(group_key_bytes))
905}

907/// Produce a revision-bound complete rotation using an explicit cryptographic RNG.
908///
909/// All post-state recipient records are validated before randomness is consumed.
910/// Each recipient then receives an independent ephemeral secret and nonce, and
911/// the private artifact constructor rejects reuse or incomplete recipient coverage.
912pub fn rotate_group_key_with_rng<R: CryptoRng + ?Sized>(
913    snapshot: &GroupKeyDistributionSnapshot,
914    group_key: &GroupKey,
915    rng: &mut R,
916) -> Result<GroupKeyRotation, IdentityError> {
917    let mut wraps = Vec::with_capacity(snapshot.recipients.len());
918    for recipient in &snapshot.recipients {
919        let mut ephemeral_secret = Zeroizing::new([0_u8; 32]);
920        rng.fill_bytes(&mut ephemeral_secret[..]);
921        let mut nonce = [0_u8; 24];
922        rng.fill_bytes(&mut nonce);
923        wraps.push(wrap_group_key_with_material(
924            snapshot,
925            group_key,
926            recipient,
927            ephemeral_secret,
928            nonce,
929        )?);
930    }
931    let wraps = RecipientKeyWraps::new(wraps)?;
932    GroupKeyRotation::from_snapshot(snapshot, wraps)
933}
934
935/// Produce a revision-bound complete rotation using operating-system entropy.
936///
937/// Entropy failure aborts the complete operation without returning a partial artifact.
938#[cfg(feature = "os-rng")]
939#[cfg_attr(krikos_docsrs, doc(cfg(feature = "os-rng")))]
940pub fn rotate_group_key(
941    snapshot: &GroupKeyDistributionSnapshot,
942    group_key: &GroupKey,
943) -> Result<GroupKeyRotation, IdentityError> {
944    let mut wraps = Vec::with_capacity(snapshot.recipients.len());
945    for recipient in &snapshot.recipients {
946        let (ephemeral_secret, nonce) = generate_wrap_randomness_with(getrandom::fill)?;
947        wraps.push(wrap_group_key_with_material(
948            snapshot,
949            group_key,
950            recipient,
951            ephemeral_secret,
952            nonce,
953        )?);
954    }
955    let wraps = RecipientKeyWraps::new(wraps)?;
956    GroupKeyRotation::from_snapshot(snapshot, wraps)
957}

959/// One bounded ciphertext carrying a group key to the recipient named by its header.

967impl WrappedGroupKey {
968    /// Construct one recipient wrap and enforce both ciphertext and complete-envelope bounds.
969    pub fn new(
970        header: GroupKeyWrapHeader,
971        ciphertext: Vec<u8>,
972        extensions: Extensions,
973    ) -> Result<Self, IdentityError> {
974        if ciphertext.len() != KEY_WRAP_CIPHERTEXT_BYTES {
975            return Err(IdentityError::InvalidRelationship {
976                resource: "group key wrap ciphertext length",
977            });
978        }
979        let ciphertext = BoundedBytes::new("group key wrap ciphertext bytes", ciphertext)?;
980        extensions.validate_critical(&[])?;
981        let wrapped = Self {
982            header,
983            ciphertext,
984            extensions,
985        };
986        let encoded_len = encode_wire(&wrapped)?.len();
987        if encoded_len > MAX_KEY_WRAP_BYTES {
988            return Err(IdentityError::limit(
989                "wrapped group key bytes",
990                encoded_len,
991                MAX_KEY_WRAP_BYTES,
992            ));
993        }
994        Ok(wrapped)
995    }

1000    }

1004        self.header.recipient_device_id
1005    }
1006

1009        self.ciphertext.as_slice()
1010    }

1013    pub const fn extensions(&self) -> &Extensions {

1015    }

1054/// Canonically sorted, duplicate-free recipient wraps for one group-key distribution.
1055#[derive(Debug, Clone, PartialEq, Eq, Serialize)]
1056pub struct RecipientKeyWraps(BoundedVec<WrappedGroupKey, MAX_DEVICES>);
1057
1058impl RecipientKeyWraps {
1059    /// Construct a nonempty set sorted by recipient `DeviceId`.
1060    pub fn new(mut wraps: Vec<WrappedGroupKey>) -> Result<Self, IdentityError> {
1061        if wraps.len() > MAX_DEVICES {
1062            return Err(IdentityError::limit(
1063                "recipient key wraps",
1064                wraps.len(),
1065                MAX_DEVICES,
1066            ));
1067        }
1068        wraps.sort_unstable_by_key(WrappedGroupKey::recipient_device_id);
1069        Self::from_fields(wraps)
1070    }
1071
1072    fn from_fields(wraps: Vec<WrappedGroupKey>) -> Result<Self, IdentityError> {
1073        if wraps.is_empty() {
1074            return Err(IdentityError::EmptyCollection {
1075                resource: "recipient key wraps",
1076            });
1077        }
1078        let wraps = BoundedVec::new("recipient key wraps", wraps)?;
1079        let first = &wraps.as_slice()[0];
1080        for pair in wraps.as_slice().windows(2) {
1081            if pair[0].recipient_device_id() == pair[1].recipient_device_id() {
1082                return Err(IdentityError::DuplicateElement {
1083                    resource: "recipient key wrap devices",
1084                });
1085            }
1086            if pair[0].recipient_device_id() > pair[1].recipient_device_id() {
1087                return Err(IdentityError::NonCanonical);
1088            }
1089        }
1090        for wrap in wraps.as_slice() {
1091            if !first.header.same_distribution_context(&wrap.header) {
1092                return Err(IdentityError::InvalidRelationship {
1093                    resource: "recipient key wrap distribution context",
1094                });
1095            }
1096        }
1097        for (index, wrap) in wraps.as_slice().iter().enumerate() {
1098            for other in &wraps.as_slice()[index.saturating_add(1)..] {
1099                if wrap.header.ephemeral_public_key == other.header.ephemeral_public_key {
1100                    return Err(IdentityError::DuplicateElement {
1101                        resource: "recipient key wrap ephemeral public keys",
1102                    });
1103                }
1104                if wrap.header.nonce == other.header.nonce {
1105                    return Err(IdentityError::DuplicateElement {
1106                        resource: "recipient key wrap nonces",
1107                    });
1108                }
1109            }
1110        }
1111
1112        let set = Self(wraps);
1113        let encoded_len = encode_wire(&set)?.len();
1114        if encoded_len > MAX_ENCODED_OBJECT_BYTES {
1115            return Err(IdentityError::limit(
1116                "recipient key wrap set bytes",
1117                encoded_len,
1118                MAX_ENCODED_OBJECT_BYTES,
1119            ));
1120        }
1121        Ok(set)
1122    }
1123
1124    /// Borrow wraps in canonical recipient order.
1125    pub fn as_slice(&self) -> &[WrappedGroupKey] {
1126        self.0.as_slice()
1127    }
1128}

1134    {

1136        Self::from_fields(wraps.into_vec()).map_err(de::Error::custom)
