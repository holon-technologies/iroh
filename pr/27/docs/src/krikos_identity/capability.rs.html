37macro_rules! bounded_capability_name {
38    ($name:ident, $resource:literal, $doc:literal) => {
39        #[doc = $doc]
40        #[derive(Debug, Clone, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize)]
41        pub struct $name(Box<str>);
42
43        impl $name {
44            /// Construct a nonempty, byte-for-byte v1 name.
45            pub fn new(value: impl AsRef<str>) -> Result<Self, IdentityError> {
46                let value = value.as_ref();
47                if value.is_empty() {
48                    return Err(IdentityError::EmptyCollection {
49                        resource: $resource,
50                    });
51                }
52                if value.len() > MAX_CAPABILITY_NAME_BYTES {
53                    return Err(IdentityError::limit(
54                        $resource,
55                        value.len(),
56                        MAX_CAPABILITY_NAME_BYTES,
57                    ));
58                }
59                Ok(Self(Box::from(value)))
60            }
61
62            /// Borrow the exact UTF-8 name bytes as text.
63            pub fn as_str(&self) -> &str {
64                &self.0
65            }
66        }
67
68        impl<'de> Deserialize<'de> for $name {
69            fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
70            where
71                D: Deserializer<'de>,
72            {
73                struct Visitor;
74
75                impl<'de> de::Visitor<'de> for Visitor {
76                    type Value = $name;
77
78                    fn expecting(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
79                        write!(
80                            formatter,
81                            "a nonempty UTF-8 name of at most {MAX_CAPABILITY_NAME_BYTES} bytes"
82                        )
83                    }
84
85                    fn visit_borrowed_str<E>(self, value: &'de str) -> Result<Self::Value, E>
86                    where
87                        E: de::Error,
88                    {
89                        $name::new(value).map_err(E::custom)
90                    }
91
92                    fn visit_str<E>(self, value: &str) -> Result<Self::Value, E>
93                    where
94                        E: de::Error,
95                    {
96                        $name::new(value).map_err(E::custom)
97                    }
98                }
99
100                deserializer.deserialize_str(Visitor)
101            }
102        }
103
104        canonical_schema!($name, $resource);
105    };
106}

155/// A semantic-order sequence of nonempty opaque resource path segments.
156#[derive(Debug, Clone, PartialEq, Eq, Serialize)]
157pub struct ResourcePath(Vec<ResourceSegment>);
158
159impl ResourcePath {
160    /// Construct a path from semantic-order opaque segment bytes.
161    pub fn new(segments: Vec<Vec<u8>>) -> Result<Self, IdentityError> {
162        if segments.is_empty() {
163            return Err(IdentityError::EmptyCollection {
164                resource: "resource path segments",
165            });
166        }
167        if segments.len() > MAX_RESOURCE_PATH_SEGMENTS {
168            return Err(IdentityError::limit(
169                "resource path segments",
170                segments.len(),
171                MAX_RESOURCE_PATH_SEGMENTS,
172            ));
173        }
174        let segments = segments
175            .into_iter()
176            .map(ResourceSegment::new)
177            .collect::<Result<Vec<_>, _>>()?;
178        Self::from_segments(segments)
179    }
180
181    fn from_segments(segments: Vec<ResourceSegment>) -> Result<Self, IdentityError> {
182        if segments.is_empty() {
183            return Err(IdentityError::EmptyCollection {
184                resource: "resource path segments",
185            });
186        }
187        if segments.len() > MAX_RESOURCE_PATH_SEGMENTS {
188            return Err(IdentityError::limit(
189                "resource path segments",
190                segments.len(),
191                MAX_RESOURCE_PATH_SEGMENTS,
192            ));
193        }
194        let path = Self(segments);
195        path.validate_selector_size()?;
196        Ok(path)
197    }
198
199    /// Borrow path segments in their semantic order.
200    pub fn segments(&self) -> &[ResourceSegment] {
201        &self.0
202    }
203
204    fn validate_selector_size(&self) -> Result<(), IdentityError> {
205        let mut encoded_len = 1usize.checked_add(varint_len(self.0.len())).ok_or(
206            IdentityError::ArithmeticOverflow {
207                resource: "resource selector bytes",
208            },
209        )?;
210        for segment in &self.0 {
211            encoded_len = encoded_len
212                .checked_add(varint_len(segment.as_bytes().len()))
213                .and_then(|length| length.checked_add(segment.as_bytes().len()))
214                .ok_or(IdentityError::ArithmeticOverflow {
215                    resource: "resource selector bytes",
216                })?;
217            if encoded_len > MAX_RESOURCE_SELECTOR_BYTES {
218                return Err(IdentityError::limit(
219                    "resource selector bytes",
220                    encoded_len,
221                    MAX_RESOURCE_SELECTOR_BYTES,
222                ));
223            }
224        }
225        Ok(())
226    }
227
228    fn starts_with(&self, prefix: &Self) -> bool {
229        self.0.len() >= prefix.0.len()
230            && self
231                .0
232                .iter()
233                .zip(&prefix.0)
234                .all(|(segment, prefix_segment)| segment == prefix_segment)
235    }
236}

239    fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
240    where

243        let segments =
244            BoundedVec::<ResourceSegment, MAX_RESOURCE_PATH_SEGMENTS>::deserialize(deserializer)?;

251const fn varint_len(mut value: usize) -> usize {
252    let mut length = 1;
253    while value >= 128 {
254        value >>= 7;
255        length += 1;
256    }
257    length
258}

260/// A closed v1 exact or complete-segment-prefix resource selector.
261#[derive(Debug, Clone, PartialEq, Eq)]
262pub enum ResourceSelector {
263    /// Select exactly one resource path.
264    Exact(ResourcePath),
265    /// Select every resource path beginning with these complete segments.
266    Prefix(ResourcePath),
267}
268
269impl ResourceSelector {
270    /// Construct an exact selector after rechecking its encoded size.
271    pub fn exact(path: ResourcePath) -> Result<Self, IdentityError> {
272        path.validate_selector_size()?;
273        Ok(Self::Exact(path))
274    }
275
276    /// Construct a complete-segment-prefix selector after rechecking its encoded size.
277    pub fn prefix(path: ResourcePath) -> Result<Self, IdentityError> {
278        path.validate_selector_size()?;
279        Ok(Self::Prefix(path))
280    }
281
282    /// Stable closed v1 selector codepoint.
283    pub const fn code(&self) -> u16 {
284        match self {
285            Self::Exact(_) => 1,
286            Self::Prefix(_) => 2,
287        }
288    }
289
290    /// Borrow the selected path or prefix.
291    pub const fn path(&self) -> &ResourcePath {
292        match self {
293            Self::Exact(path) | Self::Prefix(path) => path,
294        }
295    }
296}

299    fn serialize<S>(&self, serializer: S) -> Result<S::Ok, S::Error>
300    where
301        S: Serializer,
302    {
303        (self.code(), self.path()).serialize(serializer)
304    }
305}

309    where

312        let (code, path) = <(u16, ResourcePath)>::deserialize(deserializer)?;

317                registry: "resource selector",

319            })),
320        }
321    }

337/// A closed v1 conjunctive capability constraint.
338#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
339pub enum CapabilityConstraint {
340    /// Require an authorization account epoch at least this value.
341    AccountEpochAtLeast(Epoch),
342    /// Require an authorization account epoch at most this value.
343    AccountEpochAtMost(Epoch),
344    /// Require use no earlier than this explicit timestamp.
345    ValidFrom(Timestamp),
346}
347
348impl CapabilityConstraint {
349    /// Stable closed v1 constraint codepoint.
350    pub const fn code(self) -> u16 {
351        match self {
352            Self::AccountEpochAtLeast(_) => 1,
353            Self::AccountEpochAtMost(_) => 2,
354            Self::ValidFrom(_) => 3,
355        }
356    }
357
358    const fn value(self) -> u64 {
359        match self {
360            Self::AccountEpochAtLeast(epoch) | Self::AccountEpochAtMost(epoch) => epoch.get(),
361            Self::ValidFrom(timestamp) => timestamp.as_unix_millis(),
362        }
363    }
364}
365
366impl Serialize for CapabilityConstraint {
367    fn serialize<S>(&self, serializer: S) -> Result<S::Ok, S::Error>
368    where
369        S: Serializer,
370    {
371        (self.code(), self.value()).serialize(serializer)
372    }
373}
374
375impl<'de> Deserialize<'de> for CapabilityConstraint {
376    fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
377    where
378        D: Deserializer<'de>,
379    {
380        let (code, value) = <(u16, u64)>::deserialize(deserializer)?;
381        match code {
382            1 => Ok(Self::AccountEpochAtLeast(Epoch::new(value))),
383            2 => Ok(Self::AccountEpochAtMost(Epoch::new(value))),
384            3 => Ok(Self::ValidFrom(Timestamp::from_unix_millis(value))),
385            unsupported => Err(de::Error::custom(IdentityError::UnsupportedCodepoint {
386                registry: "capability constraint",
387                code: unsupported,
388            })),
389        }
390    }
391}

395/// Whether and how far a capability grant may be delegated.
396#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
397pub enum DelegationPermission {
398    /// This grant cannot be delegated further.
399    NotDelegable,
400    /// This grant can be delegated with bounded remaining depth.
401    Delegable {
402        /// Maximum number of remaining delegation links.
403        remaining: DelegationDepth,
404    },
405}
406
407impl DelegationPermission {
408    /// Construct a permission with a previously validated remaining depth.
409    pub const fn delegable(remaining: DelegationDepth) -> Self {
410        Self::Delegable { remaining }
411    }
412
413    /// Stable closed v1 delegation-permission codepoint.
414    pub const fn code(self) -> u16 {
415        match self {
416            Self::NotDelegable => 1,
417            Self::Delegable { .. } => 2,
418        }
419    }
420
421    /// Return the remaining depth, or `None` when delegation is forbidden.
422    pub const fn remaining(self) -> Option<DelegationDepth> {
423        match self {
424            Self::NotDelegable => None,
425            Self::Delegable { remaining } => Some(remaining),
426        }
427    }
428}
429
430impl Serialize for DelegationPermission {
431    fn serialize<S>(&self, serializer: S) -> Result<S::Ok, S::Error>
432    where
433        S: Serializer,
434    {
435        let remaining = self.remaining().map_or(0, DelegationDepth::get);
436        (self.code(), remaining).serialize(serializer)
437    }
438}
439
440impl<'de> Deserialize<'de> for DelegationPermission {
441    fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
442    where
443        D: Deserializer<'de>,
444    {
445        let (code, remaining) = <(u16, u8)>::deserialize(deserializer)?;
446        match (code, remaining) {
447            (1, 0) => Ok(Self::NotDelegable),
448            (1, _) => Err(de::Error::custom(IdentityError::InvalidCapability {
449                reason: "non-delegable permission has nonzero remaining depth",
450            })),
451            (2, remaining) => DelegationDepth::new(remaining)
452                .map(Self::delegable)
453                .map_err(de::Error::custom),
454            (unsupported, _) => Err(de::Error::custom(IdentityError::UnsupportedCodepoint {
455                registry: "delegation permission",
456                code: unsupported,
457            })),
458        }
459    }
460}

464/// A canonical, bounded v1 capability grant.
465#[derive(Debug, Clone, PartialEq, Eq, Serialize)]
466pub struct CapabilityGrant {
467    protocol_version: ProtocolVersion,
468    namespace: CapabilityNamespace,
469    action: CapabilityAction,
470    resource: ResourceSelector,
471    constraints: Vec<CapabilityConstraint>,
472    delegation: DelegationPermission,
473    expires_at: Option<Timestamp>,
474    extensions: Extensions,
475}
476
477impl CapabilityGrant {
478    /// Validate, canonically sort, and construct a v1 capability grant.
479    #[allow(clippy::too_many_arguments)]
480    pub fn new(
481        namespace: CapabilityNamespace,
482        action: CapabilityAction,
483        resource: ResourceSelector,
484        mut constraints: Vec<CapabilityConstraint>,
485        delegation: DelegationPermission,
486        expires_at: Option<Timestamp>,
487        extensions: Extensions,
488    ) -> Result<Self, IdentityError> {
489        if constraints.len() > MAX_CONSTRAINTS_PER_CAPABILITY {
490            return Err(IdentityError::limit(
491                "capability constraints",
492                constraints.len(),
493                MAX_CONSTRAINTS_PER_CAPABILITY,
494            ));
495        }
496        constraints.sort_unstable_by_key(|constraint| constraint.code());
497        Self::from_sorted(
498            namespace,
499            action,
500            resource,
501            constraints,
502            delegation,
503            expires_at,
504            extensions,
505        )
506    }
507
508    #[allow(clippy::too_many_arguments)]
509    fn from_sorted(
510        namespace: CapabilityNamespace,
511        action: CapabilityAction,
512        resource: ResourceSelector,
513        constraints: Vec<CapabilityConstraint>,
514        delegation: DelegationPermission,
515        expires_at: Option<Timestamp>,
516        extensions: Extensions,
517    ) -> Result<Self, IdentityError> {
518        if constraints.len() > MAX_CONSTRAINTS_PER_CAPABILITY {
519            return Err(IdentityError::limit(
520                "capability constraints",
521                constraints.len(),
522                MAX_CONSTRAINTS_PER_CAPABILITY,
523            ));
524        }
525        for pair in constraints.windows(2) {
526            if pair[0].code() == pair[1].code() {
527                return Err(IdentityError::DuplicateElement {
528                    resource: "capability constraints",
529                });
530            }
531            if pair[0].code() > pair[1].code() {
532                return Err(IdentityError::NonCanonical);
533            }
534        }
535
536        let minimum_epoch = constraints.iter().find_map(|constraint| match constraint {
537            CapabilityConstraint::AccountEpochAtLeast(epoch) => Some(*epoch),
538            _ => None,
539        });
540        let maximum_epoch = constraints.iter().find_map(|constraint| match constraint {
541            CapabilityConstraint::AccountEpochAtMost(epoch) => Some(*epoch),
542            _ => None,
543        });
544        if minimum_epoch
545            .zip(maximum_epoch)
546            .is_some_and(|(minimum, maximum)| minimum > maximum)
547        {
548            return Err(IdentityError::InvalidCapability {
549                reason: "minimum account epoch exceeds maximum account epoch",
550            });
551        }
552        if constraints.iter().any(|constraint| {
553            matches!(
554                (constraint, expires_at),
555                (CapabilityConstraint::ValidFrom(valid_from), Some(expires_at))
556                    if *valid_from > expires_at
557            )
558        }) {
559            return Err(IdentityError::InvalidCapability {
560                reason: "valid-from time exceeds expiration time",
561            });
562        }
563        extensions.validate_critical(&[])?;
564
565        Ok(Self {
566            protocol_version: ProtocolVersion::V1,
567            namespace,
568            action,
569            resource,
570            constraints,
571            delegation,
572            expires_at,
573            extensions,
574        })
575    }
576
577    /// Exact capability namespace.
578    pub const fn namespace(&self) -> &CapabilityNamespace {
579        &self.namespace
580    }
581
582    /// Exact capability action.
583    pub const fn action(&self) -> &CapabilityAction {
584        &self.action
585    }
586
587    /// Resource selector governed by this grant.
588    pub const fn resource(&self) -> &ResourceSelector {
589        &self.resource
590    }
591
592    /// Canonically sorted conjunctive constraints.
593    pub fn constraints(&self) -> &[CapabilityConstraint] {
594        &self.constraints
595    }
596
597    /// Remaining delegation permission.
598    pub const fn delegation(&self) -> DelegationPermission {
599        self.delegation
600    }
601
602    /// Optional exclusive upper time bound represented by the grant.
603    pub const fn expires_at(&self) -> Option<Timestamp> {
604        self.expires_at
605    }
606
607    /// Preserved noncritical extension fields.
608    pub const fn extensions(&self) -> &Extensions {
609        &self.extensions
610    }
611
612    /// Derive the grant identifier from this exact canonical grant body.
613    pub fn capability_grant_id(&self) -> Result<CapabilityGrantId, IdentityError> {
614        CapabilityGrantId::derive(self)
615    }
616}

618impl<'de> Deserialize<'de> for CapabilityGrant {
619    fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
620    where
621        D: Deserializer<'de>,
622    {
623        #[derive(Deserialize)]
624        struct Wire {
625            protocol_version: ProtocolVersion,
626            namespace: CapabilityNamespace,
627            action: CapabilityAction,
628            resource: ResourceSelector,
629            constraints: BoundedVec<CapabilityConstraint, MAX_CONSTRAINTS_PER_CAPABILITY>,
630            delegation: DelegationPermission,
631            expires_at: Option<Timestamp>,

643            wire.action,
644            wire.resource,
645            wire.constraints.into_vec(),
646            wire.delegation,
647            wire.expires_at,
648            wire.extensions,
649        )
650        .map_err(de::Error::custom)
651    }
652}

782/// Canonical signed body for one semantic delegation edge.
783#[derive(Debug, Clone, PartialEq, Eq, Serialize)]
784pub struct DelegationBody {
785    protocol_version: ProtocolVersion,
786    parent_grant_id: CapabilityGrantId,
787    child_grant: CapabilityGrant,
788    issuer: DeviceId,
789    subject: DeviceId,
790    authorization_context: AuthorizationContext,
791    issued_at: Timestamp,
792    nonce: [u8; 16],
793    extensions: Extensions,
794}
795
796impl DelegationBody {
797    /// Construct one v1 delegation edge.
798    #[allow(clippy::too_many_arguments)]
799    pub fn new(
800        parent_grant_id: CapabilityGrantId,
801        child_grant: CapabilityGrant,
802        issuer: DeviceId,
803        subject: DeviceId,
804        authorization_context: AuthorizationContext,
805        issued_at: Timestamp,
806        nonce: [u8; 16],
807        extensions: Extensions,
808    ) -> Result<Self, IdentityError> {
809        if issuer == subject {
810            return Err(IdentityError::InvalidDelegation {
811                reason: "delegation issuer and subject are identical",
812            });
813        }
814        if parent_grant_id == child_grant.capability_grant_id()? {
815            return Err(IdentityError::InvalidDelegation {
816                reason: "delegation child repeats its parent grant",
817            });
818        }
819        extensions.validate_critical(&[])?;
820        Ok(Self {
821            protocol_version: ProtocolVersion::V1,
822            parent_grant_id,
823            child_grant,
824            issuer,
825            subject,
826            authorization_context,
827            issued_at,
828            nonce,
829            extensions,
830        })
831    }
832
833    /// Grant identifier that this edge narrows.
834    pub const fn parent_grant_id(&self) -> CapabilityGrantId {
835        self.parent_grant_id
836    }
837
838    /// Narrowed grant assigned by this edge.
839    pub const fn child_grant(&self) -> &CapabilityGrant {
840        &self.child_grant
841    }
842
843    /// Device signing and issuing this edge.
844    pub const fn issuer(&self) -> DeviceId {
845        self.issuer
846    }
847
848    /// Device receiving the child grant.
849    pub const fn subject(&self) -> DeviceId {
850        self.subject
851    }
852
853    /// Account checkpoint context authorizing issuance.
854    pub const fn authorization_context(&self) -> AuthorizationContext {
855        self.authorization_context
856    }
857
858    /// Explicit issuance time.
859    pub const fn issued_at(&self) -> Timestamp {
860        self.issued_at
861    }
862
863    /// Caller-supplied uniqueness nonce.
864    pub const fn nonce(&self) -> &[u8; 16] {
865        &self.nonce
866    }
867
868    /// Preserved noncritical extension fields.
869    pub const fn extensions(&self) -> &Extensions {
870        &self.extensions
871    }
872
873    /// Derive the delegation identifier from this exact canonical body.
874    pub fn delegation_id(&self) -> Result<DelegationId, IdentityError> {
875        DelegationId::derive(self)
876    }
877}

950/// A bounded, semantic-order, same-account delegation chain.
951#[derive(Debug, Clone, PartialEq, Eq, Serialize)]
952pub struct DelegationChain {
953    root: CapabilityRoot,
954    links: Vec<SignedDelegation>,
955}
956
957impl DelegationChain {
958    /// Validate and construct a semantic-order chain of one to eight links.
959    pub fn new(root: CapabilityRoot, links: Vec<SignedDelegation>) -> Result<Self, IdentityError> {
960        if links.is_empty() {
961            return Err(IdentityError::EmptyCollection {
962                resource: "delegation chain links",
963            });
964        }
965        if links.len() > MAX_DELEGATION_DEPTH {
966            return Err(IdentityError::limit(
967                "delegation chain links",
968                links.len(),
969                MAX_DELEGATION_DEPTH,
970            ));
971        }
972        let chain = Self { root, links };
973        chain.validate()?;
974        Ok(chain)
975    }
976
977    /// Root authority and initial holder.
978    pub const fn root(&self) -> &CapabilityRoot {
979        &self.root
980    }
981
982    /// Delegation edges in parent-to-child semantic order.
983    pub fn links(&self) -> &[SignedDelegation] {
984        &self.links
985    }
986
987    /// Grant assigned by the final delegation edge.
988    pub fn leaf_grant(&self) -> &CapabilityGrant {
989        match self.links.last() {
990            Some(link) => link.body().child_grant(),
991            None => self.root.grant(),
992        }
993    }
994
995    /// Device receiving the final delegation edge.
996    pub fn leaf_holder(&self) -> DeviceId {
997        match self.links.last() {
998            Some(link) => link.body().subject(),
999            None => self.root.holder(),
1000        }
1001    }
1002
1003    fn validate(&self) -> Result<(), IdentityError> {
1004        let root_account = self.root.authorization_context().account_id();
1005        let mut current_holder = self.root.holder();
1006        let mut current_grant = self.root.grant();
1007        let mut current_grant_id = current_grant.capability_grant_id()?;
1008
1009        let mut seen_devices = Vec::with_capacity(self.links.len().saturating_add(1));
1010        let mut seen_grants = Vec::with_capacity(self.links.len().saturating_add(1));
1011        let mut seen_delegations = Vec::with_capacity(self.links.len());
1012        seen_devices.push(current_holder);
1013        seen_grants.push(current_grant_id);
1014
1015        for link in &self.links {
1016            let body = link.body();
1017            if body.authorization_context().account_id() != root_account {
1018                return Err(IdentityError::InvalidDelegation {
1019                    reason: "delegation chain crosses account contexts",
1020                });
1021            }
1022            if body.issuer() != current_holder {
1023                return Err(IdentityError::InvalidDelegation {
1024                    reason: "delegation issuer does not hold the parent grant",
1025                });
1026            }
1027            if body.parent_grant_id() != current_grant_id {
1028                return Err(IdentityError::InvalidDelegation {
1029                    reason: "delegation parent grant is out of semantic order",
1030                });
1031            }
1032            if seen_devices.contains(&body.subject()) {
1033                return Err(IdentityError::InvalidDelegation {
1034                    reason: "delegation chain contains a device cycle",
1035                });
1036            }
1037
1038            let child_grant = body.child_grant();
1039            validate_narrowing(current_grant, child_grant)?;
1040            let child_grant_id = child_grant.capability_grant_id()?;
1041            if seen_grants.contains(&child_grant_id) {
1042                return Err(IdentityError::InvalidDelegation {
1043                    reason: "delegation chain repeats a grant identifier",
1044                });
1045            }
1046            let delegation_id = link.delegation_id()?;
1047            if seen_delegations.contains(&delegation_id) {
1048                return Err(IdentityError::InvalidDelegation {
1049                    reason: "delegation chain repeats a delegation identifier",
1050                });
1051            }
1052
1053            seen_devices.push(body.subject());
1054            seen_grants.push(child_grant_id);
1055            seen_delegations.push(delegation_id);
1056            current_holder = body.subject();
1057            current_grant = child_grant;
1058            current_grant_id = child_grant_id;
1059        }
1060        Ok(())
1061    }
1062}

1066    where
1067        D: Deserializer<'de>,
1068    {
1069        #[derive(Deserialize)]

1073        }

1082fn validate_narrowing(
1083    parent: &CapabilityGrant,
1084    child: &CapabilityGrant,
1085) -> Result<(), IdentityError> {
1086    if parent.namespace() != child.namespace() || parent.action() != child.action() {
1087        return Err(IdentityError::InvalidDelegation {
1088            reason: "delegation changes capability namespace or action",
1089        });
1090    }
1091
1092    let mut strict = validate_resource_narrowing(parent.resource(), child.resource())?;
1093    strict |= validate_constraint_narrowing(parent.constraints(), child.constraints())?;
1094    strict |= validate_expiration_narrowing(parent.expires_at(), child.expires_at())?;
1095    strict |= validate_permission_narrowing(parent.delegation(), child.delegation())?;
1096
1097    if !strict {
1098        return Err(IdentityError::InvalidDelegation {
1099            reason: "delegation child does not strictly narrow its parent",
1100        });
1101    }
1102    Ok(())
1103}
1104
1105fn validate_resource_narrowing(
1106    parent: &ResourceSelector,
1107    child: &ResourceSelector,
1108) -> Result<bool, IdentityError> {
1109    match (parent, child) {
1110        (ResourceSelector::Exact(parent), ResourceSelector::Exact(child)) if parent == child => {
1111            Ok(false)
1112        }
1113        (ResourceSelector::Prefix(parent), ResourceSelector::Prefix(child))
1114            if child.starts_with(parent) =>
1115        {
1116            Ok(parent != child)
1117        }
1118        (ResourceSelector::Prefix(parent), ResourceSelector::Exact(child))
1119            if child.starts_with(parent) =>
1120        {
1121            Ok(true)
1122        }
1123        _ => Err(IdentityError::InvalidDelegation {
1124            reason: "delegation broadens its resource selector",
1125        }),
1126    }
1127}

1129fn validate_constraint_narrowing(
1130    parent: &[CapabilityConstraint],
1131    child: &[CapabilityConstraint],
1132) -> Result<bool, IdentityError> {
1133    let mut strict = false;
1134    for code in 1..=3 {
1135        let parent_value = parent
1136            .iter()
1137            .find(|constraint| constraint.code() == code)
1138            .map(|constraint| constraint.value());
1139        let child_value = child
1140            .iter()
1141            .find(|constraint| constraint.code() == code)
1142            .map(|constraint| constraint.value());
1143
1144        match (code, parent_value, child_value) {
1145            (_, Some(_), None) => {
1146                return Err(IdentityError::InvalidDelegation {
1147                    reason: "delegation removes a parent constraint",
1148                });
1149            }
1150            (1 | 3, Some(parent_value), Some(child_value)) => {
1151                if child_value < parent_value {
1152                    return Err(IdentityError::InvalidDelegation {
1153                        reason: "delegation weakens a lower-bound constraint",
1154                    });
1155                }
1156                strict |= child_value > parent_value;
1157            }
1158            (2, Some(parent_value), Some(child_value)) => {
1159                if child_value > parent_value {
1160                    return Err(IdentityError::InvalidDelegation {
1161                        reason: "delegation weakens an upper-bound constraint",
1162                    });
1163                }
1164                strict |= child_value < parent_value;
1165            }
1166            (_, None, Some(_)) => strict = true,
1167            (_, None, None) => {}
1168            _ => {
1169                return Err(IdentityError::InvalidDelegation {
1170                    reason: "delegation has an invalid constraint relationship",
1171                });
1172            }
1173        }
1174    }
1175    Ok(strict)
1176}
1177
1178fn validate_expiration_narrowing(
1179    parent: Option<Timestamp>,
1180    child: Option<Timestamp>,
1181) -> Result<bool, IdentityError> {
1182    match (parent, child) {
1183        (None, None) => Ok(false),
1184        (None, Some(_)) => Ok(true),
1185        (Some(_), None) => Err(IdentityError::InvalidDelegation {
1186            reason: "delegation removes parent expiration",
1187        }),
1188        (Some(parent), Some(child)) if child <= parent => Ok(child < parent),
1189        (Some(_), Some(_)) => Err(IdentityError::InvalidDelegation {
1190            reason: "delegation extends parent expiration",
1191        }),
1192    }
1193}
1194
1195fn validate_permission_narrowing(
1196    parent: DelegationPermission,
1197    child: DelegationPermission,
1198) -> Result<bool, IdentityError> {
1199    match (parent, child) {
1200        (DelegationPermission::NotDelegable, _) => Err(IdentityError::InvalidDelegation {
1201            reason: "parent grant is not delegable",
1202        }),
1203        (DelegationPermission::Delegable { .. }, DelegationPermission::NotDelegable) => Ok(true),
1204        (
1205            DelegationPermission::Delegable { remaining: parent },
1206            DelegationPermission::Delegable { remaining: child },
1207        ) if child.get() < parent.get() => Ok(true),
1208        (DelegationPermission::Delegable { .. }, DelegationPermission::Delegable { .. }) => {
1209            Err(IdentityError::InvalidDelegation {
1210                reason: "delegation remaining depth does not decrease",
1211            })
1212        }
1213    }
