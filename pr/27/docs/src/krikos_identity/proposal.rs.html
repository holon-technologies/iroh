39/// Complete pairing result submitted to later account-event construction.
40///
41/// This object is not account authority. Account policy still decides whether and how it becomes
42/// an `AuthorizeDevice` event, and controllers sign that exact later event independently.
43#[derive(Debug, Clone, PartialEq, Eq, Serialize)]
44pub struct DeviceAuthorizationProposal {
45    protocol_version: ProtocolVersion,
46    account_id: AccountId,
47    proposed_device: DeviceDescriptor,
48    proposed_device_id: DeviceId,
49    ticket_id: PairingTicketId,
50    transcript_id: PairingTranscriptId,
51    proof_id: PairingProofId,
52    confirmation: PairingConfirmationContext,
53    extensions: Extensions,
54}
55
56impl DeviceAuthorizationProposal {
57    #[allow(clippy::too_many_arguments)]
58    pub(crate) fn from_confirmed_pairing(
59        account_id: AccountId,
60        proposed_device: DeviceDescriptor,
61        proposed_device_id: DeviceId,
62        ticket_id: PairingTicketId,
63        transcript_id: PairingTranscriptId,
64        proof_id: PairingProofId,
65        confirmation: PairingConfirmationContext,
66    ) -> Result<Self, IdentityError> {
67        if proposed_device.id()? != proposed_device_id {
68            return Err(IdentityError::InvalidIdentifier {
69                resource: "paired authorization proposal device",
70            });
71        }
72        if confirmation.transcript_id() != transcript_id {
73            return Err(IdentityError::InvalidRelationship {
74                resource: "paired authorization proposal confirmation",
75            });
76        }
77        let proposal = Self {
78            protocol_version: ProtocolVersion::V1,
79            account_id,
80            proposed_device,
81            proposed_device_id,
82            ticket_id,
83            transcript_id,
84            proof_id,
85            confirmation,
86            extensions: Extensions::default(),
87        };
88        let encoded_len = encode_wire(&proposal)?.len();
89        if encoded_len > MAX_ACCOUNT_EVENT_BYTES {
90            return Err(IdentityError::limit(
91                "device authorization proposal bytes",
92                encoded_len,
93                MAX_ACCOUNT_EVENT_BYTES,
94            ));
95        }
96        Ok(proposal)
97    }
98
99    /// Domain-separated proposal identifier.
100    pub fn proposal_id(&self) -> Result<DeviceAuthorizationProposalId, IdentityError> {
101        Ok(DeviceAuthorizationProposalId(Digest::new(
102            HashAlgorithm::Blake3_256,
103            blake3::derive_key(
104                DEVICE_AUTHORIZATION_PROPOSAL_ID_CONTEXT,
105                &crate::CanonicalWire::to_canonical_bytes(self)?,
106            ),
107        )))
108    }
109
110    /// Account this proposal asks to extend.
111    pub const fn account_id(&self) -> AccountId {
112        self.account_id
113    }
114
115    /// Complete proposed descriptor.
116    pub const fn proposed_device(&self) -> &DeviceDescriptor {
117        &self.proposed_device
118    }
119
120    /// Exact identifier derived from the descriptor.
121    pub const fn proposed_device_id(&self) -> DeviceId {
122        self.proposed_device_id
123    }
124
125    /// One-time ticket whose ceremony produced this proposal.
126    pub const fn ticket_id(&self) -> PairingTicketId {
127        self.ticket_id
128    }
129
130    /// Complete authenticated pairing transcript.
131    pub const fn transcript_id(&self) -> PairingTranscriptId {
132        self.transcript_id
133    }
134
135    /// Complete four-role possession proof.
136    pub const fn proof_id(&self) -> PairingProofId {
137        self.proof_id
138    }
139
140    /// Exact two-sided short-auth confirmation context.
141    pub const fn confirmation(&self) -> PairingConfirmationContext {
142        self.confirmation
143    }
144}

146impl<'de> Deserialize<'de> for DeviceAuthorizationProposal {
147    fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
148    where
149        D: Deserializer<'de>,
150    {
151        let (
152            protocol_version,
153            account_id,
154            proposed_device,
155            proposed_device_id,
156            ticket_id,
157            transcript_id,
158            proof_id,
159            confirmation,
160            extensions,
161        ) = <(
162            ProtocolVersion,
163            AccountId,
164            DeviceDescriptor,
165            DeviceId,
166            PairingTicketId,
167            PairingTranscriptId,
168            PairingProofId,
169            PairingConfirmationContext,
170            Extensions,
171        )>::deserialize(deserializer)?;
172        if protocol_version != ProtocolVersion::V1 {
173            return Err(de::Error::custom(IdentityError::UnsupportedVersion {
174                version: protocol_version.get(),
175            }));
176        }
177        if proposed_device.id().map_err(de::Error::custom)? != proposed_device_id {
178            return Err(de::Error::custom(IdentityError::InvalidIdentifier {
179                resource: "paired authorization proposal device",
180            }));
181        }
182        if confirmation.transcript_id() != transcript_id {
183            return Err(de::Error::custom(IdentityError::InvalidRelationship {
184                resource: "paired authorization proposal confirmation",
185            }));
186        }
187        extensions
188            .validate_critical(&[])
189            .map_err(de::Error::custom)?;
190        if !extensions.as_slice().is_empty() {
191            return Err(de::Error::custom(IdentityError::InvalidRelationship {
192                resource: "device authorization proposal extensions",
193            }));
194        }
195        Ok(Self {
196            protocol_version,
197            account_id,
198            proposed_device,
199            proposed_device_id,
200            ticket_id,
201            transcript_id,
202            proof_id,
203            confirmation,
204            extensions,
205        })
206    }
207}
