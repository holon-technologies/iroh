19macro_rules! nonzero_bytes {
20    ($name:ident, $resource:literal, $debug:literal) => {
21        #[doc = $resource]
22        #[derive(Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize)]
23        pub struct $name([u8; 32]);
24
25        impl $name {
26            /// Validate exact nonzero bytes.
27            pub fn new(bytes: [u8; 32]) -> Result<Self, IdentityError> {
28                if bytes == [0; 32] {
29                    return Err(IdentityError::ZeroValue {
30                        resource: $resource,
31                    });
32                }
33                Ok(Self(bytes))
34            }
35
36            /// Borrow the exact bytes.
37            pub const fn as_bytes(&self) -> &[u8; 32] {
38                &self.0
39            }
40        }
41
42        impl fmt::Debug for $name {
43            fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
44                formatter.write_str($debug)
45            }
46        }
47
48        impl<'de> Deserialize<'de> for $name {
49            fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
50            where
51                D: Deserializer<'de>,
52            {
53                Self::new(<[u8; 32]>::deserialize(deserializer)?).map_err(de::Error::custom)
54            }
55        }
56
57        impl CanonicalCodec for $name {
58            const RESOURCE: &'static str = $resource;
59
60            fn encode_canonical(&self) -> Result<Vec<u8>, IdentityError> {
61                encode_wire(self)
62            }
63
64            fn decode_canonical(bytes: &[u8]) -> Result<Self, IdentityError> {
65                decode_wire(bytes)
66            }
67        }
68    };
69}

82fn validate_lifetime(issued_at: Timestamp, expires_at: Timestamp) -> Result<(), IdentityError> {
83    let lifetime = expires_at
84        .as_unix_millis()
85        .checked_sub(issued_at.as_unix_millis())
86        .ok_or(IdentityError::InvalidRelationship {
87            resource: "presence proof validity interval",
88        })?;
89    if lifetime == 0 {
90        return Err(IdentityError::ZeroValue {
91            resource: "presence proof lifetime",
92        });
93    }
94    if u128::from(lifetime) > MAX_PRESENCE_LIFETIME.as_millis() {
95        return Err(IdentityError::LimitExceeded {
96            resource: "presence proof lifetime milliseconds",
97            actual: usize::try_from(lifetime).unwrap_or(usize::MAX),
98            maximum: usize::try_from(MAX_PRESENCE_LIFETIME.as_millis()).unwrap_or(usize::MAX),
99        });
100    }
101    Ok(())
102}

106pub struct DevicePresenceChallenge {
107    protocol_version: ProtocolVersion,
108    account_id: AccountId,
109    device_id: DeviceId,
110    verifier_challenge: PresenceVerifierChallenge,
111    session_id: PresenceSessionId,
112    transcript_binding: Digest,
113    checkpoint_id: CheckpointId,
114    issued_at: Timestamp,
115    expires_at: Timestamp,
116    signing_key: SigningPublicKey,
117    extensions: Extensions,
118}

120impl DevicePresenceChallenge {
121    /// Construct a complete presence challenge with at most five minutes of validity.
122    #[allow(clippy::too_many_arguments)]
123    pub fn new(
124        account_id: AccountId,
125        device_id: DeviceId,
126        verifier_challenge: PresenceVerifierChallenge,
127        session_id: PresenceSessionId,
128        transcript_binding: Digest,
129        checkpoint_id: CheckpointId,
130        issued_at: Timestamp,
131        expires_at: Timestamp,
132        signing_key: SigningPublicKey,
133        extensions: Extensions,
134    ) -> Result<Self, IdentityError> {
135        validate_lifetime(issued_at, expires_at)?;
136        if transcript_binding.as_bytes() == &[0; 32] {
137            return Err(IdentityError::ZeroValue {
138                resource: "presence transcript binding",
139            });
140        }
141        extensions.validate_critical(&[])?;
142        let challenge = Self {
143            protocol_version: ProtocolVersion::V1,
144            account_id,
145            device_id,
146            verifier_challenge,
147            session_id,
148            transcript_binding,
149            checkpoint_id,
150            issued_at,
151            expires_at,
152            signing_key,
153            extensions,
154        };
155        let encoded_len = encode_wire(&challenge)?.len();
156        if encoded_len > MAX_ACCOUNT_EVENT_BYTES {
157            return Err(IdentityError::limit(
158                "device presence challenge bytes",
159                encoded_len,
160                MAX_ACCOUNT_EVENT_BYTES,
161            ));
162        }
163        Ok(challenge)
164    }
165
166    /// Exact domain-separated bytes signed by the named device key.
167    pub fn signing_bytes(&self) -> Result<Vec<u8>, IdentityError> {
168        let body = self.to_canonical_bytes()?;
169        let capacity = PRESENCE_SIGNATURE_DOMAIN
170            .len()
171            .checked_add(1)
172            .and_then(|length| length.checked_add(body.len()))
173            .ok_or(IdentityError::ArithmeticOverflow {
174                resource: "presence signature message bytes",
175            })?;
176        let mut message = Vec::with_capacity(capacity);
177        message.extend_from_slice(PRESENCE_SIGNATURE_DOMAIN);
178        message.push(0);
179        message.extend_from_slice(&body);
180        Ok(message)
181    }

293/// Complete signed response to one exact presence challenge.
294#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
295pub struct PresenceProof {
296    challenge: DevicePresenceChallenge,
297    signature: ProtocolSignature,
298}
299
300impl PresenceProof {
301    /// Construct a signed response. Cryptographic and state checks occur during verification.
302    pub fn new(
303        challenge: DevicePresenceChallenge,
304        signature: ProtocolSignature,
305    ) -> Result<Self, IdentityError> {
306        let proof = Self {
307            challenge,
308            signature,
309        };
310        let encoded_len = encode_wire(&proof)?.len();
311        if encoded_len > MAX_ACCOUNT_EVENT_BYTES {
312            return Err(IdentityError::limit(
313                "device presence proof bytes",
314                encoded_len,
315                MAX_ACCOUNT_EVENT_BYTES,
316            ));
317        }
318        Ok(proof)
319    }
320
321    /// Exact challenge that was signed.
322    pub const fn challenge(&self) -> &DevicePresenceChallenge {
323        &self.challenge
324    }
325
326    /// Exact Ed25519 response signature.
327    pub const fn signature(&self) -> ProtocolSignature {
328        self.signature
329    }
330
331    /// Domain-separated identifier of the complete proof.
332    pub fn proof_id(&self) -> Result<PresenceProofId, IdentityError> {
333        Ok(PresenceProofId(Digest::new(
334            HashAlgorithm::Blake3_256,
335            blake3::derive_key(PRESENCE_PROOF_ID_CONTEXT, &self.to_canonical_bytes()?),
336        )))
337    }
338}

376/// Verify exact-session liveness under one already-authenticated known checkpoint.
377///
378/// Success proves possession of the exact active device key in the caller-supplied checkpoint
379/// view. It does not establish that checkpoint's global freshness, network reachability outside
380/// this session, or continued authorization after the proof expires.
381pub fn verify_presence_proof(
382    proof: &PresenceProof,
383    expected_challenge: &DevicePresenceChallenge,
384    now: Timestamp,
385    view: &impl ApplicationAuthorizationView,
386) -> Result<PresenceProofId, IdentityError> {
387    if proof.challenge != *expected_challenge {
388        return Err(IdentityError::InvalidRelationship {
389            resource: "presence expected challenge context",
390        });
391    }
392    let future_skew = u64::try_from(MAX_FUTURE_CLOCK_SKEW.as_millis()).map_err(|_| {
393        IdentityError::ArithmeticOverflow {
394            resource: "presence future clock skew milliseconds",
395        }
396    })?;
397    let maximum_issue_time = now.checked_add(crate::DurationMillis::new(future_skew))?;
398    if proof.challenge.issued_at > maximum_issue_time {
399        return Err(IdentityError::InvalidRelationship {
400            resource: "presence proof future issue time",
401        });
402    }
403    if now > proof.challenge.expires_at {
404        return Err(IdentityError::StaleEvidence);
405    }
406
407    let context = view.authorization_context();
408    if proof.challenge.account_id != context.account_id() {
409        return Err(IdentityError::AccountMismatch);
410    }
411    if proof.challenge.checkpoint_id != context.checkpoint_id() {
412        return Err(IdentityError::InvalidRelationship {
413            resource: "presence authorization checkpoint",
414        });
415    }
416    match view.device_status(proof.challenge.device_id) {
417        ApplicationDeviceStatus::Unknown => return Err(IdentityError::DeviceNotAuthorized),
418        ApplicationDeviceStatus::Active => {}
419        ApplicationDeviceStatus::Suspended => return Err(IdentityError::DeviceSuspended),
420        ApplicationDeviceStatus::Revoked => return Err(IdentityError::DeviceRevoked),
421    }
422    let authorization = view
423        .device_authorization(proof.challenge.device_id)
424        .ok_or(IdentityError::DeviceNotAuthorized)?;
425    if authorization.device_id() != proof.challenge.device_id {
426        return Err(IdentityError::InvalidIdentifier {
427            resource: "presence authorized device",
428        });
429    }
430    if authorization.authorization_epoch() > context.epoch() {
431        return Err(IdentityError::InvalidEpoch);
432    }
433    if authorization.descriptor().application_signing_key() != proof.challenge.signing_key {
434        return Err(IdentityError::InvalidRelationship {
435            resource: "presence exact device signing key",
436        });
437    }
438
439    let public_key = Ed25519PublicKey::from_bytes(proof.challenge.signing_key.as_bytes())
440        .map_err(|_| IdentityError::InvalidSignature)?;
441    let signature = Ed25519Signature::try_from(proof.signature.as_bytes().as_slice())
442        .map_err(|_| IdentityError::InvalidSignature)?;
443    public_key
444        .verify(&proof.challenge.signing_bytes()?, &signature)
445        .map_err(|_| IdentityError::InvalidSignature)?;
446    proof.proof_id()
