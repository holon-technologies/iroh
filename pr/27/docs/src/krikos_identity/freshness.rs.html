1//! Explicit checkpoint freshness decisions with monotonic caller tightening.
2
3use crate::{
4    AuthorizationContext, DurationMillis, FreshnessEvidence, FreshnessRequirement, IdentityError,
5    ProviderMode, ProviderPolicy, ProviderPolicyId, ProviderQuorum, Timestamp,
6};
7
8/// Verified freshness basis for one exact account/checkpoint/epoch context.
9#[derive(Debug, Clone, Copy, PartialEq, Eq)]
10pub struct FreshnessDecision {
11    context: AuthorizationContext,
12    provider_policy_id: ProviderPolicyId,
13    required_quorum: Option<ProviderQuorum>,
14    maximum_age: Option<DurationMillis>,
15    provider_observed_at: Option<Timestamp>,
16}
17
18impl FreshnessDecision {
19    /// Exact account, epoch, and checkpoint to which the decision applies.
20    pub const fn context(self) -> AuthorizationContext {
21        self.context
22    }
23
24    /// Authenticated provider policy used by the decision.
25    pub const fn provider_policy_id(self) -> ProviderPolicyId {
26        self.provider_policy_id
27    }
28
29    /// Effective distinct-provider quorum, absent for latest-known-only evaluation.
30    pub const fn required_quorum(self) -> Option<ProviderQuorum> {
31        self.required_quorum
32    }
33
34    /// Effective age bound at the explicit verifier time, absent for latest-known-only evaluation.
35    pub const fn maximum_age(self) -> Option<DurationMillis> {
36        self.maximum_age
37    }
38
39    /// Deterministic quorum-th checkpoint observation time, if online evidence was required.
40    pub const fn provider_observed_at(self) -> Option<Timestamp> {
41        self.provider_observed_at
42    }
43}
44
45/// Evaluate account and caller freshness requirements without permitting caller weakening.
46///
47/// Provider maximum age is measured from the signed checkpoint-log observation to the explicit
48/// verifier time. A later tree head proves continued inclusion but cannot refresh the checkpoint's
49/// original observation. The checkpoint's account-supplied metadata timestamp is never an
50/// authority source. `LatestKnown` establishes only the exact locally trusted checkpoint context.
51pub fn evaluate_freshness(
52    context: AuthorizationContext,
53    provider_policy: &ProviderPolicy,
54    account_requirement: FreshnessRequirement,
55    caller_requirement: FreshnessRequirement,
56    evidence: &FreshnessEvidence,
57    verified_at: Timestamp,
58) -> Result<FreshnessDecision, IdentityError> {
59    if evidence.checkpoint_id() != context.checkpoint_id() {
60        return Err(IdentityError::InvalidRelationship {
61            resource: "freshness decision checkpoint",
62        });
63    }
64    let provider_policy_id = provider_policy.id()?;
65    let requested = combine_requirements(account_requirement, caller_requirement);
66    let Some((requested_quorum, requested_maximum_age)) = requested else {
67        return Ok(FreshnessDecision {
68            context,
69            provider_policy_id,
70            required_quorum: None,
71            maximum_age: None,
72            provider_observed_at: None,
73        });
74    };
75
76    let replicated = match provider_policy.mode() {
77        ProviderMode::LocalOnly => return Err(IdentityError::FreshnessUnavailable),
78        ProviderMode::Replicated(replicated) => replicated,
79    };
80    if evidence.provider_policy_id() != Some(provider_policy_id) {
81        return Err(IdentityError::PolicyVersionMismatch);
82    }
83    let receipts = evidence
84        .provider_receipts()
85        .ok_or(IdentityError::FreshnessUnavailable)?;
86    let required = usize::from(
87        requested_quorum
88            .get()
89            .max(replicated.sufficient_threshold().get()),
90    );
91    let maximum_age = DurationMillis::new(
92        requested_maximum_age
93            .get()
94            .min(replicated.maximum_evidence_age().get()),
95    );
96    let required_quorum = ProviderQuorum::new(u16::try_from(required).map_err(|_| {
97        IdentityError::ArithmeticOverflow {
98            resource: "freshness decision provider quorum",
99        }
100    })?)?;
101    let future_skew =
102        u64::try_from(crate::limits::MAX_FUTURE_CLOCK_SKEW.as_millis()).map_err(|_| {
103            IdentityError::ArithmeticOverflow {
104                resource: "freshness future clock skew milliseconds",
105            }
106        })?;
107    let maximum_observation = verified_at.checked_add(DurationMillis::new(future_skew))?;
108    let mut valid_times = Vec::new();
109    let mut stale_configured = false;
110    for receipt in receipts.as_slice() {
111        if receipt.entry().account_id() != context.account_id() {
112            return Err(IdentityError::AccountMismatch);
113        }
114        let Some(provider) = replicated
115            .providers()
116            .iter()
117            .find(|provider| provider.id() == Ok(receipt.provider_id()))
118        else {
119            continue;
120        };
121        receipt.verify(provider)?;
122        let entry_time = receipt.entry().observed_at().as_unix_millis();
123        let head_time = receipt.signed_head().body().observed_at().as_unix_millis();
124        if entry_time > maximum_observation.as_unix_millis()
125            || head_time > maximum_observation.as_unix_millis()
126        {
127            stale_configured = true;
128            continue;
129        }
130        let age = if verified_at.as_unix_millis() >= entry_time {
131            verified_at.as_unix_millis() - entry_time
132        } else {
133            0
134        };
135        if age > maximum_age.get() {
136            stale_configured = true;
137            continue;
138        }
139        valid_times.push(receipt.entry().observed_at());
140    }
141    if valid_times.len() < required {
142        return Err(if stale_configured {
143            IdentityError::StaleEvidence
144        } else {
145            IdentityError::FreshnessUnavailable
146        });
147    }
148    valid_times.sort_unstable();
149    Ok(FreshnessDecision {
150        context,
151        provider_policy_id,
152        required_quorum: Some(required_quorum),
153        maximum_age: Some(maximum_age),
154        provider_observed_at: Some(valid_times[required - 1]),
155    })
156}
157
158fn combine_requirements(
159    account: FreshnessRequirement,
160    caller: FreshnessRequirement,
161) -> Option<(ProviderQuorum, DurationMillis)> {
162    match (account, caller) {
163        (FreshnessRequirement::LatestKnown, FreshnessRequirement::LatestKnown) => None,
164        (FreshnessRequirement::ProviderQuorum(requirement), FreshnessRequirement::LatestKnown)
165        | (FreshnessRequirement::LatestKnown, FreshnessRequirement::ProviderQuorum(requirement)) => {
166            Some((requirement.required(), requirement.maximum_age()))
167        }
168        (
169            FreshnessRequirement::ProviderQuorum(account),
170            FreshnessRequirement::ProviderQuorum(caller),
171        ) => {
172            let required = if account.required() >= caller.required() {
173                account.required()
174            } else {
175                caller.required()
176            };
177            Some((
178                required,
179                DurationMillis::new(account.maximum_age().get().min(caller.maximum_age().get())),
180            ))
181        }
182    }
183}
