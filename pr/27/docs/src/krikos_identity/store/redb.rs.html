70impl StoredAccountWire {
71    fn from_stored(stored: &StoredAccount) -> Result<Self, IdentityError> {
72        Ok(Self {
73            version: 2,
74            genesis: stored.genesis.clone(),
75            events: BoundedVec::new(
76                "stored account event envelopes",
77                stored.events.values().cloned().collect(),
78            )?,
79            event_journal: BoundedVec::new(
80                "stored account event journal",
81                stored.event_journal.clone(),
82            )?,
83            effects: BoundedVec::new(
84                "stored account effects",
85                stored
86                    .outbox
87                    .values()
88                    .map(EffectWire::from_record)
89                    .collect::<Result<Vec<_>, IdentityError>>()?,
90            )?,
91            rotations: BoundedVec::new(
92                "stored group key rotations",
93                stored
94                    .group_key_rotations
95                    .values()
96                    .map(RotationWire::from_record)
97                    .collect::<Result<Vec<_>, IdentityError>>()?,
98            )?,
99            checkpoints: BoundedVec::new(
100                "stored account checkpoints",
101                stored
102                    .checkpoint_journal
103                    .iter()
104                    .map(|checkpoint_id| {
105                        stored
106                            .checkpoints
107                            .get(checkpoint_id)
108                            .map(|record| StoredCheckpointWire {
109                                checkpoint: record.checkpoint.clone(),
110                                transition_event: record.transition_event.clone(),
111                            })
112                            .ok_or(IdentityError::StorageCorruption)
113                    })
114                    .collect::<Result<Vec<_>, IdentityError>>()?,
115            )?,
116        })
117    }
118
119    fn into_stored(self, expected_account_id: AccountId) -> Result<StoredAccount, IdentityError> {
120        if self.version != 2 || self.genesis.account_id()? != expected_account_id {
121            return Err(IdentityError::StorageCorruption);
122        }
123        let mut events = std::collections::BTreeMap::new();
124        for event in self.events.into_vec() {
125            if event.body().account_id() != expected_account_id {
126                return Err(IdentityError::StorageCorruption);
127            }
128            let authorization_id = event.event_authorization_id()?;
129            if events.insert(authorization_id, event).is_some() {
130                return Err(IdentityError::StorageCorruption);
131            }
132        }
133        let mut outbox = std::collections::BTreeMap::new();
134        for wire in self.effects.into_vec() {
135            let record = wire.into_record(expected_account_id)?;
136            if outbox.insert(record.id, record).is_some() {
137                return Err(IdentityError::StorageCorruption);
138            }
139        }
140        let mut group_key_rotations = std::collections::BTreeMap::new();
141        for wire in self.rotations.into_vec() {
142            let record = wire.into_record(expected_account_id)?;
143            let key = (record.application_id, record.group_id);
144            if group_key_rotations.insert(key, record).is_some() {
145                return Err(IdentityError::StorageCorruption);
146            }
147        }
148        let mut checkpoints = std::collections::BTreeMap::new();
149        let mut checkpoint_journal = Vec::new();
150        for wire in self.checkpoints.into_vec() {
151            if wire.checkpoint.body().account_id() != expected_account_id {
152                return Err(IdentityError::StorageCorruption);
153            }
154            let checkpoint_id = wire.checkpoint.checkpoint_id()?;
155            let retained = StoredCheckpoint {
156                checkpoint: wire.checkpoint,
157                transition_event: wire.transition_event,
158            };
159            if checkpoints.insert(checkpoint_id, retained).is_some() {
160                return Err(IdentityError::StorageCorruption);
161            }
162            checkpoint_journal.push(checkpoint_id);
163        }
164        let mut stored = StoredAccount {
165            projection: crate::AccountState::from_genesis(&self.genesis)?,
166            genesis: self.genesis,
167            events,
168            event_journal: self.event_journal.into_vec(),
169            outbox,
170            group_key_rotations,
171            checkpoints,
172            checkpoint_journal,
173        };
174        let (projection, required_effects) = stored.rebuild_projection_and_effects()?;
175        if required_effects.len() != stored.outbox.len()
176            || required_effects.iter().any(|(id, effect)| {
177                stored
178                    .outbox
179                    .get(id)
180                    .is_none_or(|record| record.effect != *effect)
181            })
182        {
183            return Err(IdentityError::StorageCorruption);
184        }
185        stored.projection = projection;
186        stored.validate_checkpoint_journal()?;
187        let _ = stored.snapshot()?;
188        Ok(stored)
189    }

194        Ok(Self {
195            application_id: record.application_id,
196            group_id: record.group_id,
197            authorizing_account_epoch: record.authorizing_account_epoch,
198            group_key_epoch: record.group_key_epoch,
199            revision_heads: BoundedVec::new(
200                "stored group rotation revision heads",
201                record.revision_heads.clone(),
202            )?,
203            recipient_key_wraps: record.recipient_key_wraps.clone(),

206
207    fn into_record(self, account_id: AccountId) -> Result<StoredGroupKeyRotation, IdentityError> {
208        for wrap in self.recipient_key_wraps.as_slice() {
209            let header = wrap.header();
210            if header.account_id() != account_id
211                || header.application_id() != self.application_id
212                || header.group_id() != self.group_id
213                || header.authorizing_account_epoch() != self.authorizing_account_epoch
214                || header.group_key_epoch() != self.group_key_epoch
215            {
216                return Err(IdentityError::StorageCorruption);
217            }
218        }
219        Ok(StoredGroupKeyRotation {
220            account_id,
221            application_id: self.application_id,
222            group_id: self.group_id,
223            authorizing_account_epoch: self.authorizing_account_epoch,
224            group_key_epoch: self.group_key_epoch,
225            revision_heads: self.revision_heads.into_vec(),
226            recipient_key_wraps: self.recipient_key_wraps,
227        })
228    }

231impl EffectWire {
232    fn from_record(record: &EffectRecord) -> Result<Self, IdentityError> {
233        let (effect_code, event_id, epoch) = match record.effect {
234            ProjectionEffect::PublishAccountEvent { event_id } => (1, event_id, None),
235            ProjectionEffect::RotateGroupKeys { event_id, epoch } => (2, event_id, Some(epoch)),
236            ProjectionEffect::NotifyAccountChanged { event_id } => (3, event_id, None),
237            ProjectionEffect::NotifyForkDetected { event_id } => (4, event_id, None),
238        };
239        let (state_code, state_at, lease_id) = match record.state {
240            EffectState::Pending(PendingEffect::Scheduled(at)) => (1, at, None),
241            EffectState::Pending(PendingEffect::Exhausted(at)) => (2, at, None),
242            EffectState::Claimed {
243                lease_id,
244                leased_until,
245            } => (3, leased_until, Some(*lease_id.as_bytes())),
246            EffectState::Completed {
247                lease_id,
248                completed_at,
249            } => (4, completed_at, Some(*lease_id.as_bytes())),
250        };
251        let failure = record.last_failure.map(|failure| match failure {
252            EffectFailure::Transient(code) => (1, code),
253            EffectFailure::Permanent(code) => (2, code),
254        });
255        Ok(Self {
256            id: *record.id.as_bytes(),
257            account_id: record.account_id,
258            effect_code,
259            event_id,
260            epoch,
261            state_code,
262            state_at,
263            lease_id,
264            attempt_count: record.attempt_count,
265            failure,
266        })
267    }
268
269    fn into_record(self, expected_account_id: AccountId) -> Result<EffectRecord, IdentityError> {
270        if self.account_id != expected_account_id || self.attempt_count > MAX_RETRIES {
271            return Err(IdentityError::StorageCorruption);
272        }
273        let effect = match (self.effect_code, self.epoch) {
274            (1, None) => ProjectionEffect::PublishAccountEvent {
275                event_id: self.event_id,
276            },
277            (2, Some(epoch)) => ProjectionEffect::RotateGroupKeys {
278                event_id: self.event_id,
279                epoch,
280            },
281            (3, None) => ProjectionEffect::NotifyAccountChanged {
282                event_id: self.event_id,
283            },
284            (4, None) => ProjectionEffect::NotifyForkDetected {
285                event_id: self.event_id,
286            },
287            _ => return Err(IdentityError::StorageCorruption),
288        };
289        let state = match (self.state_code, self.lease_id) {
290            (1, None) => EffectState::Pending(PendingEffect::Scheduled(self.state_at)),
291            (2, None) => EffectState::Pending(PendingEffect::Exhausted(self.state_at)),
292            (3, Some(lease)) => EffectState::Claimed {
293                lease_id: LeaseId::new(lease).map_err(|_| IdentityError::StorageCorruption)?,
294                leased_until: self.state_at,
295            },
296            (4, Some(lease)) => EffectState::Completed {
297                lease_id: LeaseId::new(lease).map_err(|_| IdentityError::StorageCorruption)?,
298                completed_at: self.state_at,
299            },
300            _ => return Err(IdentityError::StorageCorruption),
301        };
302        let last_failure = match self.failure {
303            None => None,
304            Some((1, code)) => {
305                Some(EffectFailure::transient(code).map_err(|_| IdentityError::StorageCorruption)?)
306            }
307            Some((2, code)) => {
308                Some(EffectFailure::permanent(code).map_err(|_| IdentityError::StorageCorruption)?)
309            }
310            Some(_) => return Err(IdentityError::StorageCorruption),
311        };
312        let id = EffectId(self.id);
313        if derive_effect_id(expected_account_id, effect)? != id {
314            return Err(IdentityError::StorageCorruption);
315        }
316        Ok(EffectRecord {
317            id,
318            account_id: expected_account_id,
319            effect,
320            state,
321            attempt_count: self.attempt_count,
322            last_failure,
323        })
324    }

333impl RedbAccountStore {
334    /// Open or create a database and authenticate every retained account projection.
335    pub fn open(path: impl AsRef<Path>) -> Result<Self, IdentityError> {
336        let path = path.as_ref();
337        crate::redb_guard::validate_existing_redb_file(path)?;
338        let database = Database::create(path).map_err(|_| IdentityError::StorageCorruption)?;
339        {
340            let write = database
341                .begin_write()
342                .map_err(|_| IdentityError::StorageCorruption)?;
343            let _ = write
344                .open_table(ACCOUNT_TABLE)
345                .map_err(|_| IdentityError::StorageCorruption)?;
346            write
347                .commit()
348                .map_err(|_| IdentityError::StorageCorruption)?;
349        }
350        let store = Self {
351            database: Arc::new(database),
352        };
353        store.validate_all()?;
354        Ok(store)
355    }
356
357    fn validate_all(&self) -> Result<(), IdentityError> {
358        let read = self
359            .database
360            .begin_read()
361            .map_err(|_| IdentityError::StorageCorruption)?;
362        let table = read
363            .open_table(ACCOUNT_TABLE)
364            .map_err(|_| IdentityError::StorageCorruption)?;
365        let iterator = table.iter().map_err(|_| IdentityError::StorageCorruption)?;
366        for entry in iterator {
367            let (key, value) = entry.map_err(|_| IdentityError::StorageCorruption)?;
368            let account_id = AccountId::from_canonical_bytes(key.value())
369                .map_err(|_| IdentityError::StorageCorruption)?;
370            let _ = decode_stored(account_id, value.value())?;
371        }
372        Ok(())
373    }
374
375    fn load_sync(&self, account_id: AccountId) -> Result<Option<StoredAccount>, IdentityError> {
376        let key = account_id.to_canonical_bytes()?;
377        let read = self
378            .database
379            .begin_read()
380            .map_err(|_| IdentityError::StorageCorruption)?;
381        let table = read
382            .open_table(ACCOUNT_TABLE)
383            .map_err(|_| IdentityError::StorageCorruption)?;
384        table
385            .get(key.as_slice())
386            .map_err(|_| IdentityError::StorageCorruption)?
387            .map(|value| decode_stored(account_id, value.value()))
388            .transpose()
389    }
390
391    fn write_stored(
392        table: &mut redb::Table<'_, &[u8], &[u8]>,
393        account_id: AccountId,
394        stored: &StoredAccount,
395    ) -> Result<(), IdentityError> {
396        let key = account_id.to_canonical_bytes()?;
397        let bytes = encode_stored(stored)?;
398        table
399            .insert(key.as_slice(), bytes.as_slice())
400            .map_err(|_| IdentityError::StorageCorruption)?;
401        Ok(())
402    }
403}

405impl AccountStore for RedbAccountStore {
406    fn create_account(&self, genesis: AccountGenesis) -> StoreFuture<'_, AccountSnapshot> {
407        Box::pin(async move {
408            let account_id = genesis.account_id()?;
409            let write = self
410                .database
411                .begin_write()
412                .map_err(|_| IdentityError::StorageCorruption)?;
413            {
414                let mut table = write
415                    .open_table(ACCOUNT_TABLE)
416                    .map_err(|_| IdentityError::StorageCorruption)?;
417                let key = account_id.to_canonical_bytes()?;
418                if table
419                    .get(key.as_slice())
420                    .map_err(|_| IdentityError::StorageCorruption)?
421                    .is_some()
422                {
423                    return Err(IdentityError::InvalidRelationship {
424                        resource: "account store duplicate genesis",
425                    });
426                }
427                let stored = StoredAccount {
428                    projection: crate::AccountState::from_genesis(&genesis)?,
429                    genesis,
430                    events: std::collections::BTreeMap::new(),
431                    event_journal: Vec::new(),
432                    outbox: std::collections::BTreeMap::new(),
433                    group_key_rotations: std::collections::BTreeMap::new(),
434                    checkpoints: std::collections::BTreeMap::new(),
435                    checkpoint_journal: Vec::new(),
436                };
437                let snapshot = stored.snapshot()?;
438                Self::write_stored(&mut table, account_id, &stored)?;
439                drop(table);
440                write
441                    .commit()
442                    .map_err(|_| IdentityError::StorageCorruption)?;
443                Ok(snapshot)
444            }
445        })
446    }

448    fn load_account(&self, account_id: AccountId) -> StoreFuture<'_, Option<AccountSnapshot>> {
449        Box::pin(async move {

451                .map(|stored| stored.snapshot())
452                .transpose()
453        })
454    }

456    fn commit_event(
457        &self,
458        expected_revision: AccountRevision,
459        event: AuthorizedEvent,
460    ) -> StoreFuture<'_, CommitReceipt> {
461        Box::pin(async move {
462            let account_id = expected_revision.account_id();
463            let write = self
464                .database
465                .begin_write()
466                .map_err(|_| IdentityError::StorageCorruption)?;
467            let receipt;
468            {
469                let mut table = write
470                    .open_table(ACCOUNT_TABLE)
471                    .map_err(|_| IdentityError::StorageCorruption)?;
472                let key = account_id.to_canonical_bytes()?;
473                let value = table
474                    .get(key.as_slice())
475                    .map_err(|_| IdentityError::StorageCorruption)?
476                    .ok_or(IdentityError::InvalidRelationship {
477                        resource: "account store missing account",
478                    })?;
479                let mut stored = decode_stored(account_id, value.value())?;
480                drop(value);
481                receipt = stored.commit_event(&expected_revision, event)?;
482                Self::write_stored(&mut table, account_id, &stored)?;
483            }
484            write
485                .commit()
486                .map_err(|_| IdentityError::StorageCorruption)?;
487            Ok(receipt)
488        })
489    }

655        &self,
656        account_id: AccountId,
657        update: impl FnOnce(&mut StoredAccount) -> Result<T, IdentityError>,
658    ) -> Result<T, IdentityError> {
659        let write = self
660            .database
661            .begin_write()
662            .map_err(|_| IdentityError::StorageCorruption)?;
663        let output;
664        {
665            let mut table = write
666                .open_table(ACCOUNT_TABLE)
667                .map_err(|_| IdentityError::StorageCorruption)?;
668            let key = account_id.to_canonical_bytes()?;
669            let value = table
670                .get(key.as_slice())
671                .map_err(|_| IdentityError::StorageCorruption)?
672                .ok_or(IdentityError::InvalidRelationship {
673                    resource: "account store missing account",
674                })?;
675            let mut stored = decode_stored(account_id, value.value())?;
676            drop(value);
677            output = update(&mut stored)?;
678            Self::write_stored(&mut table, account_id, &stored)?;
679        }
680        write
681            .commit()
682            .map_err(|_| IdentityError::StorageCorruption)?;
683        Ok(output)
684    }
685}
686
687fn encode_stored(stored: &StoredAccount) -> Result<Vec<u8>, IdentityError> {
688    let wire = StoredAccountWire::from_stored(stored)?;
689    let bytes = encode_wire(&wire)?;
690    if bytes.len() > MAX_ACCOUNT_RECORD_BYTES {
691        return Err(IdentityError::limit(
692            "stored account source bytes",
693            bytes.len(),
694            MAX_ACCOUNT_RECORD_BYTES,
695        ));
696    }
697    Ok(bytes)
698}
699
700fn decode_stored(account_id: AccountId, bytes: &[u8]) -> Result<StoredAccount, IdentityError> {
701    if bytes.len() > MAX_ACCOUNT_RECORD_BYTES {
702        return Err(IdentityError::StorageCorruption);
703    }
704    let wire =
705        decode_wire::<StoredAccountWire>(bytes).map_err(|_| IdentityError::StorageCorruption)?;
706    wire.into_stored(account_id)
707        .map_err(|_| IdentityError::StorageCorruption)
708}
