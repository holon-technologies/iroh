3use std::time::Duration;

5/// Largest accepted canonical protocol object.
6pub const MAX_ENCODED_OBJECT_BYTES: usize = 1024 * 1024;
7/// Largest accepted account-control event.
8pub const MAX_ACCOUNT_EVENT_BYTES: usize = 256 * 1024;
9/// Largest accepted device-pairing ticket.
10pub const MAX_PAIRING_TICKET_BYTES: usize = 16 * 1024;
11/// Largest accepted synchronization frame.
12pub const MAX_SYNC_FRAME_BYTES: usize = 4 * 1024 * 1024;
13/// Maximum account events in one synchronization batch.
14pub const MAX_EVENTS_PER_SYNC_BATCH: usize = 256;
15/// Maximum controllers retained by one account.
16pub const MAX_CONTROLLERS: usize = 64;
17/// Maximum devices retained by one account, including tombstones.
18pub const MAX_DEVICES: usize = 1024;
19/// Maximum rules in one control policy.
20pub const MAX_POLICY_RULES: usize = 64;
21/// Maximum signatures accepted as authorization for one object.
22pub const MAX_AUTHORIZATION_SIGNATURES: usize = 64;
23/// Maximum simultaneously accepted controller-signature suites during migration.
24pub const MAX_ACTIVE_CRYPTO_SUITES: usize = 2;
25/// Maximum encoded public signing key in an explicit crypto migration.
26pub const MAX_ALGORITHM_PUBLIC_KEY_BYTES: usize = 4 * 1024;
27/// Maximum encoded signature in an explicit crypto migration.
28pub const MAX_ALGORITHM_SIGNATURE_BYTES: usize = 8 * 1024;
29/// Maximum capabilities granted to one device.
30pub const MAX_CAPABILITIES_PER_DEVICE: usize = 128;
31/// Maximum constraints attached to one capability.
32pub const MAX_CONSTRAINTS_PER_CAPABILITY: usize = 32;
33/// Maximum capability-delegation chain depth.
34pub const MAX_DELEGATION_DEPTH: usize = 8;
35/// Maximum transparency providers configured for one account.
36pub const MAX_TRANSPARENCY_PROVIDERS: usize = 16;
37/// Maximum recovery guardians configured for one account.
38pub const MAX_RECOVERY_GUARDIANS: usize = 16;
39/// Maximum hashes accepted in one Merkle proof path.
40pub const MAX_MERKLE_PROOF_HASHES: usize = 64;
41/// Maximum leaves materialized in one account-state sorted Merkle set.
42pub const MAX_MERKLE_SET_LEAVES: usize = 2_048;
43/// Maximum leaves held by one bounded in-memory provider-log generation.
44///
45/// Durable providers rotate to a new versioned log generation before reaching this bound.
46pub const MAX_MERKLE_LOG_LEAVES: usize = 1_048_576;
47/// Maximum extension fields on one extensible v1 object.
48pub const MAX_EXTENSIONS: usize = 32;
49/// Maximum value bytes in one extension field.
50pub const MAX_EXTENSION_VALUE_BYTES: usize = 16 * 1024;
51/// Maximum combined value bytes across an object's extension fields.
52pub const MAX_TOTAL_EXTENSION_BYTES: usize = 64 * 1024;
53/// Maximum retained branch heads for one account fork.
54pub const MAX_FORK_HEADS: usize = 16;
55/// Maximum encoded fork evidence retained in one account revision.
56pub const MAX_FORK_EVIDENCE_BYTES: usize = 4 * 1024 * 1024;
57/// Maximum UTF-8 bytes in a capability namespace or action.
58pub const MAX_CAPABILITY_NAME_BYTES: usize = 128;
59/// Maximum encoded bytes in a capability resource selector.
60pub const MAX_RESOURCE_SELECTOR_BYTES: usize = 1024;
61/// Maximum encrypted private-metadata envelope.
62pub const MAX_PRIVATE_METADATA_BYTES: usize = 256 * 1024;
63/// Maximum encrypted account backup envelope.
64pub const MAX_PRIVATE_BACKUP_BYTES: usize = MAX_ENCODED_OBJECT_BYTES;
65/// Maximum application-private bytes carried by one account backup.
66pub const MAX_APPLICATION_BACKUP_DATA_BYTES: usize = 256 * 1024;
67/// Maximum private relationship-label bytes accepted before blinding.
68pub const MAX_PRIVATE_LABEL_BYTES: usize = 128;
69/// Maximum normalized relying-party context bytes.
70pub const MAX_RELYING_PARTY_CONTEXT_BYTES: usize = 253;
71/// Maximum selectively disclosed claims in one portable credential.
72pub const MAX_CREDENTIAL_CLAIMS: usize = 64;
73/// Maximum portable-credential claim-name bytes.
74pub const MAX_CREDENTIAL_CLAIM_NAME_BYTES: usize = 64;
75/// Maximum disclosed value bytes in one portable-credential claim.
76pub const MAX_CREDENTIAL_CLAIM_VALUE_BYTES: usize = 16 * 1024;
77/// Maximum complete canonical portable credential export.
78pub const MAX_PORTABLE_CREDENTIAL_BYTES: usize = 512 * 1024;
79/// Maximum exact canonical bytes sent to an offline or hardware signer.
80pub const MAX_OFFLINE_SIGNING_REQUEST_BYTES: usize = MAX_ACCOUNT_EVENT_BYTES;
81/// Maximum complete canonical signed application event.
82pub const MAX_APPLICATION_EVENT_BYTES: usize = 1024 * 1024;
83/// Maximum application payload, reserving bounded envelope/signature overhead.
84pub const MAX_APPLICATION_PAYLOAD_BYTES: usize = MAX_APPLICATION_EVENT_BYTES - 4 * 1024;
85/// Maximum encoded wrapped group key for one recipient.
86pub const MAX_KEY_WRAP_BYTES: usize = 4 * 1024;
87/// Maximum pending account proposals retained locally.
88pub const MAX_PENDING_PROPOSALS: usize = 128;
89/// Maximum live, unconsumed pairing tickets retained locally.
90pub const MAX_LIVE_PAIRING_TICKETS: usize = 64;
91/// Maximum durable pairing nonce tombstones retained by one nonce store instance.
92///
93/// Reaching this bound is an availability failure; an implementation must never evict a
94/// tombstone in a way that permits a previously consumed ticket to revive.
95pub const MAX_PAIRING_NONCE_TOMBSTONES: usize = 65_536;
96/// Maximum concurrent recovery attempts for one account.
97pub const MAX_CONCURRENT_RECOVERY_ATTEMPTS: usize = 8;
98/// Maximum complete canonical social attestation.
99pub const MAX_SOCIAL_ATTESTATION_BYTES: usize = 64 * 1024;
100/// Maximum explicitly enabled social-attestation chain depth.
101pub const MAX_SOCIAL_TRANSITIVITY_DEPTH: usize = 8;
102/// Maximum lowercase ASCII DNS-style name bytes.
103pub const MAX_NORMALIZED_NAME_BYTES: usize = 253;
104/// Maximum complete canonical signed name claim.
105pub const MAX_NAME_CLAIM_BYTES: usize = 64 * 1024;
106/// Maximum name claims returned by one resolver query.
107pub const MAX_NAME_CLAIMS: usize = 64;
108/// Maximum history events processed in one page or projection call.
109pub const MAX_HISTORY_PAGE_EVENTS: usize = 256;
110/// Maximum encoded history bytes returned in one page.
111pub const MAX_HISTORY_PAGE_BYTES: usize = 4 * 1024 * 1024;
112/// Maximum total bytes exchanged in one synchronization session.
113pub const MAX_SYNC_SESSION_BYTES: usize = 16 * 1024 * 1024;
114/// Maximum bytes returned by one provider account-history request.
115pub const MAX_PROVIDER_ACCOUNT_RESPONSE_BYTES: usize = 4 * 1024 * 1024;
116/// Maximum concurrently running identity streams or child tasks.
117pub const MAX_CONCURRENT_IDENTITY_TASKS: usize = 64;
118/// Capacity of identity actor and effect queues.
119pub const IDENTITY_QUEUE_CAPACITY: usize = 256;
120/// Maximum attempts for a bounded retry policy.
121pub const MAX_RETRIES: u8 = 8;
122/// Maximum validity of a device-pairing ticket.
123pub const MAX_PAIRING_LIFETIME: Duration = Duration::from_secs(10 * 60);
124/// Maximum validity of a presence proof.
125pub const MAX_PRESENCE_LIFETIME: Duration = Duration::from_secs(5 * 60);
126/// Maximum accepted future clock skew for time-bound proofs.
