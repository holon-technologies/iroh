31impl PendingRecoveryObservation {
32    fn from_admission(
33        evidence: &AdmissionEvidence,
34        recovery_policy: &RecoveryPolicy,
35    ) -> Result<Self, IdentityError> {
36        let delay = evidence.delay();
37        let provider_policy_id = delay
38            .provider_policy_id()
39            .ok_or(IdentityError::FreshnessUnavailable)?;
40        let required_quorum = delay
41            .required_quorum()
42            .ok_or(IdentityError::FreshnessUnavailable)?;
43        let observed_at = delay
44            .observed_at()
45            .ok_or(IdentityError::FreshnessUnavailable)?;
46        let receipts = delay
47            .provider_receipts()
48            .ok_or(IdentityError::FreshnessUnavailable)?
49            .clone();
50        Ok(Self {
51            admission_evidence_id: evidence.admission_evidence_id()?,
52            provider_policy_id,
53            required_quorum,
54            observed_at,
55            delay_deadline: observed_at.checked_add(recovery_policy.delay())?,
56            lifetime_deadline: observed_at.checked_add(recovery_policy.lifetime())?,
57            receipts,
58        })
59    }
60
61    fn matches_completion_anchor(&self, anchor: &crate::RecoveryDelayAnchor) -> bool {
62        self.provider_policy_id == anchor.provider_policy_id()
63            && self.required_quorum == anchor.required_quorum()
64            && self.observed_at == anchor.observed_at()
65            && receipt_entries_match(&self.receipts, anchor.receipts())
66    }
67}
68
69fn receipt_entries_match(begin: &ProviderReceipts, completion: &ProviderReceipts) -> bool {
70    begin.as_slice().len() == completion.as_slice().len()
71        && begin
72            .as_slice()
73            .iter()
74            .zip(completion.as_slice())
75            .all(|(begin, completion)| {
76                begin.provider_id() == completion.provider_id()
77                    && begin.entry() == completion.entry()
78                    && begin.leaf_index() == completion.leaf_index()
79            })
80}

319impl AccountRevision {
320    pub(crate) fn from_frozen_heads(
321        account_id: AccountId,
322        heads: Vec<EventId>,
323    ) -> Result<Self, IdentityError> {
324        if heads.len() > crate::limits::MAX_FORK_HEADS {
325            return Err(IdentityError::limit(
326                "account revision heads",
327                heads.len(),
328                crate::limits::MAX_FORK_HEADS,
329            ));
330        }
331        for pair in heads.windows(2) {
332            if pair[0] == pair[1] {
333                return Err(IdentityError::DuplicateElement {
334                    resource: "account revision heads",
335                });
336            }
337            if pair[0] > pair[1] {
338                return Err(IdentityError::NonCanonical);
339            }
340        }
341        Ok(Self { account_id, heads })
342    }

346        self.account_id
347    }

534    /// Apply a possible conflict whose pre-state was evicted from the bounded memory cache.
535    ///
536    /// `accepted_event` and `historical_pre_state` must come from the account's authenticated
537    /// durable lineage. This method revalidates both the accepted and incoming bodies from that
538    /// exact pre-state before opening a fork; storage integration remains responsible for proving
539    /// that the accepted event belongs to the durable lineage selected by this projection.
540    pub(crate) fn validate_and_apply_historical_conflict(
541        &mut self,
542        historical_pre_state: &AccountState,
543        accepted_path: &[AuthorizedEvent],
544        incoming_event: &AuthorizedEvent,
545    ) -> Result<ApplyOutcome, IdentityError> {
546        let accepted_event = accepted_path
547            .first()
548            .ok_or(IdentityError::StorageCorruption)?;
549        let sequence = incoming_event.body().sequence();
550        if self
551            .historical_state_required_through
552            .is_none_or(|through| sequence > through)
553        {
554            return Err(IdentityError::InvalidRelationship {
555                resource: "historical conflict cache boundary",
556            });
557        }
558        if historical_pre_state.account_id() != self.account_id
559            || historical_pre_state.genesis_anchor() != self.genesis_anchor
560        {
561            return Err(IdentityError::AccountMismatch);
562        }
563        if accepted_event.body().sequence() != sequence
564            || accepted_event.body().predecessors() != incoming_event.body().predecessors()
565            || accepted_event.event_id()? == incoming_event.event_id()?
566        {
567            return Err(IdentityError::InvalidRelationship {
568                resource: "historical conflict event pair",
569            });
570        }
571        let accepted_expected_epoch =
572            expected_epoch(historical_pre_state, accepted_event.body().operation())?;
573        crate::verifier::validate_event(
574            historical_pre_state,
575            historical_pre_state,
576            accepted_event,
577            accepted_expected_epoch,
578        )?;
579        let incoming_epoch =
580            expected_epoch(historical_pre_state, incoming_event.body().operation())?;
581        crate::verifier::validate_event(
582            historical_pre_state,
583            historical_pre_state,
584            incoming_event,
585            incoming_epoch,
586        )?;
587
588        let mut accepted_projection = historical_pre_state.detached_snapshot();
589        let mut accepted_transitions = Vec::new();
590        let mut accepted_path_bytes = 0_usize;
591        for accepted in accepted_path {
592            let pre_state = accepted_projection.detached_snapshot();
593            let accepted_epoch = expected_epoch(&pre_state, accepted.body().operation())?;
594            crate::verifier::validate_event(&pre_state, &pre_state, accepted, accepted_epoch)?;
595            accepted_projection.apply_new_linear(accepted, accepted.event_id()?)?;
596            let transition = LineageEntry {
597                pre_state: Box::new(pre_state),
598                authority_state: None,
599                expected_epoch: accepted_epoch,
600                event: accepted.clone(),
601            };
602            checked_evidence_add(
603                &mut accepted_path_bytes,
604                lineage_entry_evidence_bytes(&transition)?,
605            )?;
606            if accepted_path_bytes > MAX_FORK_EVIDENCE_BYTES {
607                return Err(IdentityError::limit(
608                    "account fork evidence bytes",
609                    accepted_path_bytes,
610                    MAX_FORK_EVIDENCE_BYTES,
611                ));
612            }
613            accepted_transitions.push(transition);
614        }
615        let mut authenticated_tip = accepted_projection.detached_snapshot();
616        authenticated_tip.historical_state_required_through = None;
617        let mut current_tip = self.detached_snapshot();
618        current_tip.historical_state_required_through = None;
619        if authenticated_tip != current_tip {
620            return Err(IdentityError::StorageCorruption);
621        }
622
623        let mut staged = self.clone();
624        let outcome = staged.open_fork(
625            incoming_event,
626            incoming_event.event_id()?,
627            LineageEntry {
628                pre_state: Box::new(historical_pre_state.detached_snapshot()),
629                authority_state: None,
630                expected_epoch: accepted_expected_epoch,
631                event: accepted_event.clone(),
632            },
633            Some(accepted_transitions),
634        )?;
635        *self = staged;
636        Ok(outcome)
637    }

722    /// Domain-separated commitment to the complete projected cryptographic state.
723    pub(crate) fn crypto_state_id(&self) -> Result<crate::CryptoStateId, IdentityError> {
724        let material = match &self.crypto {
725            CryptoProjection::Stable(stable) => {
726                CryptoStateMaterial::Stable(stable_crypto_state_material(stable)?)
727            }
728            CryptoProjection::Candidate {
729                previous,
730                migration,
731                begin_event_id,
732            } => CryptoStateMaterial::Candidate {
733                previous: stable_crypto_state_material(previous)?,
734                migration_id: migration.crypto_migration_id()?,
735                candidate_suite_id: migration.to_suite().crypto_suite_id()?,
736                begin_event_id: *begin_event_id,
737            },
738            CryptoProjection::Dual {
739                previous,
740                migration,
741                begin_event_id,
742                activation_event_id,
743            } => CryptoStateMaterial::Dual {
744                previous: stable_crypto_state_material(previous)?,
745                migration_id: migration.crypto_migration_id()?,
746                candidate_suite_id: migration.to_suite().crypto_suite_id()?,
747                begin_event_id: *begin_event_id,
748                activation_event_id: *activation_event_id,
749            },
750        };
751        let bytes = crate::codec::encode_wire(&material)?;
752        Ok(crate::CryptoStateId::from_digest(crate::types::hash_bytes(
753            crate::types::HashDomain::CryptoState,
754            &bytes,
755        )))
756    }

814    fn validate_and_apply_inner(
815        &mut self,
816        event: &AuthorizedEvent,
817    ) -> Result<ApplyOutcome, IdentityError> {
818        let event_id = event.event_id()?;
819        if let Some(outcome) = self.try_replay(event, event_id)? {
820            return Ok(outcome);
821        }
822        if self.fork.is_some() {
823            return self.apply_while_forked(event, event_id);
824        }
825        if let Some(conflict) = self.find_lineage_conflict(event, event_id)? {
826            return self.open_fork(event, event_id, conflict, None);
827        }
828        if event.body().account_id() == self.account_id
829            && self
830                .historical_state_required_through
831                .is_some_and(|through| event.body().sequence() <= through)
832        {
833            return Err(IdentityError::HistoricalStateRequired {
834                sequence: event.body().sequence().get(),
835            });
836        }
837        self.apply_new_linear(event, event_id)
838    }

840    fn try_replay(
841        &mut self,
842        event: &AuthorizedEvent,
843        event_id: EventId,
844    ) -> Result<Option<ApplyOutcome>, IdentityError> {
845        if let Some(index) = self
846            .lineage
847            .iter()
848            .position(|entry| entry.event.event_id() == Ok(event_id))
849        {
850            let changed = {
851                let entry = &mut self.lineage[index];
852                crate::verifier::validate_event(
853                    &entry.pre_state,
854                    entry.authority_state.as_deref().unwrap_or(&entry.pre_state),
855                    event,
856                    entry.expected_epoch,
857                )?;
858                let merged = merge_event_evidence(&entry.event, event)?;
859                if merged != entry.event {
860                    entry.event = merged;
861                    true
862                } else {
863                    false
864                }
865            };
866            let disposition = if changed {
867                self.normalize_lineage_bound()?;
868                ApplyDisposition::ApprovalsMerged
869            } else {
870                ApplyDisposition::Replay
871            };
872            return Ok(Some(ApplyOutcome {
873                disposition,
874                event_id,
875                effects: Vec::new(),
876            }));
877        }
878
879        let fork_transitions = self.fork.as_ref().map_or_else(Vec::new, |fork| {
880            fork.branches
881                .iter()
882                .enumerate()
883                .flat_map(|(branch_index, branch)| {
884                    branch.transitions.iter().enumerate().filter_map(
885                        move |(transition_index, transition)| {
886                            (transition.event.event_id() == Ok(event_id))
887                                .then_some((branch_index, transition_index))
888                        },
889                    )
890                })
891                .collect::<Vec<_>>()
892        });
893        if !fork_transitions.is_empty() {
894            let fork = self.fork.as_ref().ok_or(IdentityError::StorageCorruption)?;
895            let mut merged = event.clone();
896            for (branch_index, transition_index) in &fork_transitions {
897                let transition = &fork.branches[*branch_index].transitions[*transition_index];
898                crate::verifier::validate_event(
899                    &transition.pre_state,
900                    transition
901                        .authority_state
902                        .as_deref()
903                        .unwrap_or(&transition.pre_state),
904                    event,
905                    transition.expected_epoch,
906                )?;
907                merged = merge_event_evidence(&transition.event, &merged)?;
908            }
909            let fork = self.fork.as_mut().ok_or(IdentityError::StorageCorruption)?;
910            let changed = fork_transitions
911                .iter()
912                .any(|(branch_index, transition_index)| {
913                    fork.branches[*branch_index].transitions[*transition_index].event != merged
914                });
915            let disposition = if changed {
916                for (branch_index, transition_index) in fork_transitions {
917                    fork.branches[branch_index].transitions[transition_index].event =
918                        merged.clone();
919                }
920                validate_fork_evidence_bound(&fork.common_state, &fork.branches)?;
921                ApplyDisposition::ApprovalsMerged
922            } else {
923                ApplyDisposition::Replay
924            };
925            return Ok(Some(ApplyOutcome {
926                disposition,
927                event_id,
928                effects: Vec::new(),
929            }));
930        }
931        Ok(None)
932    }

951    fn apply_new_linear(
952        &mut self,
953        event: &AuthorizedEvent,
954        event_id: EventId,
955    ) -> Result<ApplyOutcome, IdentityError> {
956        self.validate_lifecycle_gate(event.body().operation())?;
957        let expected_epoch = expected_epoch(self, event.body().operation())?;
958        let validated = crate::verifier::validate_event(self, self, event, expected_epoch)?;
959
960        let pre_state = self.detached_snapshot();
961        self.apply_operation(event, event_id, validated.provider_authority_time())?;
962        self.sequence = event.body().sequence();
963        self.epoch = event.body().resulting_epoch();
964        self.heads.clear();
965        self.heads.push(event_id);
966        self.retain_lineage_entry(LineageEntry {
967            pre_state: Box::new(pre_state.clone()),
968            authority_state: None,
969            expected_epoch,
970            event: event.clone(),
971        })?;
972        self.fork = None;
973
974        Ok(ApplyOutcome {
975            disposition: ApplyDisposition::Applied,
976            event_id,
977            effects: transition_effects(
978                event_id,
979                self.epoch,
980                operation_changes_epoch(event.body().operation()),
981            ),
982        })
983    }

985    fn open_fork(
986        &mut self,
987        event: &AuthorizedEvent,
988        event_id: EventId,
989        conflict: LineageEntry,
990        authenticated_left_path: Option<Vec<LineageEntry>>,
991    ) -> Result<ApplyOutcome, IdentityError> {
992        if conflict.pre_state.fork.is_some()
993            && matches!(event.body().operation(), AccountOperation::ResolveFork(_))
994        {
995            return self.open_resolution_fork(event, event_id, &conflict);
996        }
997        let common_state = conflict.pre_state.detached_snapshot();
998        let expected_epoch = expected_epoch(&common_state, event.body().operation())?;
999        crate::verifier::validate_event(&common_state, &common_state, event, expected_epoch)?;
1000
1001        let left_state = self.detached_snapshot();
1002        let conflict_id = conflict.event.event_id()?;
1003        let left_start = self
1004            .lineage
1005            .iter()
1006            .position(|transition| transition.event.event_id() == Ok(conflict_id));
1007        let left_transitions = match authenticated_left_path {
1008            Some(path) => path,
1009            None => {
1010                let index = left_start.ok_or(IdentityError::HistoricalStateRequired {
1011                    sequence: conflict.event.body().sequence().get(),
1012                })?;
1013                self.lineage[index..].to_vec()
1014            }
1015        };
1016        let mut right_state = common_state.detached_snapshot();
1017        right_state.apply_new_linear(event, event_id)?;
1018        let right_transition = right_state
1019            .lineage
1020            .last()
1021            .ok_or(IdentityError::StorageCorruption)?;
1022        let right_state = right_state.detached_snapshot();
1023        let mut branches = vec![
1024            ForkBranch {
1025                transitions: left_transitions,
1026                projected_state: left_state,
1027            },
1028            ForkBranch {
1029                transitions: vec![LineageEntry {
1030                    pre_state: right_transition.pre_state.clone(),
1031                    authority_state: right_transition.authority_state.clone(),
1032                    expected_epoch: right_transition.expected_epoch,
1033                    event: right_transition.event.clone(),
1034                }],
1035                projected_state: right_state,
1036            },
1037        ];
1038        sort_fork_branches(&mut branches)?;
1039        validate_fork_evidence_bound(&common_state, &branches)?;
1040
1041        *self = common_state.detached_snapshot();
1042        self.sequence = branches
1043            .iter()
1044            .map(|branch| branch.projected_state.sequence())
1045            .max()
1046            .ok_or(IdentityError::StorageCorruption)?;
1047        self.heads = fork_head_ids(&branches)?;
1048        self.lifecycle = ProjectionLifecycle::Forked;
1049        let common_ancestor = if common_state.sequence() == Sequence::GENESIS {
1050            ForkCommonAncestor::Genesis(common_state.genesis_anchor())
1051        } else {
1052            let [ancestor] = common_state.heads() else {
1053                return Err(IdentityError::StorageCorruption);
1054            };
1055            ForkCommonAncestor::Event(*ancestor)
1056        };
1057        self.fork = Some(ForkProjection {
1058            common_state: Box::new(common_state),
1059            common_ancestor,
1060            conflict_sequence: event.body().sequence(),
1061            conflict_predecessors: event.body().predecessors().clone(),
1062            branches,
1063        });
1064
1065        Ok(ApplyOutcome {
1066            disposition: ApplyDisposition::ForkDetected,
1067            event_id,
1068            effects: vec![
1069                ProjectionEffect::PublishAccountEvent { event_id },
1070                ProjectionEffect::NotifyForkDetected { event_id },
1071            ],
1072        })
1073    }

1099            .last()
1100            .ok_or(IdentityError::StorageCorruption)?;

1125            .ok_or(IdentityError::StorageCorruption)?;

1127        self.lifecycle = ProjectionLifecycle::Forked;
1128        self.fork = Some(ForkProjection {

1145    fn apply_while_forked(
1146        &mut self,
1147        event: &AuthorizedEvent,
1148        event_id: EventId,
1149    ) -> Result<ApplyOutcome, IdentityError> {
1150        if matches!(event.body().operation(), AccountOperation::ResolveFork(_))
1151            && event.body().predecessors().event_heads() == Some(self.heads())
1152        {
1153            return self.apply_fork_resolution(event, event_id);
1154        }
1155
1156        let mut fork = self
1157            .fork
1158            .as_ref()
1159            .ok_or(IdentityError::StorageCorruption)?
1160            .clone();
1161        if fork.branches.len() < 2 {
1162            return Err(IdentityError::StorageCorruption);
1163        }
1164
1165        let parent_index = fork.branches.iter().position(|branch| {
1166            branch.projected_state.sequence().checked_next().ok() == Some(event.body().sequence())
1167                && event.body().predecessors().event_heads() == Some(branch.projected_state.heads())
1168        });
1169        if let Some(parent_index) = parent_index {
1170            let (transition, projected_state) = project_transition(
1171                &fork.branches[parent_index].projected_state,
1172                event,
1173                event_id,
1174            )?;
1175            fork.branches[parent_index].transitions.push(transition);
1176            fork.branches[parent_index].projected_state = projected_state;
1177        } else {
1178            let conflict = fork
1179                .branches
1180                .iter()
1181                .enumerate()
1182                .find_map(|(branch_index, branch)| {
1183                    branch
1184                        .transitions
1185                        .iter()
1186                        .enumerate()
1187                        .find(|(_, transition)| {
1188                            transition.event.body().sequence() == event.body().sequence()
1189                                && transition.event.body().predecessors()
1190                                    == event.body().predecessors()
1191                        })
1192                        .map(|(transition_index, transition)| {
1193                            (branch_index, transition_index, transition.clone())
1194                        })
1195                })
1196                .ok_or(IdentityError::AccountForked)?;
1197            if fork.branches.len() >= MAX_FORK_HEADS {
1198                return Err(IdentityError::limit(
1199                    "account fork heads",
1200                    fork.branches.len().saturating_add(1),
1201                    MAX_FORK_HEADS,
1202                ));
1203            }
1204            let (branch_index, transition_index, conflicting_transition) = conflict;
1205            let (incoming_transition, projected_state) =
1206                project_transition(&conflicting_transition.pre_state, event, event_id)?;
1207            let mut transitions =
1208                fork.branches[branch_index].transitions[..transition_index].to_vec();
1209            transitions.push(incoming_transition);
1210            fork.branches.push(ForkBranch {
1211                transitions,
1212                projected_state,
1213            });
1214        }
1215        sort_fork_branches(&mut fork.branches)?;
1216        validate_fork_evidence_bound(&fork.common_state, &fork.branches)?;
1217        self.sequence = fork
1218            .branches
1219            .iter()
1220            .map(|branch| branch.projected_state.sequence())
1221            .max()
1222            .ok_or(IdentityError::StorageCorruption)?;
1223        self.heads = fork_head_ids(&fork.branches)?;
1224        self.fork = Some(fork);
1225        Ok(ApplyOutcome {
1226            disposition: ApplyDisposition::ForkDetected,
1227            event_id,
1228            effects: vec![
1229                ProjectionEffect::PublishAccountEvent { event_id },
1230                ProjectionEffect::NotifyForkDetected { event_id },
1231            ],
1232        })
1233    }

1243    fn retain_lineage_entry(&mut self, entry: LineageEntry) -> Result<(), IdentityError> {
1244        let entry_bytes = lineage_entry_evidence_bytes(&entry)?;
1245        while !self.lineage.is_empty()
1246            && (self.lineage.len() >= MAX_HISTORY_PAGE_EVENTS
1247                || self.lineage_bytes.checked_add(entry_bytes).ok_or(
1248                    IdentityError::ArithmeticOverflow {
1249                        resource: "account lineage bytes",
1250                    },
1251                )? > MAX_HISTORY_PAGE_BYTES)
1252        {
1253            let removed = self.lineage.remove(0);
1254            let removed_bytes = lineage_entry_evidence_bytes(&removed)?;
1255            self.lineage_bytes = self
1256                .lineage_bytes
1257                .checked_sub(removed_bytes)
1258                .ok_or(IdentityError::StorageCorruption)?;
1259            self.historical_state_required_through = Some(removed.event.body().sequence());
1260        }
1261        if entry_bytes > MAX_HISTORY_PAGE_BYTES {
1262            self.historical_state_required_through = Some(entry.event.body().sequence());
1263            return Ok(());
1264        }
1265        self.lineage_bytes = self.lineage_bytes.checked_add(entry_bytes).ok_or(
1266            IdentityError::ArithmeticOverflow {
1267                resource: "account lineage bytes",
1268            },
1269        )?;
1270        self.lineage.push(entry);
1271        Ok(())
1272    }

1283    fn validate_lifecycle_gate(&self, operation: &AccountOperation) -> Result<(), IdentityError> {
1284        match self.lifecycle {
1285            ProjectionLifecycle::Active => Ok(()),
1286            ProjectionLifecycle::RecoveryPending
1287                if matches!(
1288                    operation,
1289                    AccountOperation::VetoRecovery(_)
1290                        | AccountOperation::CancelRecovery(_)
1291                        | AccountOperation::FinalizeRecovery(_)
1292                ) =>
1293            {
1294                Ok(())
1295            }
1296            ProjectionLifecycle::RecoveryPending => Err(IdentityError::RecoveryPending),
1297            ProjectionLifecycle::MigrationPending
1298                if matches!(
1299                    operation,
1300                    AccountOperation::ActivateCryptoMigration(_)
1301                        | AccountOperation::RetireCryptoSuite(_)
1302                ) =>
1303            {
1304                Ok(())
1305            }
1306            ProjectionLifecycle::MigrationDual
1307                if matches!(operation, AccountOperation::RetireCryptoSuite(_)) =>
1308            {
1309                Ok(())
1310            }
1311            ProjectionLifecycle::MigrationPending | ProjectionLifecycle::MigrationDual => {
1312                Err(IdentityError::InvalidRelationship {
1313                    resource: "cryptographic migration phase operation",
1314                })
1315            }
1316            ProjectionLifecycle::UpgradePending => Err(IdentityError::UnsupportedVersion {
1317                version: self.protocol_major.get(),
1318            }),
1319            ProjectionLifecycle::Retired => Err(IdentityError::AccountRetired),
1320            ProjectionLifecycle::Forked => Err(IdentityError::AccountForked),
1321        }
1322    }

1324    fn apply_operation(
1325        &mut self,
1326        event: &AuthorizedEvent,
1327        event_id: EventId,
1328        provider_authority_time: Option<Timestamp>,
1329    ) -> Result<(), IdentityError> {
1330        match event.body().operation() {
1331            AccountOperation::AuthorizeDevice(authorization) => {
1332                self.authorize_device(authorization, event.body().resulting_epoch())
1333            }
1334            AccountOperation::UpdateDeviceAuthorization(update) => {
1335                self.update_device_authorization(update, event.body().resulting_epoch())
1336            }
1337            AccountOperation::UpdateDeviceMetadata(update) => {
1338                let device = self.device_mut(update.device_id())?;
1339                device.metadata_commitment = update.metadata_commitment();
1340                Ok(())
1341            }
1342            AccountOperation::SuspendDevice(suspend) => {
1343                let device = self.device_mut(suspend.device_id())?;
1344                match device.lifecycle {
1345                    ProjectedDeviceLifecycle::Active => {
1346                        device.lifecycle = ProjectedDeviceLifecycle::Suspended;
1347                        Ok(())
1348                    }
1349                    ProjectedDeviceLifecycle::Suspended => Err(IdentityError::DeviceSuspended),
1350                    ProjectedDeviceLifecycle::Revoked => Err(IdentityError::DeviceRevoked),
1351                }
1352            }
1353            AccountOperation::ReinstateDevice(reinstate) => {
1354                let device = self.device_mut(reinstate.device_id())?;
1355                match device.lifecycle {
1356                    ProjectedDeviceLifecycle::Suspended => {
1357                        device.lifecycle = ProjectedDeviceLifecycle::Active;
1358                        Ok(())
1359                    }
1360                    ProjectedDeviceLifecycle::Active => Err(IdentityError::InvalidRelationship {
1361                        resource: "reinstate active device",
1362                    }),
1363                    ProjectedDeviceLifecycle::Revoked => Err(IdentityError::DeviceRevoked),
1364                }
1365            }
1366            AccountOperation::RevokeDevice(revoke) => {
1367                let device = self.device_mut(revoke.device_id())?;
1368                if device.lifecycle == ProjectedDeviceLifecycle::Revoked {
1369                    return Err(IdentityError::DeviceRevoked);
1370                }
1371                device.lifecycle = ProjectedDeviceLifecycle::Revoked;
1372                Ok(())
1373            }
1374            AccountOperation::RotateDeviceKeys(rotation) => {
1375                let old = self.device_mut(rotation.old_device_id())?;
1376                if old.lifecycle == ProjectedDeviceLifecycle::Revoked {
1377                    return Err(IdentityError::DeviceRevoked);
1378                }
1379                old.lifecycle = ProjectedDeviceLifecycle::Revoked;
1380                self.authorize_device(rotation.new_authorization(), event.body().resulting_epoch())
1381            }
1382            AccountOperation::AddController(descriptor) => self.add_controller(descriptor),
1383            AccountOperation::RemoveController(controller_id) => {
1384                self.remove_controller(*controller_id)
1385            }
1386            AccountOperation::ChangeControlPolicy(policy) => {
1387                policy.validate_satisfiable(&self.active_controller_descriptors())?;
1388                self.control_policy_id = policy.id()?;
1389                self.control_policy = policy.clone();
1390                Ok(())
1391            }
1392            AccountOperation::ChangeRecoveryPolicy(policy) => {
1393                if policy.policy_version()
1394                    != self.recovery_policy.policy_version().checked_next()?
1395                {
1396                    return Err(IdentityError::PolicyVersionMismatch);
1397                }
1398                policy.validate_controller_authority(&self.active_controller_descriptors())?;
1399                self.recovery_policy_id = policy.id()?;
1400                self.recovery_policy = policy.clone();
1401                Ok(())
1402            }
1403            AccountOperation::ChangeProviderPolicy(policy) => {
1404                if policy.policy_version()
1405                    != self.provider_policy.policy_version().checked_next()?
1406                {
1407                    return Err(IdentityError::PolicyVersionMismatch);
1408                }
1409                self.provider_policy_id = policy.id()?;
1410                self.provider_policy = policy.clone();
1411                Ok(())
1412            }
1413            AccountOperation::BeginRecovery(begin) => self.begin_recovery(
1414                begin,
1415                event.admission_evidence(),
1416                event_id,
1417                event.body().proposal_id()?,
1418            ),
1419            AccountOperation::VetoRecovery(veto) => self.veto_recovery(veto),
1420            AccountOperation::CancelRecovery(cancel) => self.cancel_recovery(cancel),
1421            AccountOperation::FinalizeRecovery(finalize) => {
1422                self.finalize_recovery(finalize, provider_authority_time)
1423            }
1424            AccountOperation::ResolveFork(_) => Err(IdentityError::InvalidPredecessor),
1425            AccountOperation::BeginCryptoMigration(begin) => {
1426                self.begin_crypto_migration(begin, event_id)
1427            }
1428            AccountOperation::ActivateCryptoMigration(activate) => {
1429                self.activate_crypto_migration(activate, event_id)
1430            }
1431            AccountOperation::RetireCryptoSuite(retire) => {
1432                self.retire_crypto_suite(retire, event.body().resulting_epoch())
1433            }
1434            AccountOperation::UpgradeProtocol(upgrade) => {
1435                if upgrade.from_major() != self.protocol_major {
1436                    return Err(IdentityError::UnsupportedVersion {
1437                        version: upgrade.from_major().get(),
1438                    });
1439                }
1440                self.protocol_major = upgrade.to_major();
1441                self.upgrade = Some(upgrade.clone());
1442                self.lifecycle = ProjectionLifecycle::UpgradePending;
1443                Ok(())
1444            }
1445            AccountOperation::RetireAccount(retire) => {
1446                self.retirement = Some(RetirementProjection::Account(retire.clone()));
1447                self.lifecycle = ProjectionLifecycle::Retired;
1448                Ok(())
1449            }
1450        }
1451    }

1453    fn apply_fork_resolution(
1454        &mut self,
1455        event: &AuthorizedEvent,
1456        event_id: EventId,
1457    ) -> Result<ApplyOutcome, IdentityError> {
1458        let AccountOperation::ResolveFork(resolution) = event.body().operation() else {
1459            return Err(IdentityError::AccountForked);
1460        };
1461        let fork = self
1462            .fork
1463            .as_ref()
1464            .ok_or(IdentityError::StorageCorruption)?
1465            .clone();
1466        if resolution.fork().account_id() != self.account_id {
1467            return Err(IdentityError::AccountMismatch);
1468        }
1469        if resolution.fork().heads() != self.heads
1470            || resolution.fork().common_ancestor() != fork.common_ancestor
1471        {
1472            return Err(IdentityError::InvalidPredecessor);
1473        }
1474        let selected = fork
1475            .branches
1476            .iter()
1477            .find(|branch| fork_branch_event_id(branch) == Ok(resolution.selected_head()))
1478            .ok_or(IdentityError::InvalidPredecessor)?;
1479        let maximum_branch_epoch = fork
1480            .branches
1481            .iter()
1482            .map(|branch| branch.projected_state.epoch())
1483            .max()
1484            .ok_or(IdentityError::StorageCorruption)?;
1485        let expected_epoch = maximum_branch_epoch.checked_next()?;
1486        crate::verifier::validate_event(self, &fork.common_state, event, expected_epoch)?;
1487
1488        let mut pre_state = self.clone();
1489        pre_state.lineage.clear();
1490        pre_state.lineage_bytes = 0;
1491        let mut resolved = selected.projected_state.detached_snapshot();
1492        for controller_id in resolution.revoked_controllers() {
1493            resolved.revoke_controller_for_resolution(*controller_id)?;
1494        }
1495        for device_id in resolution.revoked_devices() {
1496            let device = resolved.device_mut(*device_id)?;
1497            device.lifecycle = ProjectedDeviceLifecycle::Revoked;
1498        }
1499        if resolved.active_controllers.is_empty() {
1500            return Err(IdentityError::UnsatisfiableThreshold);
1501        }
1502        let descriptors = resolved.active_controller_descriptors();
1503        resolved.control_policy.validate_satisfiable(&descriptors)?;
1504        resolved
1505            .recovery_policy
1506            .validate_controller_authority(&descriptors)?;
1507        resolved.sequence = event.body().sequence();
1508        resolved.epoch = event.body().resulting_epoch();
1509        resolved.heads = vec![event_id];
1510        resolved.retain_lineage_entry(LineageEntry {
1511            pre_state: Box::new(pre_state),
1512            authority_state: Some(Box::new(fork.common_state.detached_snapshot())),
1513            expected_epoch,
1514            event: event.clone(),
1515        })?;
1516        resolved.fork = None;
1517        *self = resolved;
1518
1519        Ok(ApplyOutcome {
1520            disposition: ApplyDisposition::Applied,
1521            event_id,
1522            effects: transition_effects(event_id, self.epoch, true),
1523        })
1524    }

1527        &mut self,
1528        controller_id: ControllerId,
1529    ) -> Result<(), IdentityError> {
1530        if self.revoked_controller(controller_id).is_some() {

1532        }

1534            .active_controllers
1535            .binary_search_by_key(&controller_id, ProjectedController::id)
1536            .map_err(|_| IdentityError::UnknownController)?;

1541        {
1542            Ok(_) => return Err(IdentityError::StorageCorruption),

1544        };

1549    fn authorize_device(
1550        &mut self,
1551        authorization: &crate::DeviceAuthorization,
1552        resulting_epoch: Epoch,
1553    ) -> Result<(), IdentityError> {
1554        if authorization.authorization_epoch() != resulting_epoch {
1555            return Err(IdentityError::InvalidEpoch);
1556        }
1557        let device_id = authorization.device_id();
1558        match self
1559            .devices
1560            .binary_search_by_key(&device_id, ProjectedDevice::id)
1561        {
1562            Ok(index) if self.devices[index].lifecycle == ProjectedDeviceLifecycle::Revoked => {
1563                return Err(IdentityError::DeviceRevoked);
1564            }
1565            Ok(_) => {
1566                return Err(IdentityError::InvalidRelationship {
1567                    resource: "duplicate active device authorization",
1568                });
1569            }
1570            Err(index) => {
1571                self.validate_new_device_descriptor(authorization.descriptor())?;
1572                if self.devices.len() >= MAX_DEVICES {
1573                    return Err(IdentityError::limit(
1574                        "projected devices",
1575                        self.devices.len().saturating_add(1),
1576                        MAX_DEVICES,
1577                    ));
1578                }
1579                self.devices.insert(
1580                    index,
1581                    ProjectedDevice {
1582                        id: device_id,
1583                        descriptor: authorization.descriptor().clone(),
1584                        device_class: authorization.device_class(),
1585                        metadata_commitment: authorization.metadata_commitment(),
1586                        capabilities: authorization.capabilities().to_vec(),
1587                        authorization_epoch: authorization.authorization_epoch(),
1588                        lifecycle: ProjectedDeviceLifecycle::Active,
1589                    },
1590                );
1591            }
1592        }
1593        Ok(())
1594    }

1596    fn validate_new_device_descriptor(
1597        &self,
1598        descriptor: &crate::DeviceDescriptor,
1599    ) -> Result<(), IdentityError> {
1600        if self
1601            .devices
1602            .iter()
1603            .any(|device| device_descriptors_reuse_key(device.descriptor(), descriptor))
1604        {
1605            return Err(IdentityError::InvalidRelationship {
1606                resource: "retained device public-key reuse",
1607            });
1608        }
1609        if self
1610            .active_controllers
1611            .iter()
1612            .chain(&self.revoked_controllers)
1613            .any(|controller| {
1614                device_descriptor_reuses_controller_key(descriptor, controller.signing_key())
1615            })
1616        {
1617            return Err(IdentityError::InvalidRelationship {
1618                resource: "controller/device public-key role separation",
1619            });
1620        }
1621        let application_key = descriptor.application_signing_key();
1622        let agreement_key = descriptor.agreement_key();
1623        let endpoint_key = descriptor.endpoint_key().as_signing_key();
1624        let descriptor_keys = [
1625            application_key.as_bytes().as_slice(),
1626            agreement_key.as_bytes().as_slice(),
1627            endpoint_key.as_bytes().as_slice(),
1628        ];
1629        if descriptor_keys
1630            .iter()
1631            .any(|key| self.crypto_retains_key_material(key))
1632        {
1633            return Err(IdentityError::InvalidRelationship {
1634                resource: "device/cryptographic key tombstone separation",
1635            });
1636        }
1637        Ok(())
1638    }

1666    fn add_controller(&mut self, descriptor: &ControllerDescriptor) -> Result<(), IdentityError> {
1667        let controller_id = descriptor.id()?;
1668        if self.crypto_retains_key_material(descriptor.signing_key().as_bytes()) {
1669            return Err(IdentityError::DuplicateSigningKey);
1670        }
1671        let migrated_key = match &self.crypto {
1672            CryptoProjection::Stable(stable)
1673                if stable.suite != CryptoSuiteDescriptor::v1()?
1674                    || !stable.migrated_keys.is_empty() =>
1675            {
1676                let algorithm_code = stable.suite.signature_algorithm_code();
1677                if algorithm_code != crate::SignatureAlgorithm::Ed25519.code() {
1678                    return Err(IdentityError::UnsupportedPolicyFeature {
1679                        feature: "post-migration controller enrollment for non-Ed25519 suites",
1680                    });
1681                }
1682                let key = AlgorithmPublicKey::new(
1683                    algorithm_code,
1684                    descriptor.signing_key().as_bytes().to_vec(),
1685                )?;
1686                if stable.migrated_keys.iter().any(|retained| {
1687                    retained.key.algorithm_code() == key.algorithm_code()
1688                        && retained.key.as_bytes() == key.as_bytes()
1689                }) {
1690                    return Err(IdentityError::DuplicateSigningKey);
1691                }
1692                Some(key)
1693            }
1694            CryptoProjection::Stable(_) => None,
1695            CryptoProjection::Candidate { .. } | CryptoProjection::Dual { .. } => {
1696                return Err(IdentityError::InvalidRelationship {
1697                    resource: "controller enrollment cryptographic migration phase",
1698                });
1699            }
1700        };
1701        if self.revoked_controller(controller_id).is_some() {
1702            return Err(IdentityError::RevokedController);
1703        }
1704        if self.active_controller(controller_id).is_some() {
1705            return Err(IdentityError::InvalidRelationship {
1706                resource: "duplicate active controller",
1707            });
1708        }
1709        if self
1710            .active_controllers
1711            .iter()
1712            .chain(&self.revoked_controllers)
1713            .any(|controller| controller.signing_key() == descriptor.signing_key())
1714        {
1715            return Err(IdentityError::DuplicateSigningKey);
1716        }
1717        if self.devices.iter().any(|device| {
1718            device_descriptor_reuses_controller_key(device.descriptor(), descriptor.signing_key())
1719        }) {
1720            return Err(IdentityError::InvalidRelationship {
1721                resource: "controller/device public-key role separation",
1722            });
1723        }
1724        let retained_count = self
1725            .active_controllers
1726            .len()
1727            .checked_add(self.revoked_controllers.len())
1728            .and_then(|value| value.checked_add(1))
1729            .ok_or(IdentityError::ArithmeticOverflow {
1730                resource: "projected controller count",
1731            })?;
1732        if retained_count > MAX_CONTROLLERS {
1733            return Err(IdentityError::limit(
1734                "projected controllers",
1735                retained_count,
1736                MAX_CONTROLLERS,
1737            ));
1738        }
1739        let index = match self
1740            .active_controllers
1741            .binary_search_by_key(&controller_id, ProjectedController::id)
1742        {
1743            Ok(_) => return Err(IdentityError::StorageCorruption),
1744            Err(index) => index,
1745        };
1746        self.active_controllers.insert(
1747            index,
1748            ProjectedController {
1749                id: controller_id,
1750                descriptor: descriptor.clone(),
1751            },
1752        );
1753        if let Some(key) = migrated_key {
1754            let CryptoProjection::Stable(stable) = &mut self.crypto else {
1755                return Err(IdentityError::StorageCorruption);
1756            };
1757            let key_index = match stable
1758                .migrated_keys
1759                .binary_search_by_key(&controller_id, |retained| retained.controller_id)
1760            {
1761                Ok(_) => return Err(IdentityError::StorageCorruption),
1762                Err(key_index) => key_index,
1763            };
1764            stable
1765                .migrated_keys
1766                .insert(key_index, MigrationKey { controller_id, key });
1767        }
1768        Ok(())
1769    }

1771    fn crypto_retains_key_material(&self, candidate: &[u8]) -> bool {
1772        let stable_reuses = |stable: &StableCrypto| {
1773            stable
1774                .migrated_keys
1775                .iter()
1776                .any(|retained| retained.key.as_bytes() == candidate)
1777                || stable
1778                    .key_tombstones
1779                    .iter()
1780                    .any(|retired| retired.as_bytes() == candidate)
1781        };
1782        match &self.crypto {
1783            CryptoProjection::Stable(stable) => stable_reuses(stable),
1784            CryptoProjection::Candidate {
1785                previous,
1786                migration,
1787                ..
1788            }
1789            | CryptoProjection::Dual {
1790                previous,
1791                migration,
1792                ..
1793            } => {
1794                stable_reuses(previous)
1795                    || migration
1796                        .bindings()
1797                        .iter()
1798                        .any(|binding| binding.new_signing_key().as_bytes() == candidate)
1799            }
1800        }
1801    }

1803    fn remove_controller(&mut self, controller_id: ControllerId) -> Result<(), IdentityError> {
1804        if self.revoked_controller(controller_id).is_some() {
1805            return Err(IdentityError::RevokedController);
1806        }
1807        let index = self
1808            .active_controllers
1809            .binary_search_by_key(&controller_id, ProjectedController::id)
1810            .map_err(|_| IdentityError::UnknownController)?;
1811        let removed = self.active_controllers.remove(index);
1812        if self.active_controllers.is_empty() {
1813            return Err(IdentityError::UnsatisfiableThreshold);
1814        }
1815        let descriptors = self.active_controller_descriptors();
1816        self.control_policy.validate_satisfiable(&descriptors)?;
1817        self.recovery_policy
1818            .validate_controller_authority(&descriptors)?;
1819        let revoked_index = match self
1820            .revoked_controllers
1821            .binary_search_by_key(&controller_id, ProjectedController::id)
1822        {
1823            Ok(_) => return Err(IdentityError::StorageCorruption),
1824            Err(index) => index,
1825        };
1826        self.revoked_controllers.insert(revoked_index, removed);
1827        // A migrated signing key remains retained under its revoked controller identifier.
1828        // This is a permanent key tombstone and also keeps the sorted active-key projection
1829        // stable for every controller that remains active.
1830        Ok(())
1831    }

1840    fn begin_recovery(
1841        &mut self,
1842        begin: &crate::BeginRecovery,
1843        admission_evidence: &AdmissionEvidence,
1844        event_id: EventId,
1845        begin_proposal_id: ProposalId,
1846    ) -> Result<(), IdentityError> {
1847        self.require_v1_recovery_crypto()?;
1848        if self.pending_recovery.is_some() || !begin.requires_vacant_recovery_slot() {
1849            return Err(IdentityError::RecoveryPending);
1850        }
1851        let plan = begin.proposal().plan();
1852        let current_head = self
1853            .heads
1854            .first()
1855            .copied()
1856            .ok_or(IdentityError::InvalidPredecessor)?;
1857        if plan.account_id() != self.account_id {
1858            return Err(IdentityError::AccountMismatch);
1859        }
1860        if plan.prior_event_head() != current_head {
1861            return Err(IdentityError::InvalidPredecessor);
1862        }
1863        if plan.recovery_policy_id() != self.recovery_policy_id
1864            || plan.recovery_policy_version() != self.recovery_policy.policy_version()
1865            || begin.threshold_evidence().recovery_policy_id() != self.recovery_policy_id
1866            || begin.threshold_evidence().recovery_policy_version()
1867                != self.recovery_policy.policy_version()
1868        {
1869            return Err(IdentityError::PolicyVersionMismatch);
1870        }
1871        self.pending_recovery = Some(PendingRecovery {
1872            recovery_id: begin.recovery_id(),
1873            proposal: begin.proposal().clone(),
1874            pre_recovery_control_policy_id: self.control_policy_id,
1875            begin_event_id: event_id,
1876            begin_proposal_id,
1877            begin_observation: PendingRecoveryObservation::from_admission(
1878                admission_evidence,
1879                &self.recovery_policy,
1880            )?,
1881        });
1882        self.lifecycle = ProjectionLifecycle::RecoveryPending;
1883        Ok(())
1884    }

1929    fn finalize_recovery(
1930        &mut self,
1931        finalize: &crate::FinalizeRecovery,
1932        provider_authority_time: Option<Timestamp>,
1933    ) -> Result<(), IdentityError> {
1934        self.require_v1_recovery_crypto()?;
1935        let pending = self
1936            .pending_recovery
1937            .as_ref()
1938            .ok_or(IdentityError::InvalidRelationship {
1939                resource: "finalize without pending recovery",
1940            })?
1941            .clone();
1942        if let Some(begin_transition) = self
1943            .lineage
1944            .iter()
1945            .find(|entry| entry.event.event_id() == Ok(pending.begin_event_id))
1946            && begin_transition
1947                .event
1948                .admission_evidence()
1949                .admission_evidence_id()?
1950                != pending.begin_observation.admission_evidence_id
1951        {
1952            return Err(IdentityError::StorageCorruption);
1953        }
1954        if finalize.expected_pending_recovery() != pending.recovery_id {
1955            return Err(IdentityError::InvalidRelationship {
1956                resource: "finalize pending recovery compare-and-set",
1957            });
1958        }
1959        let anchor = finalize.delay_anchor();
1960        if anchor.account_id() != self.account_id || anchor.recovery_id() != pending.recovery_id {
1961            return Err(IdentityError::AccountMismatch);
1962        }
1963        if anchor.begin_proposal_id() != pending.begin_proposal_id {
1964            return Err(IdentityError::InvalidRelationship {
1965                resource: "recovery begin proposal delay anchor",
1966            });
1967        }
1968        if pending.begin_observation.provider_policy_id != self.provider_policy_id {
1969            return Err(IdentityError::StorageCorruption);
1970        }
1971        if anchor.provider_policy_id() != self.provider_policy_id {
1972            return Err(IdentityError::PolicyVersionMismatch);
1973        }
1974        let replicated_provider_policy = match self.provider_policy.mode() {
1975            crate::ProviderMode::LocalOnly => return Err(IdentityError::FreshnessUnavailable),
1976            crate::ProviderMode::Replicated(policy) => policy,
1977        };
1978        let finalize_rule = self
1979            .control_policy
1980            .rule_for(crate::OperationKind::FinalizeRecovery)
1981            .ok_or(IdentityError::AuthorizationDenied)?;
1982        let freshness_quorum = match finalize_rule.freshness() {
1983            FreshnessRequirement::LatestKnown => 0,
1984            FreshnessRequirement::ProviderQuorum(requirement) => {
1985                usize::from(requirement.required().get())
1986            }
1987        };
1988        let minimum_required = usize::from(replicated_provider_policy.sufficient_threshold().get())
1989            .max(freshness_quorum);
1990        if usize::from(anchor.required_quorum().get()) < minimum_required {
1991            return Err(IdentityError::FreshnessUnavailable);
1992        }
1993        if !pending.begin_observation.matches_completion_anchor(anchor) {
1994            return Err(IdentityError::InvalidRelationship {
1995                resource: "recovery begin observation binding",
1996            });
1997        }
1998        let required = usize::from(anchor.required_quorum().get());
1999        let mut configured_receipts = Vec::new();
2000        for receipt in anchor.receipts().as_slice() {
2001            if receipt.entry().account_id() != self.account_id {
2002                return Err(IdentityError::AccountMismatch);
2003            }
2004            if receipt.entry().subject()
2005                != crate::ProviderLogSubject::EventIntent(pending.begin_proposal_id)
2006            {
2007                return Err(IdentityError::InvalidRelationship {
2008                    resource: "recovery delay receipt subject",
2009                });
2010            }
2011            let Some(provider) = crate::verifier::configured_provider(
2012                replicated_provider_policy.providers(),
2013                receipt.provider_id(),
2014            )?
2015            else {
2016                continue;
2017            };
2018            receipt.verify(provider)?;
2019            configured_receipts.push(receipt);
2020        }
2021        if configured_receipts.len() < required {
2022            return Err(IdentityError::FreshnessUnavailable);
2023        }
2024        let mut configured_observations = configured_receipts
2025            .iter()
2026            .map(|receipt| receipt.entry().observed_at())
2027            .collect::<Vec<_>>();
2028        configured_observations.sort_unstable();
2029        if configured_observations[required - 1] != anchor.observed_at() {
2030            return Err(IdentityError::InvalidRelationship {
2031                resource: "configured-provider recovery delay observation anchor",
2032            });
2033        }
2034        let delay_deadline = pending.begin_observation.delay_deadline;
2035        let mut completion_times = configured_receipts
2036            .iter()
2037            .map(|receipt| receipt.signed_head().body().observed_at())
2038            .filter(|observed_at| *observed_at >= delay_deadline)
2039            .collect::<Vec<_>>();
2040        if completion_times.len() < required {
2041            return Err(IdentityError::DelayNotElapsed);
2042        }
2043        completion_times.sort_unstable();
2044        let nested_authority_time = completion_times[required - 1];
2045        let provider_authority_time = provider_authority_time
2046            .map_or(nested_authority_time, |outer| {
2047                outer.max(nested_authority_time)
2048            });
2049        let lifetime_deadline = pending.begin_observation.lifetime_deadline;
2050        let plan = pending.proposal.plan();
2051        if provider_authority_time > lifetime_deadline
2052            || provider_authority_time > plan.expires_at()
2053        {
2054            return Err(IdentityError::StaleEvidence);
2055        }
2056
2057        self.install_recovery_plan(plan)?;
2058        self.pending_recovery = None;
2059        self.lifecycle = ProjectionLifecycle::Active;
2060        Ok(())
2061    }

2079    fn install_recovery_plan(
2080        &mut self,
2081        plan: &crate::RecoveryAuthorityPlan,
2082    ) -> Result<(), IdentityError> {
2083        for descriptor in plan.replacement_controllers() {
2084            let identifier = descriptor.id()?;
2085            if self.revoked_controller(identifier).is_some() {
2086                return Err(IdentityError::RevokedController);
2087            }
2088            if self.active_controllers.iter().any(|controller| {
2089                controller.id() != identifier
2090                    && controller.signing_key() == descriptor.signing_key()
2091            }) {
2092                return Err(IdentityError::DuplicateSigningKey);
2093            }
2094            if self
2095                .revoked_controllers
2096                .iter()
2097                .any(|controller| controller.signing_key() == descriptor.signing_key())
2098            {
2099                return Err(IdentityError::DuplicateSigningKey);
2100            }
2101            if self.devices.iter().any(|device| {
2102                device_descriptor_reuses_controller_key(
2103                    device.descriptor(),
2104                    descriptor.signing_key(),
2105                )
2106            }) {
2107                return Err(IdentityError::InvalidRelationship {
2108                    resource: "controller/device public-key role separation",
2109                });
2110            }
2111        }
2112
2113        let replacement_ids = plan
2114            .replacement_controllers()
2115            .iter()
2116            .map(ControllerDescriptor::id)
2117            .collect::<Result<Vec<_>, IdentityError>>()?;
2118        let removed = self
2119            .active_controllers
2120            .iter()
2121            .filter(|controller| replacement_ids.binary_search(&controller.id()).is_err())
2122            .cloned()
2123            .collect::<Vec<_>>();
2124        for controller in removed {
2125            let index = match self
2126                .revoked_controllers
2127                .binary_search_by_key(&controller.id(), ProjectedController::id)
2128            {
2129                Ok(_) => return Err(IdentityError::StorageCorruption),
2130                Err(index) => index,
2131            };
2132            self.revoked_controllers.insert(index, controller);
2133        }
2134        let retained_count = plan
2135            .replacement_controllers()
2136            .len()
2137            .checked_add(self.revoked_controllers.len())
2138            .ok_or(IdentityError::ArithmeticOverflow {
2139                resource: "recovery controller tombstones",
2140            })?;
2141        if retained_count > MAX_CONTROLLERS {
2142            return Err(IdentityError::limit(
2143                "recovery controller tombstones",
2144                retained_count,
2145                MAX_CONTROLLERS,
2146            ));
2147        }
2148        self.active_controllers = plan
2149            .replacement_controllers()
2150            .iter()
2151            .map(|descriptor| {
2152                Ok(ProjectedController {
2153                    id: descriptor.id()?,
2154                    descriptor: descriptor.clone(),
2155                })
2156            })
2157            .collect::<Result<Vec<_>, IdentityError>>()?;
2158
2159        for retained in plan.retained_devices() {
2160            let index = self
2161                .devices
2162                .binary_search_by_key(retained, ProjectedDevice::id)
2163                .map_err(|_| IdentityError::DeviceNotAuthorized)?;
2164            if self.devices[index].lifecycle == ProjectedDeviceLifecycle::Revoked {
2165                return Err(IdentityError::DeviceRevoked);
2166            }
2167        }
2168        for device in &mut self.devices {
2169            if device.lifecycle != ProjectedDeviceLifecycle::Revoked
2170                && plan.retained_devices().binary_search(&device.id()).is_err()
2171            {
2172                device.lifecycle = ProjectedDeviceLifecycle::Revoked;
2173            }
2174        }
2175
2176        self.control_policy = plan.replacement_control_policy().clone();
2177        self.control_policy_id = self.control_policy.id()?;
2178        self.recovery_policy = plan.replacement_recovery_policy().clone();
2179        self.recovery_policy_id = self.recovery_policy.id()?;
2180        Ok(())
2181    }

2183    fn begin_crypto_migration(
2184        &mut self,
2185        begin: &crate::BeginCryptoMigration,
2186        event_id: EventId,
2187    ) -> Result<(), IdentityError> {
2188        let CryptoProjection::Stable(previous) = &self.crypto else {
2189            return Err(IdentityError::InvalidRelationship {
2190                resource: "nested cryptographic migration",
2191            });
2192        };
2193        let previous = previous.clone();
2194        let migration = begin.migration();
2195        let migration_id = migration.crypto_migration_id()?;
2196        if migration.account_id() != self.account_id {
2197            return Err(IdentityError::AccountMismatch);
2198        }
2199        if migration.from_suite_id() != previous.suite.crypto_suite_id()? {
2200            return Err(IdentityError::InvalidRelationship {
2201                resource: "migration active suite",
2202            });
2203        }
2204        let candidate_suite_id = migration.to_suite().crypto_suite_id()?;
2205        if previous
2206            .retired_suites
2207            .iter()
2208            .any(|retired| retired.suite_id == candidate_suite_id)
2209        {
2210            return Err(IdentityError::InvalidRelationship {
2211                resource: "retired cryptographic suite reuse",
2212            });
2213        }
2214        let retiring_keys = stable_suite_keys(
2215            &previous,
2216            &self.active_controllers,
2217            &self.revoked_controllers,
2218        )?;
2219        let projected_tombstones = previous
2220            .key_tombstones
2221            .len()
2222            .checked_add(retiring_keys.len())
2223            .ok_or(IdentityError::ArithmeticOverflow {
2224                resource: "cryptographic key tombstones",
2225            })?;
2226        if projected_tombstones > MAX_HISTORY_PAGE_EVENTS {
2227            return Err(IdentityError::limit(
2228                "cryptographic key tombstones",
2229                projected_tombstones,
2230                MAX_HISTORY_PAGE_EVENTS,
2231            ));
2232        }
2233        if previous.retired_suites.len() >= MAX_HISTORY_PAGE_EVENTS {
2234            return Err(IdentityError::limit(
2235                "retired cryptographic suites",
2236                previous.retired_suites.len().saturating_add(1),
2237                MAX_HISTORY_PAGE_EVENTS,
2238            ));
2239        }
2240        if migration.bindings().len() != self.active_controllers.len()
2241            || begin.proofs().as_slice().len() != self.active_controllers.len()
2242        {
2243            return Err(IdentityError::InvalidRelationship {
2244                resource: "complete controller migration binding set",
2245            });
2246        }
2247        let signed_message = migration_id.to_canonical_bytes()?;
2248        for ((controller, binding), proof) in self
2249            .active_controllers
2250            .iter()
2251            .zip(migration.bindings())
2252            .zip(begin.proofs().as_slice())
2253        {
2254            if binding.controller_id() != controller.id()
2255                || proof.controller_id() != controller.id()
2256                || proof.migration_id() != migration_id
2257            {
2258                return Err(IdentityError::InvalidRelationship {
2259                    resource: "migration controller proof coverage",
2260                });
2261            }
2262            if migration_key_reuses_retained_material(
2263                binding.new_signing_key(),
2264                &previous,
2265                &self.active_controllers,
2266                &self.revoked_controllers,
2267                &self.devices,
2268            ) {
2269                return Err(IdentityError::DuplicateSigningKey);
2270            }
2271            let current_keys = self.verification_keys(controller.id())?;
2272            if current_keys.len() != 1 || binding.old_key_id() != current_keys[0].controller_key_id
2273            {
2274                return Err(IdentityError::InvalidRelationship {
2275                    resource: "migration old controller key",
2276                });
2277            }
2278            crate::verifier::verify_algorithm_signature(
2279                current_keys[0].algorithm_code,
2280                &current_keys[0].public_key,
2281                proof.old_key_signature(),
2282                &signed_message,
2283            )?;
2284            crate::verifier::verify_algorithm_signature(
2285                binding.new_signing_key().algorithm_code(),
2286                binding.new_signing_key().as_bytes(),
2287                proof.new_key_signature(),
2288                &signed_message,
2289            )?;
2290        }
2291        self.crypto = CryptoProjection::Candidate {
2292            previous,
2293            migration: migration.clone(),
2294            begin_event_id: event_id,
2295        };
2296        self.lifecycle = ProjectionLifecycle::MigrationPending;
2297        Ok(())
2298    }

2333        &mut self,
2334        retire: &crate::RetireCryptoSuite,
2335        resulting_epoch: Epoch,
2336    ) -> Result<(), IdentityError> {
2337        match (self.crypto.clone(), retire.mode()) {
2338            (
2339                CryptoProjection::Candidate {
2340                    previous,
2341                    migration,
2342                    begin_event_id,
2343                },
2344                crate::RetireCryptoSuiteMode::AbortCandidate,
2345            ) => {
2346                if retire.migration_id() != migration.crypto_migration_id()?
2347                    || retire.phase_event_id() != begin_event_id
2348                    || retire.successor_account_id().is_some()
2349                {
2350                    return Err(IdentityError::InvalidRelationship {
2351                        resource: "abort candidate migration compare-and-set",
2352                    });
2353                }
2354                self.crypto = CryptoProjection::Stable(previous);
2355                self.lifecycle = ProjectionLifecycle::Active;
2356                Ok(())
2357            }
2358            (
2359                CryptoProjection::Dual {
2360                    previous,
2361                    migration,
2362                    activation_event_id,
2363                    ..
2364                },
2365                crate::RetireCryptoSuiteMode::RetirePrevious,
2366            ) => {
2367                if retire.migration_id() != migration.crypto_migration_id()?
2368                    || retire.phase_event_id() != activation_event_id
2369                    || retire.successor_account_id() != migration.successor_account_id()
2370                {
2371                    return Err(IdentityError::InvalidRelationship {
2372                        resource: "retire previous suite compare-and-set",
2373                    });
2374                }
2375                let migrated_keys = migration
2376                    .bindings()
2377                    .iter()
2378                    .map(|binding| MigrationKey {
2379                        controller_id: binding.controller_id(),
2380                        key: binding.new_signing_key().clone(),
2381                    })
2382                    .collect();
2383                let mut retired_suites = previous.retired_suites.clone();
2384                retired_suites.push(RetiredCryptoSuite {
2385                    suite_id: previous.suite.crypto_suite_id()?,
2386                    retired_at: resulting_epoch,
2387                });
2388                let mut key_tombstones = previous.key_tombstones.clone();
2389                key_tombstones.extend(stable_suite_keys(
2390                    &previous,
2391                    &self.active_controllers,
2392                    &self.revoked_controllers,
2393                )?);
2394                key_tombstones.sort_unstable_by(|left, right| {
2395                    left.algorithm_code()
2396                        .cmp(&right.algorithm_code())
2397                        .then_with(|| left.as_bytes().cmp(right.as_bytes()))
2398                });
2399                key_tombstones.dedup_by(|left, right| {
2400                    left.algorithm_code() == right.algorithm_code()
2401                        && left.as_bytes() == right.as_bytes()
2402                });
2403                self.crypto = CryptoProjection::Stable(StableCrypto {
2404                    suite: migration.to_suite().clone(),
2405                    migrated_keys,
2406                    retired_suites,
2407                    key_tombstones,
2408                });
2409                self.lifecycle = if retire.successor_account_id().is_some() {
2410                    self.retirement = Some(RetirementProjection::CryptoMigration {
2411                        migration_id: retire.migration_id(),
2412                        successor_account_id: retire
2413                            .successor_account_id()
2414                            .ok_or(IdentityError::StorageCorruption)?,
2415                        retired_at: resulting_epoch,
2416                    });
2417                    ProjectionLifecycle::Retired
2418                } else {
2419                    ProjectionLifecycle::Active
2420                };
2421                Ok(())
2422            }
2423            _ => Err(IdentityError::InvalidRelationship {
2424                resource: "cryptographic suite retirement phase",
2425            }),
2426        }
2427    }

2673fn project_transition(
2674    pre_state: &AccountState,
2675    event: &AuthorizedEvent,
2676    event_id: EventId,
2677) -> Result<(LineageEntry, AccountState), IdentityError> {
2678    let mut projected = pre_state.clone();
2679    projected.lineage.clear();
2680    projected.lineage_bytes = 0;
2681    if projected.fork.is_some()
2682        && matches!(event.body().operation(), AccountOperation::ResolveFork(_))
2683    {
2684        projected.apply_fork_resolution(event, event_id)?;
2685    } else {
2686        projected.fork = None;
2687        projected.apply_new_linear(event, event_id)?;
2688    }
2689    let transition = projected
2690        .lineage
2691        .last()
2692        .ok_or(IdentityError::StorageCorruption)?;
2693    let transition = LineageEntry {
2694        pre_state: transition.pre_state.clone(),
2695        authority_state: transition.authority_state.clone(),
2696        expected_epoch: transition.expected_epoch,
2697        event: transition.event.clone(),
2698    };
2699    Ok((transition, projected.detached_snapshot()))
2700}

2761fn account_state_evidence_bytes(
2762    state: &AccountState,
2763    depth: usize,
2764) -> Result<usize, IdentityError> {
2765    if depth > MAX_HISTORY_PAGE_EVENTS {
2766        return Err(IdentityError::limit(
2767            "nested account projection evidence",
2768            depth,
2769            MAX_HISTORY_PAGE_EVENTS,
2770        ));
2771    }
2772    let mut total = semantic_projection_bytes(state)?;
2773    if let Some(fork) = &state.fork {
2774        checked_evidence_add(
2775            &mut total,
2776            account_state_evidence_bytes(&fork.common_state, depth.saturating_add(1))?,
2777        )?;
2778        checked_evidence_add(&mut total, fork.common_ancestor.to_canonical_bytes()?.len())?;
2779        checked_evidence_add(
2780            &mut total,
2781            fork.conflict_predecessors.to_canonical_bytes()?.len(),
2782        )?;
2783        for branch in &fork.branches {
2784            checked_evidence_add(
2785                &mut total,
2786                account_state_evidence_bytes(&branch.projected_state, depth.saturating_add(1))?,
2787            )?;
2788            for transition in &branch.transitions {
2789                checked_evidence_add(&mut total, transition.event.to_canonical_bytes()?.len())?;
2790                checked_evidence_add(
2791                    &mut total,
2792                    account_state_evidence_bytes(&transition.pre_state, depth.saturating_add(1))?,
2793                )?;
2794                if let Some(authority) = &transition.authority_state {
2795                    checked_evidence_add(
2796                        &mut total,
2797                        account_state_evidence_bytes(authority, depth.saturating_add(1))?,
2798                    )?;
2799                }
2800            }
2801        }
2802    }
2803    Ok(total)
2804}

2806fn sort_fork_branches(branches: &mut Vec<ForkBranch>) -> Result<(), IdentityError> {
2807    let mut identified = Vec::with_capacity(branches.len());
2808    for branch in branches.drain(..) {
2809        identified.push((fork_branch_event_id(&branch)?, branch));

2812    branches.extend(identified.into_iter().map(|(_, branch)| branch));
2813    Ok(())
