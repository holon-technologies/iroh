5/// Kind of algorithm registry entry that failed validation.
6#[derive(Debug, Clone, Copy, PartialEq, Eq)]
7#[non_exhaustive]
8pub enum AlgorithmKind {
9    /// Cryptographic hash algorithm.
10    Hash,
11    /// Digital-signature algorithm.
12    Signature,
13    /// Key-agreement algorithm.
14    Agreement,
15    /// Key-derivation algorithm.
16    Kdf,
17    /// Authenticated-encryption algorithm.
18    Aead,
19}
20
21impl fmt::Display for AlgorithmKind {
22    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
23        let name = match self {
24            Self::Hash => "hash",
25            Self::Signature => "signature",
26            Self::Agreement => "agreement",
27            Self::Kdf => "key derivation",
28            Self::Aead => "authenticated encryption",
29        };
30        formatter.write_str(name)
31    }
32}

34/// Error returned by identity protocol validation and foundational codecs.
35#[derive(Debug, Clone, PartialEq, Eq, thiserror::Error)]
36#[non_exhaustive]
37pub enum IdentityError {
38    /// Bytes are not a well-formed v1 protocol object.
39    #[error("invalid canonical protocol encoding")]
40    InvalidEncoding,
41    /// The encoded object uses a protocol version this implementation cannot verify.
42    #[error("unsupported protocol version {version}")]
43    UnsupportedVersion {
44        /// Unsupported wire version.
45        version: u16,
46    },
47    /// The encoded object uses an unknown cryptographic algorithm codepoint.
48    #[error("unsupported {kind} algorithm codepoint {code}")]
49    UnsupportedAlgorithm {
50        /// Registry containing the codepoint.
51        kind: AlgorithmKind,
52        /// Unsupported wire codepoint.
53        code: u16,
54    },
55    /// A wire codepoint is intentionally reserved and cannot be accepted as state.
56    #[error("reserved {registry} codepoint {code}")]
57    ReservedCodepoint {
58        /// Registry containing the reserved value.
59        registry: &'static str,
60        /// Reserved codepoint encountered on the wire.
61        code: u16,
62    },
63    /// A wire codepoint is unknown in a closed v1 registry.
64    #[error("unsupported {registry} codepoint {code}")]
65    UnsupportedCodepoint {
66        /// Registry containing the unknown value.
67        registry: &'static str,
68        /// Unknown codepoint encountered on the wire.
69        code: u16,
70    },
71    /// A public key is malformed or forbidden by its algorithm profile.
72    #[error("invalid {kind} public key")]
73    InvalidPublicKey {
74        /// Registry containing the key algorithm.
75        kind: AlgorithmKind,
76    },
77    /// Extension code zero is reserved for the absence of an extension.
78    #[error("invalid extension code {code}")]
79    InvalidExtensionCode {
80        /// Invalid extension code.
81        code: u32,
82    },
83    /// An extension code appears more than once in one object.
84    #[error("duplicate extension code {code}")]
85    DuplicateExtension {
86        /// Repeated extension code.
87        code: u32,
88    },
89    /// A critical extension is not understood by the validating object.
90    #[error("unknown critical extension code {code}")]
91    UnknownCriticalExtension {
92        /// Unknown critical extension code.
93        code: u32,
94    },
95    /// Bytes decode but are not the unique canonical representation.
96    #[error("non-canonical protocol encoding")]
97    NonCanonical,
98    /// A named input or collection exceeds a protocol resource bound.
99    #[error("{resource} contains {actual} items or bytes, maximum is {maximum}")]
100    LimitExceeded {
101        /// Name of the bounded resource.
102        resource: &'static str,
103        /// Observed item or byte count.
104        actual: usize,
105        /// Maximum accepted item or byte count.
106        maximum: usize,
107    },
108    /// A collection required by the schema is empty.
109    #[error("{resource} must not be empty")]
110    EmptyCollection {
111        /// Name of the empty resource.
112        resource: &'static str,
113    },
114    /// A set-like collection contains the same semantic element more than once.
115    #[error("{resource} contains a duplicate element")]
116    DuplicateElement {
117        /// Name of the duplicate-bearing resource.
118        resource: &'static str,
119    },
120    /// A typed content identifier is malformed or inconsistent with its body.
121    #[error("invalid {resource} identifier")]
122    InvalidIdentifier {
123        /// Identifier class.
124        resource: &'static str,
125    },
126    /// Two otherwise valid fields have a forbidden relationship.
127    #[error("invalid {resource} relationship")]
128    InvalidRelationship {
129        /// Relationship class.
130        resource: &'static str,
131    },
132    /// A schema field that must be nonzero was zero.
133    #[error("{resource} must be nonzero")]
134    ZeroValue {
135        /// Nonzero field class.
136        resource: &'static str,
137    },
138    /// A threshold cannot be met by its eligible authority set.
139    #[error("authorization threshold is not satisfiable")]
140    UnsatisfiableThreshold,
141    /// One public signing key was assigned to multiple active authorities.
142    #[error("duplicate active signing key")]
143    DuplicateSigningKey,
144    /// A control, recovery, or provider policy is structurally invalid.
145    #[error("invalid {resource} policy")]
146    InvalidPolicy {
147        /// Policy class.
148        resource: &'static str,
149    },
150    /// A capability grant is structurally invalid.
151    #[error("invalid capability: {reason}")]
152    InvalidCapability {
153        /// Stable validation reason.
154        reason: &'static str,
155    },
156    /// A capability delegation is structurally invalid.
157    #[error("invalid delegation: {reason}")]
158    InvalidDelegation {
159        /// Stable validation reason.
160        reason: &'static str,
161    },
162    /// A protocol feature is intentionally not accepted by v1 policy semantics.
163    #[error("unsupported v1 policy feature: {feature}")]
164    UnsupportedPolicyFeature {
165        /// Feature deliberately unavailable in v1.
166        feature: &'static str,
167    },
168    /// A cryptographic signature did not verify under its declared key and suite.
169    #[error("invalid protocol signature")]
170    InvalidSignature,
171    /// A group-key wrap names a suite other than the fixed v1 KEM/KDF/AEAD profile.
172    #[error("unsupported group-key wrap cryptographic suite")]
173    UnsupportedKeyWrapSuite,
174    /// Group-key unwrap failed authentication.
175    ///
176    /// This deliberately does not distinguish a wrong key, modified ciphertext, or
177    /// modified associated data. Retrying the same inputs cannot succeed.
178    #[error("group-key wrap authentication failed")]
179    KeyWrapAuthenticationFailed,
180    /// Private artifact authentication failed.
181    ///
182    /// This deliberately does not distinguish a wrong key or passphrase, modified ciphertext,
183    /// modified parameters, or modified associated context.
184    #[error("private artifact authentication failed")]
185    PrivateArtifactAuthenticationFailed,
186    /// The operating system could not provide cryptographic entropy.
187    ///
188    /// A caller may retry after the platform entropy source becomes available.
189    #[error("operating-system cryptographic entropy is unavailable")]
190    EntropyUnavailable,
191    /// An approval names no controller in the relevant pre-state.
192    #[error("unknown account controller")]
193    UnknownController,
194    /// A known controller is outside the operation rule's selector or immutable scope.
195    #[error("controller is ineligible for this operation")]
196    IneligibleController,
197    /// A terminally removed controller attempted to authorize a later transition.
198    #[error("controller has been revoked")]
199    RevokedController,
200    /// The same authority was counted more than once in one authorization set.
201    #[error("duplicate authorization signer")]
202    DuplicateSigner,
203    /// Valid eligible signatures do not meet the active policy threshold.
204    #[error("account authorization denied")]
205    AuthorizationDenied,
206    /// An event sequence does not advance from its predecessor exactly once.
207    #[error("invalid account event sequence")]
208    InvalidSequence,
209    /// A possible branch predates the bounded in-memory lineage cache and needs durable replay.
210    #[error("historical account state is required for event sequence {sequence}")]
211    HistoricalStateRequired {
212        /// Sequence whose authenticated pre-state must be loaded from durable lineage.
213        sequence: u64,
214    },
215    /// An event or application envelope names the wrong security epoch.
216    #[error("invalid account epoch")]
217    InvalidEpoch,
218    /// An event's predecessor reference does not match the complete current head set.
219    #[error("invalid account event predecessor")]
220    InvalidPredecessor,
221    /// A nested object belongs to a different stable account.
222    #[error("account identifier mismatch")]
223    AccountMismatch,
224    /// Conflicting valid account-control bodies were retained as a fork.
225    #[error("account fork detected")]
226    ForkDetected,
227    /// The requested transition is forbidden while the account remains forked.
228    #[error("account is forked")]
229    AccountForked,
230    /// The requested transition conflicts with an authoritative pending recovery.
231    #[error("account recovery is pending")]
232    RecoveryPending,
233    /// The account is terminally retired.
234    #[error("account is retired")]
235    AccountRetired,
236    /// A newer protocol major is authoritative, so this v1 client is read-only.
237    #[error("account is read-only after a protocol-major upgrade")]
238    ProtocolUpgradeReadOnly,
239    /// A rule requires signed provider freshness that is not available.
240    #[error("required freshness evidence is unavailable")]
241    FreshnessUnavailable,
242    /// Signed provider evidence is older than the applicable freshness bound.
243    #[error("freshness evidence is stale")]
244    StaleEvidence,
245    /// A delayed transition has not reached its signed provider-observed deadline.
246    #[error("required authorization delay has not elapsed")]
247    DelayNotElapsed,
248    /// Historical evidence names a policy revision other than the applicable pre-state revision.
249    #[error("policy version mismatch")]
250    PolicyVersionMismatch,
251    /// A device has no authorization in the supplied account state.
252    #[error("device is not authorized")]
253    DeviceNotAuthorized,
254    /// A temporarily suspended device attempted an authorized action.
255    #[error("device is suspended")]
256    DeviceSuspended,
257    /// A terminally revoked device attempted an authorized action.
258    #[error("device is revoked")]
259    DeviceRevoked,
260    /// An atomic state write raced with a different account revision.
261    #[error("stale account-store revision")]
262    StaleRevision,
263    /// Durable bytes or indices are inconsistent with their authenticated state.
264    #[error("identity store corruption")]
265    StorageCorruption,
266    /// An owned operation was explicitly cancelled before completion.
267    #[error("identity operation cancelled")]
268    Cancelled,
269    /// A bounded retry policy exhausted all permitted attempts.
270    #[error("identity operation exhausted its retry budget")]
271    RetryExhausted,
272    /// Bounded concurrency or queue capacity is currently exhausted.
273    #[error("identity resource is busy")]
274    ResourceBusy,
275    /// A configured transparency provider could not be reached.
276    #[error("transparency provider is unavailable")]
277    ProviderUnavailable,
278    /// A configured transparency-provider request exceeded its bounded deadline.
279    #[error("transparency provider request timed out")]
280    ProviderTimeout,
281    /// A transparency provider rejected work because its abuse-control limit was reached.
282    #[error("transparency provider rate limit exceeded")]
283    ProviderRateLimited,
284    /// A sealed provider generation released the requested payload to its verified archive.
285    #[error("provider archive is required for this released history")]
286    ProviderArchiveRequired,
287    /// Protected application writes are blocked pending mandatory key rotation.
288    #[error("protected writes are blocked pending group-key rotation")]
289    ProtectedWritesBlocked,
290    /// A Merkle, consistency, non-membership, or lineage proof is invalid.
291    #[error("invalid transparency proof")]
292    InvalidProof,
293    /// Two signed provider heads prove inconsistent statements for one tree size.
294    #[error("transparency provider equivocation")]
295    ProviderEquivocation,
296    /// A provider returned an older tree size or observation than a previously verified head.
297    #[error("transparency provider rollback")]
298    ProviderRollback,
299    /// A protocol counter or time calculation overflowed.
300    #[error("{resource} arithmetic overflow")]
301    ArithmeticOverflow {
302        /// Counter or unit that overflowed.
303        resource: &'static str,
304    },
305}

307impl IdentityError {
308    pub(crate) const fn unsupported_algorithm(kind: AlgorithmKind, code: u16) -> Self {
309        Self::UnsupportedAlgorithm { kind, code }
310    }
311
312    pub(crate) const fn limit(resource: &'static str, actual: usize, maximum: usize) -> Self {
313        Self::LimitExceeded {
314            resource,
315            actual,
316            maximum,
317        }
318    }
