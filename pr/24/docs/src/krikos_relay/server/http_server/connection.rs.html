100/// Requires a future to complete before the specified duration elapses, unless the timeout is cleared.
101///
102/// If the future completes before the duration has elapsed, then the completed value is returned.
103/// Otherwise, an error is returned and the future is canceled.
104///
105/// If `clear_timeout` is triggered, the timeout is cleared and the future is always run to completion.
106pub(super) async fn clearable_timeout<F: Future>(
107    timeout: Duration,
108    clear_timeout: Arc<Notify>,
109    establishment_lease: Arc<Mutex<Option<EstablishmentLease>>>,
110    fut: F,
111) -> Result<F::Output, Elapsed> {
112    tokio::pin!(fut);
113    let timeout = MaybeFuture::Some(tokio::time::sleep(timeout));
114    tokio::pin!(timeout);
115    loop {
116        tokio::select! {
117            biased;
118            res = &mut fut => {
119                return Ok(res);
120            }
121            _ = clear_timeout.notified() => {
122                release_establishment_lease(&establishment_lease);
123                timeout.as_mut().set_none();
124            },
125            _ = &mut timeout => {
126                return Err(Elapsed);
127            }
128        }
129    }
130}

132pub(super) fn release_establishment_lease(lease: &Mutex<Option<EstablishmentLease>>) {
133    let mut lease = match lease.lock() {
134        Ok(lease) => lease,
135        Err(poisoned) => poisoned.into_inner(),
136    };
137    drop(lease.take());
138}
