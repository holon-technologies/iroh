92/// A structured runtime observation.
93#[derive(Clone, Debug, Deserialize, Eq, PartialEq, Serialize)]
94pub struct TraceEvent {
95    /// Event schema version.
96    pub schema_version: u16,
97    /// Global event sequence.
98    pub sequence: TraceSequence,
99    /// Optional causal predecessor in the same global trace.
100    #[serde(default, skip_serializing_if = "Option::is_none")]
101    pub causal_parent: Option<TraceSequence>,
102    /// Optional behavioral decision responsible for this event.
103    #[serde(default, skip_serializing_if = "Option::is_none")]
104    pub decision: Option<DecisionId>,
105    /// Nanoseconds elapsed on the run's virtual monotonic timeline.
106    pub virtual_time_nanos: u64,
107    /// Related runtime/protocol entities.
108    pub context: TraceContext,
109    /// Event payload.
110    pub event: TraceEventKind,
111}
112
113impl TraceEvent {
114    /// Creates an event using the current schema version.
115    pub fn new(
116        sequence: TraceSequence,
117        virtual_time_nanos: u64,
118        context: TraceContext,
119        event: TraceEventKind,
120    ) -> Self {
121        Self {
122            schema_version: TRACE_SCHEMA_VERSION,
123            sequence,
124            causal_parent: None,
125            decision: None,
126            virtual_time_nanos,
127            context,
128            event,
129        }
130    }
131
132    /// Attaches a stable causal predecessor.
133    pub const fn with_causal_parent(mut self, parent: TraceSequence) -> Self {
134        self.causal_parent = Some(parent);
135        self
136    }
137
138    /// Attaches the behavioral decision that selected this event.
139    pub const fn with_decision(mut self, decision: DecisionId) -> Self {
140        self.decision = Some(decision);
141        self
142    }
143}

380/// Receives structured trace observations.
381pub trait TraceSink: fmt::Debug + Send + Sync + 'static {
382    /// Records one observation.
383    fn record(&self, event: TraceEvent) -> Result<(), TraceSinkError>;
384}
385
386/// Assigns one global sequence space to events sent to a trace sink.
387#[derive(Debug)]
388pub struct TraceRecorder {
389    sequences: crate::IdAllocator<TraceSequence>,
390    sink: Arc<dyn TraceSink>,
391    failure: Mutex<Option<TraceRecordError>>,
392}
393
394impl TraceRecorder {
395    /// Creates a recorder for `sink`.
396    pub fn new(sink: Arc<dyn TraceSink>) -> Self {
397        Self {
398            sequences: crate::IdAllocator::default(),
399            sink,
400            failure: Mutex::new(None),
401        }
402    }
403
404    /// Assigns a sequence number and records one event.
405    pub fn record(
406        &self,
407        virtual_time_nanos: u64,
408        context: TraceContext,
409        event: TraceEventKind,
410    ) -> Result<TraceSequence, TraceRecordError> {
411        let result: Result<TraceSequence, TraceRecordError> = (|| {
412            let sequence = self
413                .sequences
414                .allocate()
415                .map_err(|_| TraceRecordError::IdExhausted)?;
416            self.sink
417                .record(TraceEvent::new(
418                    sequence,
419                    virtual_time_nanos,
420                    context,
421                    event,
422                ))
423                .map_err(TraceRecordError::Sink)?;
424            Ok(sequence)
425        })();
426        if let Err(error) = &result {
427            self.latch(error.clone());
428        }
429        result
430    }
431
432    /// Takes the first trace failure that has not yet been handled.
433    pub fn take_failure(&self) -> Option<TraceRecordError> {
434        self.failure
435            .lock()
436            .expect("trace failure lock poisoned")
437            .take()
438    }
439
440    fn latch(&self, error: TraceRecordError) {
441        let mut failure = self.failure.lock().expect("trace failure lock poisoned");
442        if failure.is_none() {
443            *failure = Some(error);
444        }
445    }
446}
447
448/// Failure while assigning or retaining a trace event.
449#[derive(Clone, Debug, Eq, PartialEq)]
450pub enum TraceRecordError {
451    /// The global trace sequence was exhausted.
452    IdExhausted,
453    /// The configured sink rejected the event.
454    Sink(TraceSinkError),
455}
456
457impl fmt::Display for TraceRecordError {
458    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
459        match self {
460            Self::IdExhausted => f.write_str("trace sequence space exhausted"),
461            Self::Sink(err) => write!(f, "trace sink failed: {err}"),
462        }
463    }
464}
465
466impl std::error::Error for TraceRecordError {
467    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
468        match self {
469            Self::Sink(err) => Some(err),
470            Self::IdExhausted => None,
471        }
472    }
473}
