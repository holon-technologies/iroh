57///

61pub struct HttpsConfig {

68    /// Strategy used to obtain TLS certificates.
69    pub cert_mode: CertMode,

75    /// Whether to use the Let's Encrypt production endpoint instead of staging.

77    /// Only applies when [`Self::cert_mode`] is [`CertMode::LetsEncrypt`]. When

82/// The HTTP(S) server part of krikos-dns-server
83#[derive(Debug)]
84pub(crate) struct HttpServer {
85    tasks: JoinSet<std::io::Result<()>>,
86    http_addr: Option<SocketAddr>,
87    https_addr: Option<SocketAddr>,
88    cancel: CancellationToken,
89    shutdown_timeout: std::time::Duration,
90}
91
92impl HttpServer {
93    /// Spawn the server
94    pub(crate) async fn spawn(
95        http_config: Option<HttpConfig>,
96        https_config: Option<HttpsConfig>,
97        rate_limit_config: RateLimitConfig,
98        state: AppState,
99        cert_cache_dir: PathBuf,
100        ingress: IngressPolicy,
101    ) -> Result<HttpServer> {
102        if http_config.is_none() && https_config.is_none() {
103            bail_any!("Either http or https config is required");
104        }
105
106        let admission = Arc::new(AdmissionControl::new(
107            ingress.clone(),
108            state.metrics.clone(),
109            Instant::now(),
110        ));
111        let app = create_app(state, &rate_limit_config, admission.clone(), &ingress);
112
113        let mut tasks = JoinSet::new();
114        let cancel = CancellationToken::new();
115
116        // launch http
117        let http_addr = if let Some(config) = http_config {
118            let bind_addr = SocketAddr::new(
119                config.bind_addr.unwrap_or(Ipv4Addr::UNSPECIFIED.into()),
120                config.port,
121            );
122            let app = app.clone();
123            let listener = TcpListener::bind(bind_addr).await.anyerr()?;
124            let bound_addr = listener.local_addr().anyerr()?;
125            let fut = transport::serve_listener(
126                listener,
127                app,
128                None,
129                admission.clone(),
130                ingress.max_http2_streams_per_connection,
131                cancel.clone(),
132            );
133            info!("HTTP server listening on {bind_addr}");
134            tasks.spawn(fut);
135            Some(bound_addr)
136        } else {
137            None
138        };
139
140        // launch https
141        let https_addr = if let Some(config) = https_config {
142            let bind_addr = SocketAddr::new(
143                config.bind_addr.unwrap_or(Ipv4Addr::UNSPECIFIED.into()),
144                config.port,
145            );
146            let tls_runtime = {
147                tokio::fs::create_dir_all(&cert_cache_dir)
148                    .await
149                    .with_std_context(|_| {
150                        format!(
151                            "failed to create cert cache dir at {}",
152                            cert_cache_dir.display()
153                        )
154                    })?;
155                config
156                    .cert_mode
157                    .build(
158                        config.domains,
159                        cert_cache_dir,
160                        config.letsencrypt_contact,
161                        config.letsencrypt_prod.unwrap_or(false),
162                    )
163                    .await?
164            };
165            let listener = TcpListener::bind(bind_addr).await.anyerr()?;
166            let bound_addr = listener.local_addr().anyerr()?;
167            let fut = transport::serve_listener(
168                listener,
169                app,
170                Some(tls_runtime.acceptor),
171                admission,
172                ingress.max_http2_streams_per_connection,
173                cancel.clone(),
174            );
175            info!("HTTPS server listening on {bind_addr}");
176            tasks.spawn(fut);
177            if let Some(maintenance) = tls_runtime.maintenance {
178                tasks.spawn(maintenance);
179            }
180            Some(bound_addr)
181        } else {
182            None
183        };
184
185        Ok(HttpServer {
186            tasks,
187            http_addr,
188            https_addr,
189            cancel,
190            shutdown_timeout: ingress.shutdown_timeout,
191        })
192    }

195    pub(crate) fn http_addr(&self) -> Option<SocketAddr> {
196        self.http_addr

201        self.https_addr

204    /// Signals every HTTP listener and connection task to stop.
205    pub(crate) fn start_shutdown(&self) {
206        self.cancel.cancel();
207    }
208
209    /// Shutdown the server and wait for all tasks to complete.
210    pub(crate) async fn shutdown(&mut self) -> Result<()> {
211        self.start_shutdown();
212        match tokio::time::timeout(self.shutdown_timeout, self.run_tasks_until_done()).await {
213            Ok(result) => result,
214            Err(_) => {
215                self.tasks.abort_all();
216                while self.tasks.join_next().await.is_some() {}
217                Err(anyerr!(
218                    "HTTP shutdown exceeded {:?}",
219                    self.shutdown_timeout
220                ))
221            }
222        }
223    }
224
225    /// Wait for all tasks to complete.
226    ///
227    /// Runs forever unless tasks fail.
228    pub(crate) async fn run_until_done(&mut self) -> Result<()> {
229        self.run_tasks_until_done().await
230    }
231
232    async fn run_tasks_until_done(&mut self) -> Result<()> {
233        let mut final_res: Result<()> = Ok(());
234        while let Some(res) = self.tasks.join_next().await {
235            match res {
236                Ok(Ok(())) if self.cancel.is_cancelled() => {}
237                Ok(Ok(())) => {
238                    warn!("HTTP supervisor task exited unexpectedly");
239                    if final_res.is_ok() {
240                        final_res = Err(anyerr!("HTTP supervisor task exited unexpectedly"));
241                    }
242                    self.cancel.cancel();
243                }
244                Err(err) if err.is_cancelled() => {}
245                Ok(Err(err)) => {
246                    warn!(?err, "task failed");
247                    if final_res.is_ok() {
248                        final_res = Err(anyerr!(err, "task"));
249                    }
250                    self.cancel.cancel();
251                }
252                Err(err) => {
253                    warn!(?err, "task panicked");
254                    if final_res.is_ok() {
255                        final_res = Err(anyerr!(err, "join"));
256                    }
257                    self.cancel.cancel();
258                }
259            }
260        }
261        final_res
262    }
263}

286async fn hold_for_admission_test(headers: http::HeaderMap) -> impl IntoResponse {
287    const MAX_HOLD_MILLIS: u64 = 10 * 60 * 1_000;
288    let duration = headers
289        .get("x-iroh-test-hold-millis")
290        .and_then(|value| value.to_str().ok())
291        .and_then(|value| value.parse::<u64>().ok())
292        .filter(|duration| *duration > 0 && *duration <= MAX_HOLD_MILLIS);
293    let Some(duration) = duration else {
294        return StatusCode::BAD_REQUEST;
295    };
296    tokio::time::sleep(Duration::from_millis(duration)).await;
297    StatusCode::OK
298}

300pub(crate) fn create_app(
301    state: AppState,
302    rate_limit_config: &RateLimitConfig,
303    admission: Arc<AdmissionControl>,
304    ingress: &IngressPolicy,
305) -> Router {
306    // configure cors middleware
307    let cors = CorsLayer::new()
308        // allow `GET` and `POST` when accessing the resource
309        .allow_methods([Method::GET, Method::POST, Method::PUT])
310        // allow requests from any origin
311        .allow_origin(cors::Any);
312
313    // configure tracing middleware
314    let trace = TraceLayer::new_for_http().make_span_with(|request: &http::Request<_>| {
315        let conn_info = request
316            .extensions()
317            .get::<ConnectInfo<SocketAddr>>()
318            .expect("connectinfo extension to be present");
319        let span = span!(
320        Level::DEBUG,
321            "http_request",
322            method = ?request.method(),
323            uri = ?request.uri(),
324            src = %conn_info.0,
325            );
326        span
327    });
328
329    // configure rate limiting middleware
330    let rate_limit = rate_limiting::create(rate_limit_config, ingress, state.metrics.clone());
331
332    // configure routes
333    //
334    // only the pkarr::put route gets a rate limit
335    let router = Router::new()
336        .route("/dns-query", get(doh::get).post(doh::post))
337        .route(
338            "/pkarr/{key}",
339            if let Some(rate_limit) = rate_limit {
340                get(pkarr::get).put(
341                    pkarr::put
342                        .layer(DefaultBodyLimit::max(
343                            krikos_dns::pkarr::SignedPacket::MAX_RELAY_PAYLOAD_BYTES,
344                        ))
345                        .layer(middleware::from_fn_with_state(
346                            rate_limit,
347                            rate_limiting::middleware,
348                        )),
349                )
350            } else {
351                get(pkarr::get).put(pkarr::put.layer(DefaultBodyLimit::max(
352                    krikos_dns::pkarr::SignedPacket::MAX_RELAY_PAYLOAD_BYTES,
353                )))
354            },
355        )
356        // Deprecated: use /healthz instead
357        .route("/healthcheck", get(|| async { "OK" }))
358        .route("/healthz", get(healthz))
359        .route("/", get(|| async { "Hi!" }))
360        .with_state(state.clone());
361    #[cfg(feature = "test-utils")]
362    let router = router.route("/__iroh_test/hold", get(hold_for_admission_test));
363
364    // configure app
365    router
366        .layer(DefaultBodyLimit::max(ingress.max_http_body_bytes.get()))
367        .layer(cors)
368        .layer(trace)
369        .route_layer(middleware::from_fn_with_state(state, metrics_middleware))
370        .route_layer(middleware::from_fn_with_state(
371            admission,
372            admission_middleware,
373        ))
374}

376async fn admission_middleware(
377    State(admission): State<Arc<AdmissionControl>>,
378    req: Request<Body>,
379    next: Next,
380) -> Response {
381    match admission.try_request() {
382        RequestAdmission::Accepted(lease) => {
383            let _lease = lease;
384            next.run(req).await
385        }
386        RequestAdmission::CapacityFull => (
387            StatusCode::SERVICE_UNAVAILABLE,
388            [(header::RETRY_AFTER, "1")],
389        )
390            .into_response(),
391    }
392}

394/// Record request metrics.
395// TODO:
396// * Request duration would be much better tracked as a histogram.
397// * It would be great to attach labels to the metrics, so that the recorded metrics
398// can filter by method etc.
399//
400// See also
401// https://github.com/tokio-rs/axum/blob/main/examples/prometheus-metrics/src/main.rs#L114
402async fn metrics_middleware(
403    State(state): State<AppState>,
404    req: Request,
405    next: Next,
406) -> impl IntoResponse {
407    let start = Instant::now();
408    let response = next.run(req).await;
409    let latency = start.elapsed().as_millis();
410    let status = response.status();
411    state
412        .metrics
413        .http_requests_duration_ms
414        .inc_by(u64::try_from(latency).unwrap_or(u64::MAX));
415    state.metrics.http_requests.inc();
416    if status.is_success() {
417        state.metrics.http_requests_success.inc();
418    } else {
419        state.metrics.http_requests_error.inc();
420    }
421    response
422}

614            other => panic!("expected TXT record, got {other:?}"),
615        };
616        assert_eq!(&txt_data[0][..], format!("relay={RELAY_URL}").as_bytes());
617
618        server.shutdown().await?;
619        Ok(())
620    }

694                rustls::crypto::ring::default_provider(),

696            .with_safe_default_protocol_versions()
697            .unwrap()
698            .with_root_certificates(RootCertStore::empty())
