23/// The default certificate reload interval.
24pub const DEFAULT_CERT_RELOAD_INTERVAL: Duration = Duration::from_secs(60 * 60 * 24);
25const MAX_RELOAD_TLS_FILE_BYTES: usize = 1024 * 1024;

43        &self,

59async fn read_reload_file(path: &Path) -> io::Result<Vec<u8>> {
60    let file = tokio::fs::File::open(path).await?;
61    let capacity = usize::try_from(
62        file.metadata()
63            .await?
64            .len()
65            .min(MAX_RELOAD_TLS_FILE_READ_BYTES),
66    )
67    .map_err(|_| io::Error::new(io::ErrorKind::InvalidData, "TLS file length is invalid"))?;
68    let mut reader = file.take(MAX_RELOAD_TLS_FILE_READ_BYTES);
69    let mut bytes = Vec::with_capacity(capacity);
70    reader.read_to_end(&mut bytes).await?;
71    if bytes.len() > MAX_RELOAD_TLS_FILE_BYTES {
72        return Err(io::Error::new(
73            io::ErrorKind::InvalidData,
74            format!(
75                "TLS reload file {} exceeds {MAX_RELOAD_TLS_FILE_BYTES} bytes",
76                path.display()
77            ),
78        ));
79    }
80    Ok(bytes)
81}

83/// Builds a [`ResolvesServerCert`] that reloads its certificate and key from disk on an interval.
84///
85/// Loads the PEM-encoded certificate chain from `cert_path` and the PEM-encoded private key
86/// from `key_path` using `crypto_provider`'s key provider, then spawns a background task that
87/// re-reads both files every `interval`. The returned resolver hands the most recently loaded
88/// `CertifiedKey` to rustls for each TLS handshake, so certificate rotation takes effect without
89/// restarting the server. See [`DEFAULT_CERT_RELOAD_INTERVAL`] for a sensible default.
90///
91/// The reload task is tied to the returned `Arc` and is aborted when the last reference is
92/// dropped. Reload failures on the interval are silently ignored; the previously loaded
93/// certificate remains in use.
94///
95/// # Errors
96///
97/// Returns an error if the initial certificate or key load fails (for example, the files do not
98/// exist, cannot be read, or cannot be parsed as PEM).
99pub async fn reloading_resolver(
100    crypto_provider: &CryptoProvider,
101    cert_path: PathBuf,
102    key_path: PathBuf,
103    interval: std::time::Duration,
104) -> Result<Arc<dyn ResolvesServerCert>, AnyError> {
105    let key_reader = BoundedPemFileReader::new(key_path);
106    let certs_reader = BoundedPemFileReader::new(cert_path);
107    let loader = CertifiedKeyLoader {
108        key_provider: Dyn(crypto_provider.key_provider),
109        key_reader,
110        certs_reader,
111    };
112    let resolver = ReloadingResolver::init(loader, interval)
113        .await
114        .std_context("cert loading")?;
115    Ok(Arc::new(resolver))
116}

118/// A Certificate resolver that reloads the certificate every interval
119#[derive(Debug)]
120struct ReloadingResolver<Loader: Send + 'static> {
121    /// The inner reloadable value.
122    reloadable: Arc<Reloadable<CertifiedKey, Loader>>,
123    /// The handle to the task that reloads the certificate.
124    _handle: AbortOnDropHandle<()>,
125}
126
127impl<Loader> ReloadingResolver<Loader>
128where
129    Loader: Send + reloadable_state::core::Loader<Value = CertifiedKey> + 'static,
130{
131    /// Perform the initial load and construct the [`ReloadingResolver`].
132    async fn init(loader: Loader, interval: Duration) -> Result<Self, Loader::Error> {
133        let (reloadable, _) = Reloadable::init_load(loader).await?;
134        let reloadable = Arc::new(reloadable);
135
136        let cancel_token = CancellationToken::new();
137
138        // Spawn a task to reload the certificate every interval.
139        let _reloadable = reloadable.clone();
140        let _cancel_token = cancel_token.clone();
141        let _handle = task::spawn(async move {
142            let mut interval = time::interval(interval);
143            loop {
144                tokio::select! {
145                    _ = interval.tick() => {
146                        let _ = _reloadable.reload().await;
147                        tracing::info!("Reloaded the certificate");
148                    },
149                    _ = _cancel_token.cancelled() => {
150                        tracing::trace!("shutting down");
151                        break;
152                    }
153                }
154            }
155        });
156        let _handle = AbortOnDropHandle::new(_handle);
157
158        Ok(Self {
159            reloadable,
160            _handle,
161        })
162    }
163}

165impl<Loader> ResolvesServerCert for ReloadingResolver<Loader>
166where
167    Loader: reloadable_state::core::Loader<Value = CertifiedKey>,
168    Loader: Send,
