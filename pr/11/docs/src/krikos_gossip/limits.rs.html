1//! Resource limits enforced by the gossip protocol and network adapter.
2
3use std::time::Duration;
4
5/// Largest accepted encoded gossip frame.
6pub const MAX_MESSAGE_SIZE: usize = 1024 * 1024;
7/// Largest configured HyParView active view.
8pub const MAX_ACTIVE_VIEW_CAPACITY: usize = 64;
9/// Largest configured HyParView passive view.
10pub const MAX_PASSIVE_VIEW_CAPACITY: usize = 1024;
11/// Largest peer list accepted in a shuffle message.
12pub const MAX_SHUFFLE_PEERS: usize = 128;
13/// Maximum topics joined by one gossip actor.
14pub const MAX_TOPICS: usize = 1024;
15/// Maximum bootstrap peers accepted in one API command.
16pub const MAX_BOOTSTRAP_PEERS: usize = 256;
17/// Maximum subscription buffer requested through the API.
18pub const MAX_SUBSCRIPTION_CAPACITY: usize = 65_536;
19/// Maximum missing messages tracked for Plumtree recovery.
20pub const MAX_PENDING_MESSAGES: usize = 8192;
21/// Maximum peers tried while recovering one missing message.
22pub const MAX_GRAFT_RETRIES: usize = 8;
23/// Maximum message identifiers accepted in one `IHave` message.
24pub const MAX_IHAVE_ENTRIES: usize = 1024;
25/// Maximum duplicate identifiers retained per topic.
26pub const MAX_DUPLICATE_CACHE_ENTRIES: usize = 16_384;
27/// Maximum payloads retained for graft replies per topic.
28pub const MAX_CACHED_MESSAGES: usize = 4096;
29/// Maximum lazy announcements queued for one peer.
30pub const MAX_LAZY_QUEUE_PER_PEER: usize = 1024;
31/// Maximum messages queued while dialing one peer.
32pub const MAX_PENDING_SENDS_PER_PEER: usize = 256;
33/// Maximum simultaneous connection tasks owned by one actor.
34pub const MAX_CONCURRENT_CONNECTION_HANDLERS: usize = 256;
35/// Maximum simultaneous inbound topic streams on one connection.
36pub const MAX_STREAMS_PER_CONNECTION: usize = 1024;
37/// Maximum simultaneous outbound dials owned by one actor.
38pub const MAX_CONCURRENT_DIALS: usize = 256;
39/// Maximum live topic subscriptions owned by one actor.
40pub const MAX_SUBSCRIPTIONS: usize = 4096;
41/// Time allowed for graceful actor shutdown.
42pub const SHUTDOWN_TIMEOUT: Duration = Duration::from_secs(5);
43/// Portion of shutdown reserved for sending topic disconnect messages.
44pub const SHUTDOWN_DRAIN_TIMEOUT: Duration = Duration::from_secs(4);
45
46/// A protocol configuration exceeds a resource limit.
47#[derive(Debug, Clone, Copy, PartialEq, Eq)]
48pub struct ConfigError {
49    field: &'static str,
50    value: usize,
51    min: usize,
52    max: usize,
53}
54
55impl ConfigError {
56    pub(crate) fn new(field: &'static str, value: usize, min: usize, max: usize) -> Self {
57        Self {
58            field,
59            value,
60            min,
61            max,
62        }
63    }
64}
65
66impl std::fmt::Display for ConfigError {
67    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
68        write!(
69            formatter,
70            "{} must be in {}..={}, got {}",
71            self.field, self.min, self.max, self.value
72        )
73    }
74}
75
76impl std::error::Error for ConfigError {}
